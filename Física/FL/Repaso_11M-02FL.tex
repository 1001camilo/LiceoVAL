	% --- PUNTO 3 (Antes Punto 1) ---
	\noindent \textbf{3.} Se cuelga un objeto de $450\text{ g}$ a un resorte de constante $60\text{ N/m}$. Si este empieza a oscilar con una amplitud de $10\text{ cm}$, calcule:
	\begin{enumerate}[label=\alph*), itemsep=0.3cm, leftmargin=0.8cm]
		\item La energía asociada al resorte.\\
		\textcolor{blue}{\textbf{Solución:} $m = 0.45 \text{ kg}$, $k = 60 \text{ N/m}$, $A = 0.10 \text{ m}$.\\
			$E = \frac{1}{2} k \cdot A^2 = \frac{1}{2}(60 \text{ N/m})(0.10 \text{ m})^2 = 30(0.01) = \mathbf{0.30 \text{ J}}$}
		
		\item La velocidad cuando pasa por el punto de equilibrio.\\
		\textcolor{blue}{\textbf{Solución:} $E_k = E_{total} \Rightarrow \frac{1}{2} m \cdot v_{max}^2 = E$\\
			$0.5(0.45 \text{ kg}) \cdot v^2 = 0.30 \text{ J} \Rightarrow 0.225 \cdot v^2 = 0.30 \Rightarrow v = \sqrt{\frac{0.30}{0.225}} \approx \mathbf{1.15 \text{ m/s}}$}
		
		\item Si el sistema se encontrara en reposo, ¿qué valor tendría la elongación? (\textbf{\textit{Pista: Usa la ley de Hooke}})\\
		\textcolor{blue}{\textbf{Solución:} $F_k = F_g \Rightarrow k \cdot x = m \cdot g$\\
			$60 \cdot x = (0.45)(9.8) \Rightarrow 60x = 4.41 \Rightarrow x = \frac{4.41}{60} \approx \mathbf{0.0735 \text{ m}}$ (o $\mathbf{7.35 \text{ cm}}$).}
	\end{enumerate}

	% --- PUNTO 4 (Antes Punto 2) ---
	\noindent \textbf{4.} Un tren de ondas se desplaza cumpliendo $80\text{ ciclos}$ en $4\text{ segundos}$, la longitud de estos ciclos es de $15\text{ cm}$. Calcule:
	\begin{enumerate}[label=\alph*), itemsep=0.3cm, leftmargin=0.8cm]
		\item La frecuencia y el periodo asociados.\\
		\textcolor{blue}{\textbf{Solución:} Frecuencia ($f$) = $\frac{\text{No. de ciclos}}{\text{Tiempo}} = \frac{80}{4} = \mathbf{20 \text{ Hz}}$.\\
			Periodo ($T$) = $\frac{1}{f} = \frac{1}{20} = \mathbf{0.05 \text{ s}}$.}
		
		\item ¿Qué velocidad lleva la onda?\\
		\textcolor{blue}{\textbf{Solución:} Pasamos la longitud a metros: $\lambda = 0.15 \text{ m}$.\\
			$v = \lambda \cdot f = (0.15 \text{ m}) \cdot (20 \text{ Hz}) = \mathbf{3 \text{ m/s}}$ (o $\mathbf{300 \text{ cm/s}}$).}
	\end{enumerate}
