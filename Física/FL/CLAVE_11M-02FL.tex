	% --- PUNTO 1 ---
	\noindent \textbf{1.} Se cuelga un objeto de $600\text{ g}$ a un resorte de constante $55\text{ N/m}$. Si este empieza a oscilar con una amplitud de $8\text{ cm}$, calcule:
	\begin{enumerate}[label=\alph*), itemsep=0.3cm, leftmargin=0.8cm]
		\item La energía asociada al resorte.\\
		\textcolor{blue}{\textbf{Solución:}  $m = 0.6 \text{ kg}$, $k = 55 \text{ N/m}$, $A = 0.08 \text{ m}$.\\
			$E = \frac{1}{2} k \cdot A^2 = \frac{1}{2}(55 \text{ N/m})(0.08 \text{ m})^2 = 27.5(0.0064) = \mathbf{0.176 \text{ J}}$}
		
		\item La velocidad cuando pasa por el punto de equilibrio.\\
		\textcolor{blue}{\textbf{Solución:} En el punto de equilibrio, toda la energía es cinética ($E_k = E_{total}$).\\
			$\frac{1}{2} m \cdot v_{max}^2 = E \Rightarrow 0.5(0.6 \text{ kg}) \cdot v^2 = 0.176 \text{ J} \Rightarrow 0.3 \cdot v^2 = 0.176 \Rightarrow v = \sqrt{\frac{0.176}{0.3}} \approx \mathbf{0.766 \text{ m/s}}$}
		
		\item Si el sistema se encontrara en reposo, ¿qué valor tendría la elongación?(\textbf{\textit{Pista: Usa la ley de hooke}})\\
		\textcolor{blue}{\textbf{Solución:} En equilibrio estático, la fuerza elástica iguala al peso: $F_k = F_g \Rightarrow k \cdot x = m \cdot g$\\
			$55 \cdot x = (0.6)(9.8) \Rightarrow 55x = 5.88 \Rightarrow x = \frac{5.88}{55} \approx \mathbf{0.1069 \text{ m}}$ (o $\mathbf{10.69 \text{ cm}}$).\\
			\textit{*Nota: Si utilizan $g=10 \text{ m/s}^2$, el resultado es $10.9 \text{ cm}$.}}
	\end{enumerate}

	% --- PUNTO 2 ---
	\noindent \textbf{2.} Una fuente genera $60\text{ pulsos}$ en $3\text{ segundos}$, produciendo ondas de $20\text{ cm}$ de longitud. Calcule:
	\begin{enumerate}[label=\alph*), itemsep=0.3cm, leftmargin=0.8cm]
		\item La frecuencia y el periodo asociados.\\
		\textcolor{blue}{\textbf{Solución:} Frecuencia ($f$) = $\frac{\text{No. de ciclos}}{\text{Tiempo}} = \frac{60}{3} = \mathbf{20 \text{ Hz}}$.\\
			Periodo ($T$) = $\frac{1}{f} = \frac{1}{20 \text{ Hz}} = \mathbf{0.05 \text{ s}}$.}
		
		\item ¿Qué velocidad lleva la onda?\\
		\textcolor{blue}{\textbf{Solución:} $v = \lambda \cdot f = (0.20 \text{ m}) \cdot (20 \text{ Hz}) = \mathbf{4 \text{ m/s}}$ (o $\mathbf{400 \text{ cm/s}}$).}
	\end{enumerate}

	\begin{enumerate}[label=\alph*), itemsep=0.4cm, leftmargin=0.8cm]
		\item  La energía total de un sistema masa resorte es constante (no cambia) \underline{\hspace{0.5cm}\textcolor{blue}{\textbf{ V }}\hspace{0.5cm}} \\
		\textcolor{blue}{\textbf{Justificación:} Según el principio de conservación de la energía, la energía se transforma constantemente de potencial elástica a cinética y viceversa, pero la suma de ambas (energía mecánica total) permanece constante.}

	\end{enumerate}
