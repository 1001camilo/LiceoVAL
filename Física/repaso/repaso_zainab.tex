		\begin{enumerate}[label=\alph*), itemsep=0.3cm]

		\end{enumerate}
		
		\item \textbf{Problemas Aplicados con Geometría y Densidad:}
		\begin{enumerate}[label=\alph*), itemsep=0.4cm]
			\item El volumen de un recipiente esférico es de $113.04 \text{ litros}$, determina el radio exacto de la esfera expresado en \textbf{metros}.
			\item Un cilindro macizo tiene un radio de $8 \text{ cm}$ y una altura de $1.2 \text{ pies}$. Si su masa es de $5.5 \text{ kg}$, calcula su volumen y determina su densidad expresada en unidades del Sistema Internacional
		\end{enumerate}
		
		\item \textbf{Análisis Dimensional (Homogeneidad):} Determina, si las siguientes ecuaciones son dimensionalmente homogéneas:
		\begin{enumerate}[label=\alph*), itemsep=0.3cm]
			\item $v = a \cdot t - \frac{3 \cdot x}{t}$
			\item $x = 2a \cdot t^2 - v \cdot t$
		\end{enumerate}
	
		\newpage
		\item \textbf{Análisis Dimensional (Hallar Constantes Físicas):} Si la variable $a$ representa la aceleración y $t$ representa el tiempo, determina mediante igualación de dimensiones cuáles deben ser las unidades fundamentales de las constantes físicas $\alpha$, $\beta$ y $\gamma$ para que la siguiente ecuación cumpla con el principio de homogeneidad:
		$$v = \alpha t + \beta xt + \gamma t^3$$
		
		\item \textbf{Análisis Deductivo y Proporciones:} Un estudiante midió la densidad de un líquido en el laboratorio y obtuvo $0.92 \text{ g/ml}$. 	\end{enumerate}
