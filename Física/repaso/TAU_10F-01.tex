		\item \textbf{(10\%) Conversión de Unidades de Longitud, Masa y Tiempo:} 
		Aplica el método de \textbf{factores de conversión} (mostrando de forma explícita la simplificación de unidades) para realizar las siguientes transformaciones:
		\begin{enumerate}[label=\alph*), itemsep=0.2cm]
			\item Convertir $850\text{ pies}$ ($\text{ft}$) a metros ($\text{m}$). \quad (\textit{Dato: $1\text{ ft} = 0,3048\text{ m}$})
			\item Convertir $0,45\text{ toneladas}$ ($\text{ton}$) a gramos ($\text{g}$). \quad (\textit{Dato: $1\text{ ton} = 1.000\text{ kg}$, $1\text{ kg} = 1.000\text{ g}$})
			\item Convertir $3,5\text{ horas}$ ($\text{h}$) a segundos ($\text{s}$). \quad (\textit{Dato: $1\text{ h} = 3.600\text{ s}$})
		\end{enumerate}
		
		\answer{
			\textbf{Desarrollo paso a paso por factores de conversión:}
			\begin{itemize}[itemsep=0.2cm]
				\item a) $850\text{ ft} \times \left(\frac{0,3048\text{ m}}{1\text{ ft}}\right) = 850 \times 0,3048 = \mathbf{259,08\text{ m}}$.
				\item b) $0,45\text{ ton} \times \left(\frac{1.000\text{ kg}}{1\text{ ton}}\right) \times \left(\frac{1.000\text{ g}}{1\text{ kg}}\right) = 0,45 \times 10^6 = \mathbf{450.000\text{ g}}$ ($4,5 \times 10^5\text{ g}$).
				\item c) $3,5\text{ h} \times \left(\frac{3.600\text{ s}}{1\text{ h}}\right) = 3,5 \times 3.600 = \mathbf{12.600\text{ s}}$.
			\end{itemize}
		}

			\item La densidad del aluminio es de $2,7\text{ g/cm}^3$. Transforma esta densidad a kilogramos por metro cúbico ($\text{kg/m}^3$). \quad (\textit{Dato: $1\text{ kg} = 1.000\text{ g}$, $1\text{ m}^3 = 10^6\text{ cm}^3$})
		\end{enumerate}
		
		\answer{
			\textbf{Procedimiento explícito:}
			\begin{itemize}[itemsep=0.2cm]
				\item a) $108\text{ }\frac{\text{km}}{\text{h}} \times \left(\frac{1.000\text{ m}}{1\text{ km}}\right) \times \left(\frac{1\text{ h}}{3.600\text{ s}}\right) = \frac{108.000}{3.600} = \mathbf{30\text{ m/s}}$.
				\item b) $2,7\text{ }\frac{\text{g}}{\text{cm}^3} \times \left(\frac{1\text{ kg}}{1.000\text{ g}}\right) \times \left(\frac{10^6\text{ cm}^3}{1\text{ m}^3}\right) = 2,7 \times \frac{1.000.000}{1.000} = 2,7 \times 1.000 = \mathbf{2.700\text{ kg/m}^3}$.
			\end{itemize}
		}

		% PUNTO 7
		\item \textbf{(10\%) Análisis Dimensional y Coherencia de Ecuaciones:} 
		El análisis dimensional verifica si una fórmula física es correcta mediante el álgebra de dimensiones básicas: Longitud $[L]$, Masa $[M]$ y Tiempo $[T]$.
		\begin{enumerate}[label=\alph*), itemsep=0.2cm]
			\item Determina las dimensiones de la aceleración ($a = \frac{v}{t}$) y de la fuerza ($F = m \cdot a$).
			\item Verifica si la siguiente ecuación del movimiento es dimensionalmente homogénea:
			\[ d = v_0 \cdot t + \frac{1}{2} a \cdot t^2 \]
			donde $d$ es distancia $[L]$, $v_0$ es velocidad $[L T^{-1}]$, $a$ es aceleración $[L T^{-2}]$ y $t$ es tiempo $[T]$.
		\end{enumerate}
		
		\answer{
			\textbf{Demostración dimensional paso a paso:}
			\begin{itemize}[itemsep=0.15cm]
				\item a) Dimensiones de velocidad: $[v] = [L T^{-1}]$. \\
				Aceleración: $[a] = \frac{[v]}{[t]} = \frac{[L T^{-1}]}{[T]} = \mathbf{[L T^{-2}]}$. \\
				Fuerza: $[F] = [m] \cdot [a] = \mathbf{[M L T^{-2}]}$ (equivale a Newtons).
				\item b) Evaluamos cada término de la ecuación $d = v_0 \cdot t + \frac{1}{2} a \cdot t^2$:
				\begin{itemize}
					\item Lado izquierdo: $[d] = \mathbf{[L]}$.
					\item Primer término del lado derecho: $[v_0 \cdot t] = [L T^{-1}] \cdot [T] = \mathbf{[L]}$.
					\item Segundo término del lado derecho (las constantes adimensionales como $1/2$ no tienen dimensiones): $[\frac{1}{2} a \cdot t^2] = [L T^{-2}] \cdot [T^2] = \mathbf{[L]}$.
				\end{itemize}
				Como todos los términos tienen dimensiones de Longitud $[L]$, la ecuación es \textbf{dimensionalmente homogénea y correcta}.
			\end{itemize}
		}

		\item \textbf{(10\%) Práctica de Laboratorio -- El Proceso de la Medida y Márgenes de Error (10F01P):} 
		En la práctica de laboratorio "El proceso de la medida", un grupo de estudiantes midió el diámetro de una esfera de aluminio utilizando dos instrumentos diferentes:
		\begin{itemize}
			\item Medición con regla escolar básica: $2,5\text{ cm}$ ($\pm 0,1\text{ cm}$).
			\item Medición con calibrador pie de rey: $2,46\text{ cm}$ ($\pm 0,01\text{ cm}$).
			\item El valor teórico real de fábrica es de $2,48\text{ cm}$.
		\end{itemize}
		\begin{enumerate}[label=\alph*), itemsep=0.2cm]
			\item Calcula el porcentaje de error ($\% E$) de la medida obtenida con la regla escolar utilizando la fórmula:
			\[ \% E = \left| \frac{\text{Valor Teórico} - \text{Valor Experimental}}{\text{Valor Teórico}} \right| \times 100\% \]
			\item Calcula el porcentaje de error ($\% E$) de la medida obtenida con el calibrador pie de rey.
			\item Explica cuál de los dos instrumentos ofreció una mayor \textbf{precisión} y \textbf{exactitud} en la práctica.
		\end{enumerate}
		
		\answer{
			\textbf{Cálculo de márgenes de error:}
			\begin{itemize}[itemsep=0.2cm]
				\item a) Error con Regla escolar: $\% E = \left| \frac{2,48 - 2,5}{2,48} \right| \times 100\% = \left| \frac{-0,02}{2,48} \right| \times 100\% \approx \mathbf{0,806\%}$.
				\item b) Error con Calibrador pie de rey: $\% E = \left| \frac{2,48 - 2,46}{2,48} \right| \times 100\% = \left| \frac{0,02}{2,48} \right| \times 100\% \approx \mathbf{0,806\%}$.
				\item c) \textbf{Análisis de precisión y exactitud:} Ambos instrumentos resultaron ofrecer idéntica \textbf{exactitud} (el porcentaje de error respecto al valor teórico es el mismo). Sin embargo, el calibrador pie de rey ofrece una mayor \textbf{precisión} debido a su menor incertidumbre o sensibilidad de escala ($\pm 0,01\text{ cm}$ frente a $\pm 0,1\text{ cm}$ de la regla), permitiendo realizar mediciones centesimales.
			\end{itemize}
		}

		\item \textbf{(10\%) Problema Integrado de Aplicación Real:} 
		Un camión cisterna transporta $15.000\text{ litros}$ ($\text{L}$) de aceite mineral que tiene una densidad de $0,85\text{ g/cm}^3$. Durante el trayecto, viaja a una velocidad constante de $72\text{ km/h}$ durante un tiempo de $45\text{ minutos}$.
		\begin{enumerate}[label=\alph*), itemsep=0.2cm]
			\item Convierte el volumen transportado a metros cúbicos ($\text{m}^3$) y la densidad a $\text{kg/m}^3$. \quad (\textit{Dato: $1\text{ m}^3 = 1.000\text{ L}$})
			\item Calcula la masa total del aceite transportado en kilogramos ($\text{kg}$) y exprésala en notación científica. \quad (\textit{Fórmula: $\text{Masa} = \text{Densidad} \times \text{Volumen}$})
			\item Convierte la velocidad a $\text{m/s}$ y calcula la distancia total recorrida en metros ($\text{m}$) durante los $45\text{ minutos}$. \quad (\textit{Fórmula: $d = v \cdot t$})
		\end{enumerate}
		
		\answer{
			\textbf{Solución integrada paso a paso:}
			\begin{itemize}[itemsep=0.2cm]
				\item a) Volumen: $15.000\text{ L} \times \left(\frac{1\text{ m}^3}{1.000\text{ L}}\right) = \mathbf{15\text{ m}^3}$. \\
				Densidad: $0,85\text{ g/cm}^3 \times 1.000 = \mathbf{850\text{ kg/m}^3}$.
				\item b) Masa: $m = \text{Densidad} \times \text{Volumen} = 850\text{ kg/m}^3 \times 15\text{ m}^3 = \mathbf{12.750\text{ kg}}$. \\
				En notación científica: $\mathbf{1,275 \times 10^4\text{ kg}}$.
				\item c) Velocidad: $72\text{ km/h} = \mathbf{20\text{ m/s}}$. \\
				Tiempo: $45\text{ min} \times 60\text{ s/min} = \mathbf{2.700\text{ s}}$. \\
				Distancia: $d = v \cdot t = 20\text{ m/s} \times 2.700\text{ s} = \mathbf{54.000\text{ m}}$ ($54\text{ km}$ o $5,4 \times 10^4\text{ m}$).
			\end{itemize}
		}
		
	\end{enumerate}
