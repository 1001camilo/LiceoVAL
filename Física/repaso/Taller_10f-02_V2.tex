		\small
		\begin{itemize}[leftmargin=*, itemsep=0.1cm, topsep=0pt]
			\item \textbf{Movimiento Uniforme (MU)} ($v = \text{constante}$, $a = 0$): \quad $v = \dfrac{x_f - x_0}{t_f - t_0}$ \quad ó \quad $v = \dfrac{d}{t} \quad \mathbf{(2.I)}$
			\item \textbf{Movimiento Uniforme Acelerado / Variado (MUA / MUV)} ($a = \text{constante} \neq 0$):
			\[ a = \frac{v_f - v_0}{t} \quad \mathbf{(2.II)} \qquad\qquad x = x_0 + v_0 \cdot t + \frac{a \cdot t^2}{2} \quad \mathbf{(2.III)} \qquad\qquad v_f^2 = v_0^2 + 2a \cdot \Delta x \quad \mathbf{(2.IV)} \]
		\end{itemize}

		\item \textbf{(10\%) Diferenciación entre MU y MUA con conversión de unidades:} Analice las siguientes dos situaciones independientes, realice las conversiones de unidades al Sistema Internacional ($\text{m/s}$) y responda:
		\begin{enumerate}[label=\alph*., itemsep=0.2cm]
			\item \textbf{Caso A:} Un atleta trota en una pista recta a $18\text{ m/s}$ y $10\text{ segundos}$ después observa que su velocímetro marca $64,8\text{ km/h}$. ¿Qué tipo de movimiento presenta? ¿Qué distancia recorre en este intervalo?
			\item \textbf{Caso B:} Un tren de carga avanza a $126\text{ km/h}$ y, al aplicar una retención en sus frenos durante $8\text{ segundos}$, reduce su velocidad a $15\text{ m/s}$. ¿El tren acelera o desacelera? ¿Qué tipo de movimiento presenta y qué distancia recorre en esos $8\text{ s}$?
		\end{enumerate}
		\respuesta{
			\textbf{a) Caso A:} Convertimos $64,8\text{ km/h}$ a $\text{m/s}$: $64,8 \times \frac{1000}{3600} = 18\text{ m/s}$. Como $v_0 = 18\text{ m/s}$ y $v_f = 18\text{ m/s}$, la velocidad se mantiene constante ($a = 0\text{ m/s}^2$), por tanto presenta un \textbf{Movimiento Uniforme (MU)}. Con la ec. (2.I): $d = v \cdot t = (18\text{ m/s})(10\text{ s}) = \mathbf{180}\text{ \textbf{m}}$.\\[0.1cm]
			\textbf{b) Caso B:} Convertimos $126\text{ km/h}$ a $\text{m/s}$: $v_0 = 35\text{ m/s}$. Al pasar de $35\text{ m/s}$ a $15\text{ m/s}$ en $t = 8\text{ s}$, su velocidad disminuye (\textbf{desacelera}), presentando un \textbf{Movimiento Uniforme Desacelerado $\text{MUA}(-)$}.\\
			Aceleración con ec. (2.II): $a = \frac{15 - 35}{8} = \frac{-20}{8} = \mathbf{-2,5}\text{ \textbf{m/s}}^\mathbf{2}$.\\
			Distancia con ec. (2.III): $x = (35\text{ m/s})(8\text{ s}) + \frac{(-2,5\text{ m/s}^2)(8\text{ s})^2}{2} = 280 - 80 = \mathbf{200}\text{ \textbf{m}}$.
		}

En $t_0 = 0\text{ s}$ se ubica en la posición $x_0 = 180\text{ m}$ a la derecha del origen y se desplaza con velocidad constante hasta alcanzar la posición $x_f = 20\text{ m}$ en el instante $t_f = 16\text{ s}$.

		\respuesta{
			\textbf{a)} $v = \frac{20\text{ m} - 180\text{ m}}{16\text{ s} - 0\text{ s}} = \frac{-160\text{ m}}{16\text{ s}} = \mathbf{-10}\text{ \textbf{m/s}}$. El signo negativo indica que el robot avanza en sentido opuesto al eje positivo del sistema de referencia (regresando hacia el origen $x=0$).\\[0.1cm]
			\textbf{b)} Datos del frenado: $v_0 = -10\text{ m/s}$, $v_f = 0\text{ m/s}$, $t = 4\text{ s}$. Con ec. (2.II): $a = \frac{0 - (-10\text{ m/s})}{4\text{ s}} = \mathbf{+2,5}\text{ \textbf{m/s}}^\mathbf{2}$. (2.IV):} Un tren de alta velocidad se desplaza a $144\text{ km/h}$ cuando el maquinista observa una señal de detención y aplica los frenos de emergencia con aceleración constante, logrando detenerlo por completo tras recorrer $250\text{ metros}$.
		\begin{enumerate}[label=\alph*., itemsep=0.2cm]
			\item Determine la aceleración (desaceleración) del tren utilizando la ecuación \textbf{(2.IV)}.
			\item Calcule el tiempo total que tomó el proceso de frenado empleando la ecuación \textbf{(2.II)}.
		\end{enumerate}
		\respuesta{
			Convertimos: $v_0 = 144\text{ km/h} = 40\text{ m/s}$; $v_f = 0\text{ m/s}$; $\Delta x = 250\text{ m}$.\\[0.1cm]
			\textbf{a)} De $v_f^2 = v_0^2 + 2a\Delta x \implies 0^2 = (40)^2 + 2a(250) \implies -1600 = 500a \implies a = \frac{-1600}{500} = \mathbf{-3,2}\text{ \textbf{m/s}}^\mathbf{2}$.\\[0.1cm]

		}

			\textbf{a)} Con ec. (2.IV): $v_f^2 = 0^2 + 2(3,5)(280) = 1960 \implies v_f = \sqrt{1960} \approx \mathbf{44,27}\text{ \textbf{m/s}}$.\\
			En $\text{km/h}$: $44,27 \times 3,6 \approx \mathbf{159,38}\text{ \textbf{km/h}}$.\\[0.1cm]
			\textbf{b)} Con ec. (2.II): $t = \frac{v_f - v_0}{a} = \frac{44,27\text{ m/s}}{3,5\text{ m/s}^2} \approx \mathbf{12,65}\text{ \textbf{s}}$.
		}

A continuación, mantiene esa velocidad alcanzada durante $1,2\text{ km}$ con movimiento uniforme (\textbf{Etapa B}). Finalmente, el conductor aplica los frenos desacelerando uniformemente hasta detenerse por completo en $6\text{ segundos}$ (\textbf{Etapa C}).

			\textbf{Etapa A ($\text{MUA}(+)$):} $v_{0A}=0$, $a_A=2,0\text{ m/s}^2$, $t_A = 15\text{ s}$.\\

			\textbf{Etapa B ($\text{MU}$):} $v_B = 30\text{ m/s}$, $d_B = 1,2\text{ km} = 1200\text{ m}$.\\
			Tiempo B: $t_B = \frac{1200\text{ m}}{30\text{ m/s}} = \mathbf{40}\text{ \textbf{s}}$.\\[0.1cm]
			\textbf{Etapa C ($\text{MUA}(-)$):} $v_{0C} = 30\text{ m/s}$, $v_{fC} = 0$, $t_C = 6\text{ s}$.\\
			Aceleración C: $a_C = \frac{0 - 30}{6} = \mathbf{-5,0}\text{ \textbf{m/s}}^\mathbf{2}$. Distancia C: $x_C = 30(6) + \frac{1}{2}(-5,0)(6)^2 = 180 - 90 = \mathbf{90}\text{ \textbf{m}}$.\\[0.1cm]
			\textbf{Totales:} Distancia total $x_{\text{total}} = 225 + 1200 + 90 = \mathbf{1515}\text{ \textbf{m}}$ \quad | \quad Tiempo total $t_{\text{total}} = 15 + 40 + 6 = \mathbf{61}\text{ \textbf{s}}$.
		}

		\begin{enumerate}[label=\alph*., itemsep=0.2cm]
			\item Exprese la velocidad del auto de carreras en $\text{m/s}$ y plantee las ecuaciones de posición para ambos vehículos.
			\item ¿En cuánto tiempo alcanza la patrulla al auto de carreras y a qué distancia de la meta ocurre la interceptación?

			\textbf{b)} Igualando $x_i = x_p \implies 45 t = 1,5 t^2 \implies t = \frac{45}{1,5} = \mathbf{30}\text{ \textbf{s}}$.\\
			Distancia de interceptación: $x = 45(30) = \mathbf{1350}\text{ \textbf{m}}$.\\[0.1cm]

En $t = 0\text{ s}$, el vehículo A sale desde $x_0 = 0$ con velocidad constante de $20\text{ m/s}$ hacia la derecha. Simultáneamente, el vehículo B parte desde el reposo en $x_0 = 450\text{ m}$ acelerando hacia la izquierda a razón de $a = -1,5\text{ m/s}^2$.
		\begin{enumerate}[label=\alph*., itemsep=0.2cm]
			\item Escriba las ecuaciones de posición $x_A(t)$ y $x_B(t)$ para ambos móviles.
			\item Calcule el tiempo transcurrido $t$ hasta que se cruzan y la posición exacta del encuentro respecto a $x = 0$.
		\end{enumerate}
		\respuesta{
			\textbf{a)} $x_A(t) = 20 t$ \quad | \quad $x_B(t) = 450 + 0 \cdot t + \frac{1}{2}(-1,5) t^2 = 450 - 0,75 t^2$.\\[0.1cm]
			\textbf{b)} Igualando $x_A = x_B \implies 20 t = 450 - 0,75 t^2 \implies 0,75 t^2 + 20 t - 450 = 0$.\\
			Multiplicando por 4: $3 t^2 + 80 t - 1800 = 0$.\\
			Usando la fórmula cuadrática: $t = \frac{-80 \pm \sqrt{6400 - 4(3)(-1800)}}{6} = \frac{-80 \pm \sqrt{28000}}{6} \approx \frac{-80 + 167,33}{6} \approx \mathbf{14,56}\text{ \textbf{s}}$.\\
			Posición de encuentro: $x_A = 20(14,56) = \mathbf{291,2}\text{ \textbf{m}}$.
		}

El tiempo de reacción del conductor antes de presionar el pedal del freno es de $0,75\text{ segundos}$. Al accionar los frenos, el vehículo desacelera uniformemente a razón de $-4,5\text{ m/s}^2$.
		\begin{enumerate}[label=\alph*., itemsep=0.2cm]
			\item Calcule la distancia recorrida durante el tiempo de reacción ($\text{MU}$) y la distancia recorrida durante el frenado ($\text{MUA}$).
			\item Determine la distancia total de detención y responda si el conductor logra evitar la colisión con el obstáculo, indicando a qué distancia se detiene.
		\end{enumerate}
		\respuesta{
			\textbf{a)} Distancia de reacción ($\text{MU}$): $x_{\text{reacción}} = v \cdot t_{\text{reacción}} = (20\text{ m/s})(0,75\text{ s}) = \mathbf{15}\text{ \textbf{m}}$.\\
			Distancia de frenado ($\text{MUA}$): Con $v_f^2 = v_0^2 + 2a x_{\text{frenado}} \implies 0^2 = 20^2 + 2(-4,5) x_{\text{frenado}} \implies 9 x_{\text{frenado}} = 400 \implies x_{\text{frenado}} = \mathbf{44,44}\text{ \textbf{m}}$.\\[0.1cm]
			\textbf{b)} Distancia total de detención: $x_{\text{total}} = 15 + 44,44 = \mathbf{59,44}\text{ \textbf{m}}$.\\
			Como $59,44\text{ m} < 65\text{ m}$, \textbf{sí logra evitar la colisión}, deteniéndose $65 - 59,44 = \mathbf{5,56}\text{ \textbf{metros}}$ antes del obstáculo.
		}
		
		% --- PUNTO 10 ---
		\item \textbf{(10\%) Evaluación de Enunciados Conceptuales (Falso o Verdadero):} Responda Falso (\textbf{F}) o Verdadero (\textbf{V}) a cada enunciado y justifique detalladamente su respuesta con un ejemplo o argumento físico:
		\begin{enumerate}[label=\alph*., itemsep=0.3cm]
			\item \underline{\hspace{1.2cm}} Una aceleración con signo negativo indica de forma obligatoria que el objeto está disminuyendo su rapidez (frenando). \\[0.15cm]
			\textbf{Justificación:} \hrulefill \\[0.15cm] \mbox{}\hrulefill
			
			\item \underline{\hspace{1.2cm}} En una gráfica de posición contra tiempo ($x$ vs. $t$), la pendiente de la recta representa la velocidad instantánea del móvil. \\[0.15cm]
			\textbf{Justificación:} \hrulefill \\[0.15cm] \mbox{}\hrulefill
			
			\item \underline{\hspace{1.2cm}} En una gráfica de velocidad contra tiempo ($v$ vs. $t$), el área total bajo la curva siempre representa la distancia recorrida, sin importar el signo de la velocidad. \\[0.15cm]
			\textbf{Justificación:} \hrulefill \\[0.15cm] \mbox{}\hrulefill
			
			\item \underline{\hspace{1.2cm}} Si un objeto parte del reposo y acelera de forma constante, la distancia recorrida es directamente proporcional al tiempo transcurrido. \\[0.15cm]
			\textbf{Justificación:} \hrulefill \\[0.15cm] \mbox{}\hrulefill
		\end{enumerate}
		\respuesta{
			\textbf{a) Falso (F):} Si la velocidad es negativa (avanzando hacia la izquierda) y la aceleración es negativa, el objeto aumenta su rapidez en sentido negativo (acelera en reversa). Solamente desacelera cuando velocidad y aceleración tienen signos opuestos.\\[0.1cm]
			\textbf{b) Verdadero (V):} La pendiente de la curva en la gráfica $x$ vs. $t$ está dada por $m = \frac{\Delta x}{\Delta t} = v$, que corresponde físicamente a la velocidad del objeto.\\[0.1cm]
			\textbf{c) Falso (F):} El área con signo algebraico da el \textbf{desplazamiento neto}. Para obtener la \textbf{distancia total} se deben sumar los valores absolutos de las áreas positiva y negativa.\\[0.1cm]
			\textbf{d) Falso (F):} Según la ecuación $x = \frac{1}{2} a t^2$, la distancia recorrida es directamente proporcional al \textbf{cuadrado del tiempo} ($t^2$), no al tiempo lineal.
		}
	\end{enumerate}
