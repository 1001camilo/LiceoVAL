% --- PUNTO 4 ---
\noindent \textbf{\textcolor{valblue}{4. PRACTICANDO LA VELOCIDAD DE LA LUZ}}

\vspace{0.2cm}
\noindent \textit{¡Vamos a practicar la fórmula para el examen con un ejemplo diferente!}

\vspace{0.2cm}
\noindent \textbf{Contexto:} La luz es lo más rápido del universo, viaja a \textbf{300.000} km/s. Pero si entra a un diamante brillante, "frena" muchísimo.

\vspace{0.2cm}
\noindent \textbf{Ejercicio:} El índice del diamante es \textbf{3}. Vamos a dividir la velocidad original de la luz entre 3 para saber a qué velocidad viaja adentro del diamante:

\vspace{0.4cm}
\begin{center}
    \textbf{Fórmula:} \quad Velocidad en Diamante $= \frac{\textbf{Velocidad Original}}{\textbf{Índice del Diamante}}$
\end{center}

\vspace{0.4cm}
\begin{center}
    \textit{(Reemplaza y divide. Pista: ¿Cuál es la tercera parte de 300.000? Piensa en $3 \div 3$ y agrega los ceros)}\\[0.5cm]
    
    \Large 
    Velocidad en Diamante $= \frac{300.000}{3} =$ \boxfill[3cm] km/s
\end{center}
