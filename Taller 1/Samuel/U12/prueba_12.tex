	\begin{conceptbox}{Valor Absoluto — Clave para multiplicar y dividir}
		El \textbf{valor absoluto} de un entero $n$, notado $|n|$, es su
		\textbf{distancia al cero} en la recta numérica. Siempre es no negativo:
		\[
		|-7| = 7 \qquad |{+5}| = 5 \qquad |0| = 0 \qquad
		|-a| = |a| \text{ para todo } a
		\]
		\textbf{¿Por qué importa aquí?}\; Al multiplicar o dividir enteros,
		el \textit{módulo} (tamaño) del resultado se calcula con los valores
		absolutos, y el \textit{signo} se determina por una regla aparte.
		Son dos problemas independientes.
	\end{conceptbox}

				minimum width=2.7cm, minimum height=2.3cm,

	\begin{alertbox}{Propiedad del Cero en la Multiplicación}
		Para todo $a \in \mathbb{Z}$:\quad $a \times 0 = 0 \times a = 0$.
		\quad\textit{Cualquier entero multiplicado por cero da cero.}
		
		\vspace{4pt}
		\textbf{Consecuencia importante:}\; Si $a \cdot b = 0$, entonces
		$a = 0$ \textbf{o} $b = 0$ (o ambos). Esta propiedad se llama
		\textit{ley de cancelación del cero}.
	\end{alertbox}

	\begin{trickbox}{Multiplicación de más de dos factores — Cuenta los negativos}
		Cuando multiplicas \textbf{tres o más enteros}, no necesitas ir de a pares.
		Usa este truco de \textbf{conteo de signos negativos}:
		
		\vspace{0.2cm}
		
		\begin{center}
			\begin{tikzpicture}
				\node[draw=valred, fill=vallightred, rounded corners=6pt,
				minimum width=6cm, minimum height=1cm,
				align=center, font=\bfseries, line width=2pt] at (0,0)
				{Cantidad \textbf{IMPAR} de negativos $\Rightarrow$ resultado $\boldsymbol{-}$};
				\node[draw=valgreen, fill=vallightgreen, rounded corners=6pt,
				minimum width=6cm, minimum height=1cm,
				align=center, font=\bfseries, line width=2pt] at (8.2,0)
				{Cantidad \textbf{PAR} de negativos $\Rightarrow$ resultado $\boldsymbol{+}$};
				\draw[-{Stealth}, valdark, line width=1.5pt] (3.1,0) -- (5.1,0);
			\end{tikzpicture}
		\end{center}
		
		\vspace{0.2cm}
		\textbf{Ejemplo:}\;
		$\underbrace{(-2)}_{\text{neg}}\times
		\underbrace{(-3)}_{\text{neg}}\times
		\underbrace{(-1)}_{\text{neg}}\times
		(+4)\times
		\underbrace{(-2)}_{\text{neg}}$
		\quad $\Rightarrow$ \textbf{4 negativos (par)} $\Rightarrow$ signo $+$
		
		Módulo:\; $2\times3\times1\times4\times2 = 48$ \quad
		Resultado:\; $\boldsymbol{+48}$
	\end{trickbox}

	\misionheader[valorange]{4}{Descubrimiento — Regla de Signos en la División}
	
	\begin{conceptbox}{La División como Operación Inversa}
		La \textbf{división} es la operación inversa de la multiplicación.
		Esta relación fundamental permite deducir la regla de signos
		de la división \textit{directamente} de la que ya conoces:
		\[
		a \div b = c \quad \Longleftrightarrow \quad b \times c = a
		\qquad (b \neq 0)
		\]
		\textbf{Ejemplo:}\; $12 \div 4 = 3$ porque $4 \times 3 = 12$.
		\quad No necesitas memorizar nada nuevo: la misma lógica aplica.
	\end{conceptbox}
	
	\vspace{0.3cm}

	\begin{alertbox}{División por cero — La operación prohibida}
		\textbf{La división por cero no está definida.}\;
		$\dfrac{a}{0}$ \textbf{NO EXISTE} para ningún valor de $a$.
		
		\vspace{4pt}
		\textbf{¿Por qué?}\; Porque no existe ningún $c \in \mathbb{Z}$ tal que
		$0 \times c = a$ cuando $a \neq 0$. Y si $a = 0$, cualquier $c$
		serviría, lo que da un resultado \textit{indeterminado}.
		
		\vspace{4pt}
		\textbf{Sin embargo:}\; $\dfrac{0}{a} = 0$ siempre que $a \neq 0$.
		El cero \textit{sí puede} ser dividendo.
	\end{alertbox}

	\begin{multicols}{2}
		
		\begin{conceptbox}{Número Entero}
			Elemento del conjunto
			$\mathbb{Z} = \{\ldots,-2,-1,0,1,2,\ldots\}$.
			Se extiende infinitamente en ambas direcciones.
			\vspace{2pt}\\
			\textbf{Notación:} $n \in \mathbb{Z}$\\
			\textbf{Ej.:} $-7 \in \mathbb{Z}$,\; $+3 \in \mathbb{Z}$,\; $0 \in \mathbb{Z}$
		\end{conceptbox}
		
		\begin{conceptbox}{Valor Absoluto}
			Distancia de un entero al cero en la recta
			numérica. Siempre es $\geq 0$.
			\vspace{2pt}\\
			\textbf{Notación:} $|n|$\\
			\textbf{Ej.:} $|-5|=5$,\; $|+3|=3$,\; $|0|=0$\\
			\textbf{Propiedad:} $|n| = |-n|$
		\end{conceptbox}
		
		\begin{conceptbox}{Factor}
			Cada uno de los números que se multiplican
			en un producto.
			\vspace{2pt}\\
			\textbf{Ej.:} En $(-3)\times(+7)$,\;
			los factores son $-3$ y $+7$.
		\end{conceptbox}
		
		\begin{conceptbox}{Producto}
			Resultado de una multiplicación.
			\vspace{2pt}\\
			\textbf{Ej.:} En $(-3)\times(+7) = -21$,\;
			el producto es $-21$.
		\end{conceptbox}
		
		\begin{conceptbox}{Dividendo}
			El número que se divide (el que se reparte).
			\vspace{2pt}\\
			\textbf{Ej.:} En $\dfrac{-48}{6}$,\;
			el dividendo es $-48$.
		\end{conceptbox}
		
		\begin{conceptbox}{Divisor}
			El número por el cual se divide.
			\vspace{2pt}\\
			\textbf{Ej.:} En $\dfrac{-48}{6}$,\;
			el divisor es $6$.\\
			\textbf{Restricción:} Divisor $\neq 0$.
		\end{conceptbox}
		
		\begin{conceptbox}{Cociente}
			Resultado de una división exacta.
			\vspace{2pt}\\
			\textbf{Ej.:} En $(-48)\div 6 = -8$,\;
			el cociente es $-8$.\\
			\textbf{Verificación:} $6\times(-8)=-48$\;\checkmark
		\end{conceptbox}
		
		\begin{conceptbox}{Operaciones Inversas}
			Dos operaciones que se ``deshacen''
			mutuamente. Multiplicación y división
			son inversas:
			\vspace{2pt}\\
			$a\times b = c \;\Leftrightarrow\; c\div b = a$
		\end{conceptbox}
		
		\begin{conceptbox}{Propiedad Conmutativa}
			El orden de los factores no altera
			el producto:
			\vspace{2pt}\\
			$a \times b = b \times a$\\
			\textbf{Ojo:} La división \textbf{NO} es conmutativa.
		\end{conceptbox}
		
		\begin{conceptbox}{Propiedad Asociativa}
			Al agrupar tres factores de distinta
			manera, el producto no cambia:
			\vspace{2pt}\\
			$(a \times b)\times c = a\times(b\times c)$\\
			\textbf{Ojo:} La división \textbf{NO} es asociativa.
		\end{conceptbox}
		
		\begin{conceptbox}{Propiedad Distributiva}
			La multiplicación se distribuye sobre
			la suma (y la resta):
			\vspace{2pt}\\
			$a(b+c) = ab + ac$\\
			\textbf{Ojo:} No aplica para el divisor.
		\end{conceptbox}
		
		\begin{conceptbox}{Jerarquía de Operaciones}
			Orden obligatorio para evaluar expresiones:
			\vspace{2pt}\\
			1.	extsuperscript{o} Paréntesis (más internos primero)\\
			2.	extsuperscript{o} Potencias y raíces\\
			3.	extsuperscript{o} $\times$ y $\div$ (izq. a der.)\\
			4.	extsuperscript{o} $+$ y $-$ (izq. a der.)
		\end{conceptbox}
		
	\end{multicols}
