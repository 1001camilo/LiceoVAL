	\section*{Misión 1: Más allá del cero}
	
	\begin{concepto}{El Mundo de los Positivos y Negativos}
		Hasta ahora has trabajado con números naturales ($1, 2, 350, 1000...$) para contar cosas que tienes. Pero, ¿qué pasa si debes dinero? ¿O si un buzo baja al fondo del mar? ¡Ahí nacen los \textbf{Números Enteros ($\mathbb{Z}$)}!
		\begin{itemize}
			\item \textbf{Números Positivos ($+$):} Representan ganancias, temperaturas sobre cero, subir pisos o ir a la derecha. (Ej: $+50$ o simplemente $50$).
			\item \textbf{Números Negativos ($-$):} Representan deudas, temperaturas bajo cero, bajar al sótano o profundidades marinas. \textbf{Siempre} llevan el signo menos. (Ej: $-120$).
			\item \textbf{El Cero ($0$):} Es la frontera. El nivel del mar, la planta baja o cuando no debes ni tienes dinero.
		\end{itemize}
	\end{concepto}

	\section*{Misión 2: El Mapa de los Números}
	
	\begin{concepto}{Ubicación y Duelos en la Recta}
		La recta numérica es como una regla infinita donde organizamos todos los números.
		\begin{itemize}
			\item A la \textbf{derecha} del cero viven los \textcolor{valblue}{\textbf{Positivos}}. Mientras más a la derecha, ¡más grandes son!
			\item A la \textbf{izquierda} del cero viven los \textcolor{valred}{\textbf{Negativos}}. Mientras más a la izquierda, ¡más pequeños son!
		\end{itemize}
		\textbf{La Regla del Tamaño:} En un duelo de números, \textbf{el que esté más a la derecha SIEMPRE es mayor.} Por ejemplo, tener una deuda de 10.000 ($-10.000$) es mucho peor (menor) que tener una deuda de solo 100 ($-100$).
	\end{concepto}
	
	\vspace{0.2cm}
	\begin{center}

	\section*{Misión 3: Saltos en la Recta (Suma y Resta)}
	
	\begin{truco}{El Viaje del Sapito}
		Sumar y restar números enteros es exactamente igual a hacer que un sapito salte por la recta numérica:
		\begin{itemize}
			\item El \textbf{primer número} te dice en qué lugar exacto de la recta \textbf{inicias} tu viaje.
			\item El \textbf{segundo número} (con su signo) te indica hacia dónde debes saltar:
			\begin{itemize}
				\item \textbf{Sumar (+):} Saltas hacia la \textcolor{valblue}{\textbf{DERECHA}}.
				\item \textbf{Restar (-):} Saltas hacia la \textcolor{valred}{\textbf{IZQUIERDA}}.
			\end{itemize}
		\end{itemize}
	\end{truco}
	
	\begin{ejemplo}
		\textbf{Ejemplo 1: $\mathbf{-2 + 5 = ?}$} \\
		Empiezas en el $\mathbf{-2}$. Como dice $\mathbf{+5}$, saltas 5 espacios a la \textcolor{valblue}{\textbf{DERECHA}}.

	\section*{Misión 4: Detectives en la Vida Real}
	
	\begin{concepto}{Traducir a lenguaje matemático}
		Los problemas de la vida real son fáciles si extraes la información y le pones el signo correcto:
		\begin{itemize}
			\item \textbf{Bajar, gastar, deber, profundidad, bajo cero} $\rightarrow$ Signo Menos ($-$).
			\item \textbf{Subir, ganar, abonar, altura, sobre cero} $\rightarrow$ Signo Más ($+$).
		\end{itemize}
		Arma la operación con estos signos guiándote con las cajitas y resuélvela.
	\end{concepto}
	
	\begin{practica}{Problemas}
		\textit{Escribe la operación completa en los primeros recuadros (ej: $-150+300$) y el resultado final a la derecha.}
		
		\vspace{0.4cm}
		\begin{enumerate}[itemsep=0.9cm]
			\item El submarino está explorando a \textbf{450 m de profundidad} (bajo el nivel del mar). Luego, \textbf{asciende 300 m}. ¿A qué profundidad quedó? \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} Quedó a \boxfill[1.5cm] m.
			
			\item Tienes una deuda de \textbf{\$ 4.500}. Al día siguiente abonas \textbf{\$ 2.000}. ¿Cuál es tu saldo? \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} Saldo de \$\boxfill[1.5cm]
			
			\item Un halcón vuela a \textbf{850 m} de altura. Localiza una presa y \textbf{desciende 900 m} picando hacia el valle. ¿A qué altura llegó? \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} Llegó a \boxfill[1.5cm] m.
			
			\item La temperatura en el polo es de \textbf{$-15^\circ$C}. Sale el sol y la temperatura \textbf{sube $25^\circ$C}. ¿A qué temperatura quedaron? \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} Quedaron a \boxfill[1.5cm] $^\circ$C.
			
			\item El ascensor está en el \textbf{sótano 3}. Sube \textbf{12 pisos} de corrido. ¿A qué piso llegó? \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} Al piso \boxfill[1.5cm]
			
			\item Debes \textbf{\$ 3.200} en la papelería. Al otro día vas y fías unos marcadores de \textbf{\$ 1.800}. ¿A cuánto subió tu deuda? \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} Debo \$\boxfill[1.5cm]
			
			\item Un minero está a \textbf{$-250$ m} bajo tierra. Baja otros \textbf{150 m} buscando oro. ¿Dónde está? \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} Está a \boxfill[1.5cm] m.
			
			\item Tenías \textbf{\$ 15.000} en efectivo. Gastas \textbf{\$ 18.000} pagando con tarjeta de crédito. ¿Cómo queda tu cuenta? \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} Cuenta en \$\boxfill[1.5cm]
			
			% Problemas de 3 operaciones
			\item El submarino está a \textbf{$-500$ m}. Baja \textbf{200 m} y luego sube \textbf{800 m}. \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} \boxfill[1.5cm] m.
			
			\item Debes \textbf{\$ 4.500}. Abonas \textbf{\$ 2.000}. Luego fías \textbf{\$ 1.500} más. \\
			\textbf{Operación:} \boxfill[6cm] $=$ \boxfill[1cm] \hspace{0.2cm} \textbf{Rta:} \$\boxfill[1.5cm]
		\end{enumerate}
		
		\vspace{0.2cm}
		\textbf{\textit{Escribe la operación y el resultado en la caja final:}}
		\vspace{0.2cm}
		\begin{enumerate}[itemsep=0.8cm, start=11]
			\item De una cuenta de \$ 1.000, sacan \$ 1.500 y luego depositan \$ 2.000 ¿Cuánto queda en la cuenta?. \\
			$\rightarrow$ \boxfill[8cm]
			\item En el día se tenía una temperatura de $10^\circ$C baja $15^\circ$C en la noche y vuelve a bajar $5^\circ$C en la madrugada ¿Cual es la temperatura en la madrugada? \\
			$\rightarrow$ \boxfill[8cm]
			\item UN pájaro vuela a $300$m, baja $450$m a pescar (entra al agua) y vuelve a subir $200$m.¿A qué altura queda el pájaro? \\
			$\rightarrow$ \boxfill[8cm]
			\item Te regalan \$ 5.000, te gastas \$ 8.000 y te gastas otros \$ 2.000 ¿Cuánto quedas debiendo? \\
			$\rightarrow$ \boxfill[8cm]
			\item El ascensor sale del piso 5, baja 8 pisos, y sube 2 pisos. ¿Dónde quedó? \\
			$\rightarrow$ \boxfill[8cm]
		\end{enumerate}
	\end{practica}
