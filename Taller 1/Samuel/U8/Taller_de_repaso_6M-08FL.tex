	\noindent \textbf{1. Convierte los siguientes números mixtos a fracciones impropias. (20\%)}

	\begin{multicols}{3}
		\begin{enumerate}[label=\textbf{\Alph*)}, itemsep=0.8cm]
			\item $3 \frac{1}{2} = \dfrac{\boxfill[0.8cm]}{\boxfill[0.8cm]}$
			
			\item $4 \frac{2}{3} = \dfrac{\boxfill[0.8cm]}{\boxfill[0.8cm]}$
			
			\item $1 \frac{3}{5} = \dfrac{\boxfill[0.8cm]}{\boxfill[0.8cm]}$
		\end{enumerate}
	\end{multicols}
	
	\vspace{0.8cm}
	
	% --- PUNTO 2 ---
	\noindent \textbf{2. Convierte las siguientes fracciones impropias a números mixtos. (20\%)}
	
	\vspace{0.4cm}
	\begin{multicols}{2}
		\begin{enumerate}[label=\textbf{\Alph*)}, itemsep=0.8cm]
			\item $\dfrac{9}{2} = \boxfill[0.6cm] \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]}$
			
			\item $\dfrac{11}{4} = \boxfill[0.6cm] \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]}$
		\end{enumerate}
	\end{multicols}
	
	\vspace{0.8cm}
	
	% --- PUNTO 3 ---
	\noindent \textbf{3. Resuelve las siguientes sumas y restas de números mixtos. (20\%)}
	
	\vspace{0.5cm}
	\begin{multicols}{2}
		\begin{enumerate}[label=\textbf{\Alph*)}, itemsep=1.5cm]
			\item $2 \frac{1}{4} + 3 \frac{2}{4} = \boxfill[0.6cm] \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]}$
			
			\item $6 \frac{5}{8} - 2 \frac{1}{8} = \boxfill[0.6cm] \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]}$
		\end{enumerate}
	\end{multicols}
	
	\vspace{0.8cm}

	% --- PUNTO 4 ---
	\noindent \textbf{4. Resuelve las siguientes multiplicaciones y divisiones (recuerda convertirlos primero a fracciones impropias). (20\%)}
	
	\vspace{0.5cm}
	\begin{multicols}{2}
		\begin{enumerate}[label=\textbf{\Alph*)}, itemsep=1.5cm]
			\item $1 \frac{1}{3} \times 2 \frac{1}{2} = \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]} \times \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]} = \dfrac{\boxfill[0.8cm]}{\boxfill[0.8cm]}$
			
			\item $4 \frac{1}{2} \div 1 \frac{1}{3} = \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]} \div \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]} = \dfrac{\boxfill[0.8cm]}{\boxfill[0.8cm]}$
		\end{enumerate}
	\end{multicols}
	
	
	
	% --- PUNTO 5 ---
	\noindent \textbf{5. Lee la siguiente situación, completa la operación y calcula el resultado. (20\%)}
	
	\vspace{0.4cm}
	\noindent \textit{Un panadero tiene $5 \frac{3}{4}$ kilos de harina y gasta $3 \frac{1}{4}$ kilos en preparar un pastel grande. ¿Cuánta harina le sobra?}
	
	\vspace{0.6cm}
	\begin{center}
		\begin{tabular}{c c c}
			\textbf{Operación:} & \hspace{2cm} & \textbf{Respuesta:} \\[0.6cm]
			\Large $5 \frac{3}{4} - 3 \frac{1}{4} = \boxfill[0.6cm] \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]}$ & & 
			Le sobran $\boxfill[0.6cm] \dfrac{\boxfill[0.6cm]}{\boxfill[0.6cm]}$ kilos.
		\end{tabular}
	\end{center}
