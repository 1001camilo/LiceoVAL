	\section*{Misión 1: Tipos de Gráficas y Lectura de Tablas}
	
	\begin{concepto}{La Estadística y sus Representaciones}
		La \textbf{estadística} recoge, organiza e interpreta datos para poder
		sacar conclusiones sobre una situación. Un mismo conjunto de datos
		puede representarse de \textbf{varias formas} según lo que se quiera
		mostrar:
		
		\vspace{0.3cm}
		\renewcommand{\arraystretch}{1.8}
		\begin{center}
			\begin{longtable}{|>{\bfseries}m{3.3cm}|m{10.8cm}|}
				\hline
				\rowcolor{valblue!20}
				\textbf{Tipo de gráfica} & \textbf{¿Para qué se usa?} \\
				\hline
				Gráfica de \textcolor{valblue}{barras} &
				Compara \textbf{cantidades} entre distintas categorías usando
				rectángulos de distinta altura o longitud. \\
				\hline
				Gráfica \textcolor{valorange}{circular} (de torta) &
				Muestra cómo se \textbf{reparte un total} en partes
				proporcionales, generalmente en porcentajes. \\
				\hline
				\textcolor{valgreen}{Pictograma} &
				Usa \textbf{símbolos o íconos} repetidos, donde cada símbolo
				representa una cantidad fija de elementos. \\
				\hline
				Gráfica de \textcolor{valred}{líneas} &
				Conecta puntos con segmentos para mostrar la
				\textbf{evolución} de un dato a través del \textbf{tiempo}. \\
				\hline
				Histograma &
				Agrupa datos numéricos \textbf{continuos} en intervalos,
				representados con barras \textbf{unidas} entre sí (sin
				espacio). \\
				\hline
			\end{longtable}
		\end{center}
	\end{concepto}
	
	\begin{atencion}{¡Cuidado! Barras vs. Histograma}
		En la gráfica de barras hay \textbf{espacio} entre cada barra porque
		representa \textbf{categorías distintas} (ej: países, colores,
		equipos). En el histograma las barras están \textbf{pegadas} porque
		representan \textbf{intervalos consecutivos} de un mismo dato
		numérico (ej: edades de 0-10, 10-20, 20-30...).
	\end{atencion}
	
	\vspace{0.3cm}
	\begin{ejemplo}
		\textbf{Tabla de datos:} Una tabla organiza la información en filas
		y columnas \textbf{antes} de graficarla. Para leerla correctamente,
		primero identifica \textbf{qué representa cada fila} y \textbf{qué
			representa cada columna}.
	\end{ejemplo}

	\section*{Misión 2: Análisis de Gráficas de Barras}
	
	\begin{concepto}{Barras con Valores Positivos y Negativos}
		Una gráfica de barras también puede mostrar \textbf{números
			enteros}: las barras que suben desde la línea del cero representan
		\textbf{ganancias} (positivo) y las que bajan representan
		\textbf{pérdidas} (negativo). La línea del \textbf{cero} es siempre
		el punto de referencia para leer la gráfica.
	\end{concepto}

	\section*{Misión 3: Gráficas de Líneas (Tendencias Temporales)}
	
	\begin{concepto}{Lectura de una Gráfica de Líneas}
		En una gráfica de líneas, el eje \textbf{horizontal (X)}
		generalmente representa el \textbf{tiempo}, y el eje
		\textbf{vertical (Y)} representa la cantidad que cambia. La línea
		puede:
		\begin{itemize}
			\item \textbf{Subir} $\rightarrow$ el dato \textbf{aumenta}.
			\item \textbf{Bajar} $\rightarrow$ el dato \textbf{disminuye}.
			\item Mantenerse \textbf{horizontal}
			$\rightarrow$ el dato permanece \textbf{constante}.
		\end{itemize}
	\end{concepto}
	
	\vspace{0.3cm}

	\section*{Misión 4: Gráficas Circulares y Pictogramas}
	
	\begin{concepto}{Proporciones en una Gráfica Circular}
		En una gráfica circular, el círculo completo representa el
		\textbf{100\% del total}. Cada porción (o ``tajada'') representa un
		\textbf{porcentaje} de ese total. Para hallar \textbf{cuántos
			elementos} representa una porción, se multiplica el
		\textbf{porcentaje} (en forma decimal) por el \textbf{total de
			datos}.\\
			\textbf{Nota: La forma decimal es el valor del porcentaje dividido 100, es decir, 25\% en decimal es 0.25, 5\% es 0.05 y así con cualquier porcentaje}
	\end{concepto}
	
	\vspace{0.2cm}

	\begin{concepto}{Lectura de un Pictograma}
		En un pictograma, cada símbolo (o parte de él) representa una
		\textbf{cantidad fija}. Para hallar el total de una categoría, se
		\textbf{multiplica} el número de símbolos por el valor de cada
		símbolo. Medio símbolo representa la \textbf{mitad} de ese valor.\\
		\textbf{Nota: La forma decimal es el valor del porcentaje dividido 100, es decir, 25\% en decimal es 0.25, 5\% es 0.05 y así con cualquier porcentaje}
	\end{concepto}
