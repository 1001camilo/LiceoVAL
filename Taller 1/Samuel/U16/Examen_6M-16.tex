\newcommand{\drawTransversal}[1]{
	\draw[thick, <->] (-2.5, 1.5) -- (2.5, 1.5);
	\draw[thick, <->] (-2.5, 0) -- (2.5, 0);
	\coordinate (P) at (0,0);
	\coordinate (Q) at ({1.5/tan(#1)}, 1.5);
	\draw[thick, <->, >=stealth] ($(P) - ({1.8*cos(#1)}, {1.8*sin(#1)})$) -- ($(Q) + ({1.8*cos(#1)}, {1.8*sin(#1)})$);
}

\newcommand{\marcarTR}[4]{ 
	\draw[fill=#4!20, draw=#4, thick] (#1) -- ++(0.5,0) arc (0:#2:0.5) -- cycle;
	\node at ($(#1) + ({#2/2}:0.8)$) {#3};
}
\newcommand{\marcarTL}[4]{ 
	\draw[fill=#4!20, draw=#4, thick] (#1) -- ++(#2:0.5) arc (#2:180:0.5) -- cycle;
	\node at ($(#1) + ({(#2+180)/2}:0.8)$) {#3};
}
\newcommand{\marcarBL}[4]{ 
	\draw[fill=#4!20, draw=#4, thick] (#1) -- ++(180:0.5) arc (180:{180+#2}:0.5) -- cycle;
	\node at ($(#1) + ({180+#2/2}:0.8)$) {#3};
}
\newcommand{\marcarBR}[4]{ 
	\draw[fill=#4!20, draw=#4, thick] (#1) -- ++({180+#2}:0.5) arc ({180+#2}:360:0.5) -- cycle;
	\node at ($(#1) + ({270+#2/2}:0.8)$) {#3};
}
