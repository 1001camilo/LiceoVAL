	\section*{Misión 1: El Conjunto de Datos}
	
	\begin{concepto}{¿Qué es un conjunto de datos?}
		En estadística recopilamos \textbf{datos}: información numérica
		sobre algo que queremos estudiar.
		Un \textbf{conjunto de datos} es la colección organizada de todos
		esos valores.
		%\vspace{0.3cm}
		\begin{center}
			\begin{tikzpicture}[scale=0.9]
				\node[draw=valdark, fill=valgray, rounded corners=6pt,
				minimum width=13cm, minimum height=1.4cm,
				align=center, font=\bfseries, line width=1.5pt,
				text width=12.5cm] at (6,0)
				{Calificaciones de 10 estudiantes en un examen:\\[4pt]
					\textcolor{valblue}{\large $8,\; 6,\; 9,\; 8,\; 7,\; 10,\; 8,\; 6,\; 9,\; 8$}};
				\node[font=\small\bfseries, text=valblue] at (6,-1.2)
				{Este es el \textbf{conjunto de datos} (10 valores)};
			\end{tikzpicture}
		\end{center}
		
	%	\newpage
		\textbf{Tabla de frecuencias:} organiza los datos mostrando cuántas
		veces aparece cada valor. La columna ``Frecuencia'' indica cuántas
		veces se repite cada dato.
		
		%\vspace{0.3cm}
		\begin{center}
			\renewcommand{\arraystretch}{1}
			\begin{tabular}{|c|c|c|}
				\hline
				\rowcolor{valblue!25}
				\textbf{Valor (dato)}  & \textbf{Frecuencia(cuantas veces se repite)} \\
				\hline
				6   & 2 \\
				\hline
				7  & 1 \\
				\hline
				8  & 4 \\
				\hline
				9  & 2 \\
				\hline
				10 & 1 \\
				\hline
				\rowcolor{valblue!10}
				\textbf{Total} & \textbf{10} \\
				\hline
			\end{tabular}
		\end{center}
	\end{concepto}

	\section*{Misión 2: La Moda}
	
	\begin{concepto}{¿Qué es la Moda?}
		La \textbf{moda} \large{($\hat{x}$)}, es el valor que \textbf{más se repite}
		en un conjunto de datos.
		
		%\vspace{0.3cm}
		\begin{center}
			\begin{tikzpicture}[scale=0.85]
				% Datos visuales con cajas
				\node[datom] at (0,0)   {8};
				\node[dato] at (1.1,0) {6};
				\node[datom] at (2.2,0) {8};
				\node[dato] at (3.3,0) {7};
				\node[datom] at (4.4,0) {8};
				\node[dato] at (5.5,0) {6};
				\node[datom] at (6.6,0) {8};
				\node[dato] at (7.7,0) {9};
				% Flecha y etiqueta
				\draw[-{Stealth}, valred, line width=2pt]
				(2.2,-1.0) -- (2.2,-0.6);
				\draw[-{Stealth}, valred, line width=2pt]
				(0,-1.0) -- (0,-0.6);
				\draw[-{Stealth}, valred, line width=2pt]
				(4.4,-1.0) -- (4.4,-0.6);
				\draw[-{Stealth}, valred, line width=2pt]
				(6.7,-1.0) -- (6.7,-0.6);
				\node[font=\large\bfseries, text=valred] at (3.3,-1.6)
				{Moda $= \mathbf{8}$ \small(aparece 4 veces)};
				% Leyenda cajas
				%%\node[datom, minimum width=0.6cm, minimum height=0.6cm] at (8.8,0) {};
			\end{tikzpicture}
		\end{center}
		
		%\vspace{0.3cm}
		\begin{tcolorbox}[enhanced, colback=vallightorange,
			colframe=valorange, arc=3pt]
			\textbf{Casos especiales:}
			\begin{itemize}[itemsep=2pt]
				\item \textbf{Sin moda:} todos los valores aparecen el mismo número
				de veces.
				\item \textbf{Dos modas (bimodal):} dos valores comparten la mayor
				frecuencia.
				\item \textbf{Tres o más modas:} varios valores empatan en frecuencia.
			\end{itemize}
		\end{tcolorbox}
	\end{concepto}
	
	\begin{ejemplo}
		\textbf{Ejemplo 1 — Una sola moda:}
		
		Datos: $5, 3, 7, 3, 9, 3, 7, 5, 3$
		
		Frecuencias: \quad $3 \to 4$ veces\quad $5 \to 2$ veces
		\quad $7 \to 2$ veces\quad $9 \to 1$ vez
		
		\textbf{Moda $= 3$} (el que más se repite)
		
		%\vspace{0.4cm}
		\textbf{Ejemplo 2 — Dos modas:}
		
		Datos: $2, 5, 8, 5, 2, 9, 2, 5, 1$
		
		Frecuencias: \quad $2 \to 3$ veces\quad $5 \to 3$ veces
		(empatan)\quad $8, 9, 1 \to 1$ vez cada uno
		
		\textbf{Moda $= 2$ y $5$} (ambos con la mayor frecuencia)
		
		%\vspace{0.4cm}
		\textbf{Ejemplo 3 — Sin moda:}
		
		Datos: $1, 2, 3, 4, 5$ (cada uno aparece 1 vez)
		
		\textbf{No hay moda.}
	\end{ejemplo}
	
	\begin{practica}{La Moda}
		
		\textbf{A) Encuentra la moda. Muestra la frecuencia de cada valor:}
		
		%\vspace{0.2cm}
		\begin{multicols}{2}
			\begin{enumerate}[label=\textbf{\arabic*.}, itemsep=0.85cm]
				\item $4, 7, 4, 9, 4, 7, 2$
				\begin{espaciotrabajonb}{}

	\section*{Misión 3: La Mediana}
	
	\begin{concepto}{¿Qué es la Mediana?}
		La \textbf{mediana} ($\tilde{x}$) es el valor que queda en el \textbf{centro}
		cuando los datos están ordenados de menor a mayor.
		La mediana \textit{divide} el conjunto en dos mitades iguales.
		
		%\vspace{0.3cm}
		\textbf{Proceso en 2 pasos:}
		
		\begin{enumerate}[label=\textcolor{valblue}{\bfseries Paso \arabic*.}]
			\item \textbf{Ordena} los datos de menor a mayor.
			\item \textbf{Ubica el valor central.}
		\end{enumerate}
		
		%\vspace{0.3cm}
		\begin{tcolorbox}[enhanced, colback=vallightblue,
			colframe=valblue, arc=3pt]
			\centering\bfseries
			Si el número de datos es \textbf{impar}:
			la mediana es el valor del \textbf{medio}.\\[6pt]
			Si el número de datos es \textbf{par}:
			la mediana es el \textbf{promedio} de los dos valores centrales.
		\end{tcolorbox}
		
		%\vspace{0.3cm}
		\textbf{Número impar de datos — Ejemplo:}\quad
		$3,\,7,\,2,\,9,\,5\;\to\;2,\,3,\,\mathbf{5},\,7,\,9$\quad Me $=\mathbf{5}$
		
		\begin{center}
			\begin{tikzpicture}[scale=0.9]
				\node[dato] at (0,0)   {2};
				\node[dato] at (1.1,0) {3};
				\node[datoM] at (2.2,0) {5};
				\node[dato] at (3.3,0) {7};
				\node[dato] at (4.4,0) {9};
				\draw[<->, valgreen, line width=1.5pt]
				(0,-0.7) -- (2.0,-0.7)
				node[midway, below, font=\small\bfseries, text=valgreen] {2 datos};
				\draw[<->, valgreen, line width=1.5pt]
				(2.4,-0.7) -- (4.4,-0.7)
				node[midway, below, font=\small\bfseries, text=valgreen] {2 datos};
				\draw[-{Stealth}, valgreen, line width=2pt]
				(2.2,-1.4) -- (2.2,-0.6);
				\node[font=\bfseries, text=valgreen] at (2.2,-1.7)
				{Mediana $= 5$};
			\end{tikzpicture}
		\end{center}
		
		%\vspace{0.3cm}
		\textbf{Número par de datos — Ejemplo:}\quad
		$4,\,8,\,2,\,6\;\to\;2,\,\mathbf{4},\,\mathbf{6},\,8$\quad Me $= \dfrac{4+6}{2} = \dfrac{10}{2} = \mathbf{5}$
	\end{concepto}
	
	\begin{ejemplo}
		\textbf{Ejemplo 1 (impar):}
		Edades: $12, 9, 15, 11, 13$
		
		Paso 1 (ordenar): $9, 11, \mathbf{12}, 13, 15$\quad (5 datos)
		
		Paso 2 (centro): el 3.\textsuperscript{er} valor es \textbf{12}.
		
		\textbf{Mediana $= 12$ años}
		
		%\vspace{0.3cm}
		\textbf{Ejemplo 2 (par):}
		Puntos: $6, 10, 4, 8, 2, 12$
		
		Paso 1 (ordenar): $2, 4, \mathbf{6, 8}, 10, 12$\quad (6 datos)
		
		Paso 2 (centro): los valores 3.\textsuperscript{o} y 4.\textsuperscript{o} son 6 y 8.
		
		Me $= \dfrac{6+8}{2} = \dfrac{14}{2} = \mathbf{7}$
		
		\textbf{Mediana $= 7$ puntos}
	\end{ejemplo}

	\section*{Misión 4: El Rango}
	
	\begin{concepto}{¿Qué es el Rango?}
		El \textbf{rango} (R) mide la \textbf{amplitud} o \textbf{dispersión}
		de los datos: qué tanto se separan entre el mayor y el menor.
		
		%\vspace{0.3cm}
		\begin{center}
			\begin{tcolorbox}[enhanced, colback=valred!5, colframe=valred,
				arc=3pt, width=0.7\textwidth]
				\centering\Large\bfseries
				$\text{Rango} = \text{Valor máximo} - \text{Valor mínimo}$
			\end{tcolorbox}
		\end{center}
		
		%\vspace{0.4cm}
		\begin{center}
			\begin{tikzpicture}[scale=0.9]
				% Recta numérica
				\draw[-{Stealth}, valdark, line width=1.5pt] (-0.5,0) -- (11,0);
				% Marcas
				\foreach \x/\v in {1/2, 3/5, 5/7, 7/9, 9/14} {
					\draw[valdark, line width=1pt] (\x,0.15) -- (\x,-0.15);
					\node[below, font=\small\bfseries] at (\x,-0.15) {\v};
				}
				% Resaltados
				\fill[valred]   (1,0) circle (5pt) node[above, font=\small\bfseries, text=valred] {mín $=2$};
				\fill[valblue]  (9,0) circle (5pt) node[above, font=\small\bfseries, text=valblue]{máx $=14$};
				% Flecha doble
				\draw[<->, valred, line width=2pt] (1,-0.7) -- (9,-0.7)
				node[midway, below, font=\bfseries, text=valred]
				{Rango $= 14-2 = \mathbf{12}$};
			\end{tikzpicture}

	\section*{Misión 5: Probabilidad Básica}
	
	\begin{concepto}{¿Qué es la Probabilidad?}
		La \textbf{probabilidad} mide qué tan \textbf{probable} es que
		ocurra un evento en un experimento. Se expresa como una
		\textbf{fracción, decimal o porcentaje} entre 0 y 1.
		
		%\vspace{0.4cm}
		\begin{center}
			\begin{tcolorbox}[enhanced, colback=valpurple!6, colframe=valpurple,
				arc=3pt, width=0.75\textwidth]
				\centering\Large\bfseries
				$P(\text{evento}) = \dfrac{\text{casos favorables}}{\text{casos totales posibles}}$
			\end{tcolorbox}
		\end{center}
		
		%\vspace{0.4cm}
		\textbf{Escala de probabilidad:}
		
		\begin{center}
			\begin{tikzpicture}[scale=1.0]
				% Línea
				\draw[-{Stealth}, valdark, line width=2pt] (-0.5,0) -- (11.5,0);
				% Puntos clave
				\fill[valred]    (0,0)  circle (7pt);
				\fill[valorange] (5,0)  circle (7pt);
				\fill[valgreen]  (10,0) circle (7pt);
				% Etiquetas arriba
				\node[font=\bfseries, text=valred,    above] at (0,0.2)  {$0$};
				\node[font=\bfseries, text=valorange, above] at (5,0.2)  {$\dfrac{1}{2}$};
				\node[font=\bfseries, text=valgreen,  above] at (10,0.2) {$1$};
				% Etiquetas abajo
				\node[font=\small\bfseries, text=valred,    below, text width=2.8cm, align=center]
				at (0,-0.35)    {IMPOSIBLE\\(nunca ocurre)};
				\node[font=\small\bfseries, text=valorange, below, text width=3cm, align=center]
				at (5,-0.35)    {IGUALMENTE\\PROBABLE};
				\node[font=\small\bfseries, text=valgreen,  below, text width=2.8cm, align=center]
				at (10,-0.35)   {SEGURO\\(siempre ocurre)};
				% Gradiente texto
				\node[font=\small, text=valdark] at (2.5,0.8)  {poco probable};
				\node[font=\small, text=valdark] at (7.5,0.8)  {muy probable};
			\end{tikzpicture}
		\end{center}
		
		%\vspace{0.4cm}
		\textbf{Ejemplo con un dado (6 caras: 1, 2, 3, 4, 5, 6):}
		
		\begin{center}
			\renewcommand{\arraystretch}{1.6}
			\begin{tabular}{|m{5cm}|m{2.5cm}|m{3.5cm}|}
				\hline
				\rowcolor{valpurple!15}
				\textbf{Evento} & \textbf{Fracción} & \textbf{Decimal} \\
				\hline
				Sacar el número 3 &
				$\dfrac{1}{6}$ & $\approx 0.17$ \\
				\hline
				Sacar un número par (2,4,6) &
				$\dfrac{3}{6} = \dfrac{1}{2}$ & $0.5$ \\
				\hline
				Sacar un número $> 0$ &
				$\dfrac{6}{6} = 1$ & $1$ \\
				\hline
				Sacar el número 9 &
				$\dfrac{0}{6} = 0$ & $0$ \\
				\hline
			\end{tabular}
		\end{center}
	\end{concepto}
	
	\begin{ejemplo}
		\textbf{Una bolsa contiene:} 3 bolas rojas, 2 bolas azules
		y 5 bolas verdes. Total: 10 bolas.
