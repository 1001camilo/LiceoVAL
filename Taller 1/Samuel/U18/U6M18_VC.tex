	\section*{Misión 1: La Media (Promedio) y la Media Ponderada}
	
	\begin{concepto}{¿Qué es la Media o Promedio?}
		La \textbf{media} (también llamada \textbf{promedio} o \textbf{promedio aritmético},
		símbolo $\bar{x}$) es el valor central que representa a un conjunto de datos.
		Se obtiene dividiendo la \textbf{suma de todos los valores} entre la
		\textbf{cantidad de datos}.
		
		\vspace{0.3cm}
		\begin{center}
			\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue,
				arc=3pt, width=0.82\textwidth]
				\centering\Large\bfseries
				$$\bar{x} = \dfrac{\text{Suma de todos los valores}}{n}$$
				$$\bar{x}=\dfrac{x_1 + x_2 + x_3 + \cdots + x_n}{n}$$\\[6pt]
				\normalsize\normalfont donde $n$ es la cantidad total de datos.
			\end{tcolorbox}
		\end{center}
		
		\vspace{0.4cm}
		Imagina que todos los valores ``se redistribuyen''
		de manera igual entre todos los datos.
		
		\begin{center}
			\begin{tikzpicture}[scale=0.9]
				\node[dato] at (0,0)    {3};
				\node[dato] at (1.2,0)  {7};
				\node[dato] at (2.4,0)  {5};
				\node[dato] at (3.6,0)  {9};
				\node[dato] at (4.8,0)  {6};
				\draw[-{Stealth}, valblue, line width=2pt](6.2,0)--(7.2,0)
				node[midway, above, font=\small\bfseries]{$\div 5$};
				\node[datoM, minimum width=2cm] at (9.5,0)
				{$\bar{x}=6$};
				\node[font=\small, text=valdark] at (2.4,-0.9)
				{Suma: $3+7+5+9+6=30$};
				\node[font=\small, text=valgreen] at (9.5,-0.9)
				{Media: $30\div 5=6$};
			\end{tikzpicture}
		\end{center}
	\end{concepto}
	
	\begin{ejemplo}
		\textbf{Millaje de una flotilla:}\\
		Una empresa de transporte registra el rendimiento de combustible
		(km/L) de sus 7 vehículos:
		\[12{,}4 \quad 10{,}8 \quad 13{,}2 \quad 11{,}5 \quad 12{,}9 \quad 10{,}3 \quad 11{,}7\]
		
		\textbf{Paso 1 — Sumar:}
		$12{,}4 + 10{,}8 + 13{,}2 + 11{,}5 + 12{,}9 + 10{,}3 + 11{,}7 = \textcolor{red}{82{,}8}$
		
		\textbf{Paso 2 — Dividir entre $n=\textcolor{blue}{7}$:}
		$\bar{x} = \dfrac{\textcolor{red}{82{,}8}}{\textcolor{blue}{7}} \approx \mathbf{11{,}83}$ km/L
		
		\textbf{Interpretación:} en promedio, cada vehículo rinde $11{,}83$ km por litro.
		
		\vspace{0.5cm}
		\textbf{Calificaciones del semestre:}\\
		Las notas de un estudiante en 5 materias fueron:
		$8{,}5 \;|\; 7{,}0 \;|\; 9{,}0 \;|\; 6{,}5 \;|\; 8{,}0$
		
		$\bar{x} = \dfrac{8{,}5 + 7{,}0 + 9{,}0 + 6{,}5 + 8{,}0}{5}
		= \dfrac{39{,}0}{5} = \mathbf{7{,}80}$
	\end{ejemplo}
	
	\begin{concepto}{Media Ponderada}
		En la \textbf{media ponderada} (o promedio ponderado), cada valor tiene
		un \textbf{peso} (importancia o cantidad de veces que influye).
		Se usa cuando no todos los datos ``valen lo mismo''.
		
		\vspace{0.3cm}
		\begin{center}
			\begin{tcolorbox}[enhanced, colback=vallightorange, colframe=valorange,
				arc=3pt, width=0.85\textwidth]
				\centering\Large\bfseries
				$\bar{x}_p = \dfrac{v_1 \cdot w_1 + v_2 \cdot w_2 + \cdots + v_n \cdot w_n}
				{w_1 + w_2 + \cdots + w_n}$\\[6pt]
				\normalsize\normalfont
				donde $v_i$ = valor, $w_i$ = peso (ponderación).
			\end{tcolorbox}
		\end{center}
		
		\vspace{0.3cm}
		\textbf{¿Cuándo se usa?}
		Cuando las materias tienen diferente número de créditos, cuando un
		examen vale más que una tarea, o cuando algunos productos se compran
		en mayor cantidad que otros.
	\end{concepto}
	\newpage
	\begin{ejemplo}
		\textbf{Calificaciones del semestre con créditos (media ponderada):}
		
		\vspace{0.2cm}
		\begin{center}
			\renewcommand{\arraystretch}{1.6}
			\begin{tabular}{|m{4cm}|c|c|c|}
				\hline
				\rowcolor{valblue!20}
				\textbf{Materia} & \textbf{Nota} & \textbf{Créditos} &
				\textbf{Nota $\times$ Créditos} \\
				\hline
				Matemáticas  & \textcolor{blue}{8{,}5} & \textcolor{red}{4} & $\textcolor{blue}{8{,}5} \times \textcolor{red}{4} = 34{,}0$ \\
				\hline
				Español      & \textcolor{blue}{7{,}0} & \textcolor{red}{3} & $\textcolor{blue}{7{,}0} \times \textcolor{red}{3} = 21{,}0$ \\
				\hline
				Ciencias     & \textcolor{blue}{9{,}0} & \textcolor{red}{3} & $\textcolor{blue}{9{,}0} \times \textcolor{red}{3} = 27{,}0$ \\
				\hline
				Inglés       & \textcolor{blue}{6{,}5} & \textcolor{red}{2} & $\textcolor{blue}{6{,}5} \times \textcolor{red}{2} = 13{,}0$ \\
				\hline
				Artes        & \textcolor{blue}{8{,}0} & \textcolor{red}{1} & $\textcolor{blue}{8{,}0} \times \textcolor{red}{1} = \phantom{1}8{,}0$ \\
				\hline
				\rowcolor{valblue!10}
				\textbf{Total} & --- & \textbf{\textcolor{orange}{13}} & \textbf{\textcolor{brown}{103{,}0}} \\
				\hline
			\end{tabular}
		\end{center}
		
		$$\bar{x}_p = \dfrac{\textcolor{brown}{103{,}0}}{\textcolor{orange}{13}} \approx \mathbf{7{,}92}$$
		
		\textbf{Diferencia con la media aritmética} si calculamos la media simple de las 5 notas:
		$\dfrac{8{,}5+7{,}0+9{,}0+6{,}5+8{,}0}{5} = \dfrac{39{,}0}{5} = 7{,}80$\\
		La media ponderada (7,92) es mayor porque Matemáticas y Ciencias
		(notas más altas) tienen más créditos.
	\end{ejemplo}

	\section*{Misión 2: El Conjunto de Datos y Tablas de Frecuencia}
	
	\begin{concepto}{¿Qué es un conjunto de datos?}
		En estadística recopilamos \textbf{datos}: información numérica sobre algo
		que queremos estudiar. Un \textbf{conjunto de datos} es la colección
		organizada de todos esos valores.
		
		\vspace{0.3cm}
		\begin{center}
			\begin{tikzpicture}[scale=0.9]
				\node[draw=valdark, fill=valgray, rounded corners=6pt,
				minimum width=13cm, minimum height=1.4cm, align=center,
				font=\bfseries, line width=1.5pt, text width=12.5cm] at (6,0)
				{Calificaciones de 10 estudiantes en un quiz:\\[4pt]
					\textcolor{valblue}{\large $8;\; 6;\; 9;\; 8;\; 7;\; 10;\; 8;\; 6;\; 9;\; 8$}};
				\node[font=\small\bfseries, text=valblue] at (6,-1.2)
				{Este es el \textbf{conjunto de datos} (10 valores)};
			\end{tikzpicture}
		\end{center}
		
		\vspace{0.3cm}
		\textbf{Tabla de frecuencias:} organiza cuántas veces aparece cada valor.
		\vspace{0.2cm}
		\begin{center}
			\renewcommand{\arraystretch}{1.5}
			\begin{tabular}{|c|c|}
				\hline
				\rowcolor{valblue!25}
				\textbf{Valor} & \textbf{Frecuencia (Cuantas veces aparece)} \\
				\hline
				6  & 2 \\
				\hline
				7  & 1 \\
				\hline
				8  & 4 \\
				\hline
				9  & 2 \\
				\hline
				10 & 1 \\
				\hline
				\rowcolor{valblue!10}
				\textbf{Total} & \textbf{10} \\
				\hline
			\end{tabular}
		\end{center}
		
		%\vspace{0.2cm}
		La suma de todas las frecuencias siempre debe ser igual al total de datos.
	\end{concepto}
	
	\begin{ejemplo}
		\textbf{Datos:} colores de camisetas de 12 estudiantes:
		\textcolor{red}{Rojo}, \textcolor{blue}{Azul}, \textcolor{ForestGreen}{Verde}, \textcolor{blue}{Azul}, \textcolor{red}{Rojo}, \textcolor{blue}{Azul}, \textcolor{ForestGreen}{Verde}, \textcolor{blue}{Azul}, \textcolor{red}{Rojo}, \textcolor{blue}{Azul}, \textcolor{ForestGreen}{Verde}, \textcolor{blue}{Azul}.
		
		\begin{center}
			\renewcommand{\arraystretch}{1.5}
			\begin{tabular}{|m{3cm}|m{2.5cm}|}
				\hline
				\rowcolor{valblue!20}\textbf{Color} & \textbf{Frecuencia} \\
				\hline

				\hline
				Azul  & 6 \\
				\hline
				Verde & 3 \\

				\hline

		\end{center}
		
		\textbf{Verificar:} $3+6+3=12$ \checkmark
	\end{ejemplo}

	\section*{Misión 4: La Mediana}
	
	\begin{concepto}{¿Qué es la Mediana?}
		La \textbf{mediana} ($\tilde{x}$) es el valor central de un conjunto de datos
		\textbf{ordenado de menor a mayor}. Divide el conjunto en dos mitades
		exactamente iguales.
		
		\vspace{0.3cm}
		\begin{center}
			\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue,
				arc=3pt, width=0.82\textwidth]
				\centering\bfseries
				\textbf{Número impar de datos:} la mediana es el valor del centro.\\[6pt]
				\textbf{Número par de datos:} la mediana es la media de los
				dos valores centrales.
			\end{tcolorbox}
		\end{center}
		
		\vspace{0.3cm}
		\textbf{Proceso:}
		\begin{enumerate}[label=\textcolor{valblue}{\bfseries Paso \arabic*.}]
			\item \textbf{Ordena} los datos de menor a mayor.
			\item \textbf{Ubica el valor central} (o calcula la media de los dos centrales).
		\end{enumerate}
		
		\vspace{0.3cm}
		\textbf{Impar — ejemplo:}\quad
		$3,\,7,\,2,\,9,\,5\;\to\;2,\,3,\,\mathbf{5},\,7,\,9$\quad $\tilde{x}=\mathbf{5}$
		
		\begin{center}
			\begin{tikzpicture}[scale=0.9]
				\node[dato] at (0,0)    {2};
				\node[dato] at (1.1,0)  {3};
				\node[datoM] at (2.2,0) {5};
				\node[dato] at (3.3,0)  {7};
				\node[dato] at (4.4,0)  {9};
				\draw[<->, valgreen, line width=1.5pt](0,-0.7)--(2.0,-0.7)
				node[midway, below, font=\small\bfseries, text=valgreen]{2 datos};
				\draw[<->, valgreen, line width=1.5pt](2.4,-0.7)--(4.4,-0.7)
				node[midway, below, font=\small\bfseries, text=valgreen]{2 datos};
				\draw[-{Stealth}, valgreen, line width=2pt](2.2,-1.4)--(2.2,-0.6);
				\node[font=\bfseries, text=valgreen] at (2.2,-1.7){Mediana $=5$};
			\end{tikzpicture}
		\end{center}
		
		\vspace{0.3cm}
		\textbf{Par — ejemplo:}\quad
		$4,\,8,\,2,\,6\;\to\;2,\,\mathbf{\textcolor{ForestGreen}{4}},\,\mathbf{\textcolor{ForestGreen}{6}},\,8$\quad
		$\tilde{x}=\dfrac{4+6}{2}=\dfrac{10}{2}=\mathbf{5}$
	\end{concepto}
	
	\begin{ejemplo}
		\textbf{Impar:} Edades: $12;\;9;\;15;\;11;\;13$
		
		Ordenadas: $9,\;11,\;\mathbf{12},\;13,\;15$\quad \textbf{$\tilde{x}=12$ años}
		
		\vspace{0.3cm}
		\textbf{Par:} Notas: $6;\;10;\;4;\;8;\;2;\;12$
		
		Ordenadas: $2,\;4,\;\mathbf{6,\;8},\;10,\;12$\quad
		$\tilde{x}=\dfrac{6+8}{2}=\dfrac{14}{2}=\mathbf{7}$
		
		\vspace{0.3cm}
		\textbf{Diferencia entre Media y Mediana:}
		Si los datos son: $1;\;2;\;3;\;4;\;100$\\
		Media $= \dfrac{110}{5}=22$ \quad Mediana $= 3$\\
		La mediana no se ``distorsiona'' por valores extremos como la media.
	\end{ejemplo}

	\section*{Misión 5: El Rango}
	
	\begin{concepto}{¿Qué es el Rango?}
		El \textbf{rango} (R) mide la \textbf{amplitud} o \textbf{dispersión} de los datos:
		qué tanta diferencia existe entre el valor mayor y el menor.
		
		\vspace{0.3cm}
		\begin{center}
			\begin{tcolorbox}[enhanced, colback=valred!5, colframe=valred,
				arc=3pt, width=0.65\textwidth]
				\centering\Large\bfseries
				$\text{Rango} = \text{Valor máximo} - \text{Valor mínimo}$
			\end{tcolorbox}
		\end{center}
		
		\vspace{0.4cm}
		\begin{center}
			\begin{tikzpicture}[scale=0.9]
				\draw[-{Stealth}, valdark, line width=1.5pt](-0.5,0)--(11,0);
				\draw[valdark, line width=1pt](1,0.15)--(1,-0.15)
				node[below, font=\small\bfseries]{2};
				\draw[valdark, line width=1pt](3,0.15)--(3,-0.15)
				node[below, font=\small\bfseries]{5};
				\draw[valdark, line width=1pt](5,0.15)--(5,-0.15)
				node[below, font=\small\bfseries]{7};
				\draw[valdark, line width=1pt](7,0.15)--(7,-0.15)
				node[below, font=\small\bfseries]{9};
				\draw[valdark, line width=1pt](9,0.15)--(9,-0.15)
				node[below, font=\small\bfseries]{14};
				\fill[valred]  (1,0) circle(5pt)
				node[above, font=\small\bfseries, text=valred]{mín $=2$};
				\fill[valblue] (9,0) circle(5pt)
				node[above, font=\small\bfseries, text=valblue]{máx $=14$};
				\draw[<->, valred, line width=2pt](1,-0.7)--(9,-0.7)
				node[midway, below, font=\bfseries, text=valred]
				{Rango $=14-2=\mathbf{12}$};
			\end{tikzpicture}
		\end{center}
		
		\vspace{0.3cm}
		Un rango \textit{grande} indica datos muy dispersos.
		Un rango \textit{pequeño} indica datos agrupados (más constantes).
	\end{concepto}

	\begin{concepto}{Tipos de Eventos}
		\vspace{0.2cm}
		\renewcommand{\arraystretch}{1.7}
		\begin{center}
			\begin{tabular}{|m{3cm}|m{4.5cm}|m{4.5cm}|}
				\hline
				\rowcolor{valblue!25}
				\textbf{Tipo de evento} & \textbf{Definición} & \textbf{Ejemplo} \\
				\hline
				\textbf{Certero / Seguro} &
				Siempre ocurre. $P=1$ &
				Que al lanzar una moneda caiga cara o sello. \\
				\hline
				\textbf{Imposible} &
				Nunca puede ocurrir. $P=0$ &
				Que un dado de 6 caras muestre el número 9. \\
				\hline
				\textbf{Probable} &
				Alta probabilidad de ocurrir. $P$ cercana a 1 &
				Sacar rojo de una bolsa con 9 rojas y 1 azul. \\
				\hline
				\textbf{Improbable} &
				Baja probabilidad de ocurrir. $P$ cercana a 0 &
				Sacar azul de esa misma bolsa ($P=\dfrac{1}{10}$). \\
				\hline
			\end{tabular}
		\end{center}
	\end{concepto}
