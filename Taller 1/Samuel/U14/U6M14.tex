	\section*{Misión 1: ¿Qué es una Ecuación?}
	
	\begin{concepto}{Ecuación, Solución y Verificación}
		Una \textbf{ecuación} es un enunciado matemático que indica que dos
		expresiones son \textbf{iguales}. \textbf{Todas} las ecuaciones contienen un símbolo ``$=$'' que separa dos lados, el lado izquierdo \textbf{(LI)} y el lado derecho \textbf{(LD)}.\\
		La letra $x$ es la \textbf{variable} (o \textbf{incógnita}):
		\textbf{NOTA:} Los lados de la ecuación pueden \textbf{invertirse}, por lo que puede escribirse cómo:
		$$x+3=8$$
		\begin{center}
			ó
		\end{center}
		$$8=x+3$$
		
		\vspace{0.4cm}
		\begin{center}
			\begin{tikzpicture}
				\node[font=\LARGE\bfseries] (eq) at (5,0) {$x + 3 \;=\; 8$};
				\draw[-{Stealth}, valblue, line width=2pt]
				(1.8, 0.7) -- (3.2, 0.3)
				node[pos=0, above, align=center, font=\small\bfseries, valblue]
				{LADO\\IZQUIERDO};\draw[thick,draw=valblue] (4.2,0) ellipse (1cm and 0.5cm);
				\draw[thick,draw=valorange] (6.5,0) ellipse (0.5cm and 0.5cm);
				\draw[-{Stealth}, valorange, line width=2pt]
				(8.2, 0.7) -- (7.2, 0.3)
				node[pos=0, above, align=center, font=\small\bfseries, valorange]
				{LADO\\DERECHO};
				\draw[-{Stealth}, valred, line width=2pt]
				(3.5, -1.3) -- (3.5, -0.3)
				node[pos=0, below, align=center, font=\small\bfseries, valred]
				{VARIABLE\\(incógnita)};
			\end{tikzpicture}
		\end{center}
		\vspace{0.3cm}
		
		\begin{itemize}
			\item Una ecuación puede ser \textbf{verdadera}: $6+3=9$
			\item Una ecuación puede ser \textbf{falsa}: $2+4=7$
			\item Una ecuación puede no ser \textbf{verdadera} ni \textbf{falsa}. Por ejemplo, $x+5=15$ no es verdadera ni falsa debido a que no se sabe qué número representa \textbf{$x$}
		\end{itemize}
		
		\vspace{0.3cm}
		Una ecuación se hace verdadera o falsa sustituyendo la variable por un número. Si se sustituye $x$ por $5$ en $x+3=8$ la ecuación es \textbf{verdadera}. Si se sustituye $x$ por $4$, la ecuación resultante es \textbf{falsa}.\\
		
		
		\textbf{Solución:}\\
		
		El número que al \textbf{sustituir} la variable hace que la
		ecuación sea \textbf{verdadera}. Se dice que ese número \textit{satisface} la ecuación y se le llama \textbf{solución}.\\
		
		A continuación se muestra en la tabla la sustitución de $x$ por 5 y después por 4, para saber si son soluciones o no.
		
		\vspace{0.3cm}
		\begin{center}
			\renewcommand{\arraystretch}{1.5}
			\begin{tabular}{|m{3cm}|m{4.5cm}|m{2.5cm}|}
				\hline
				\rowcolor{valblue!20}
				\textbf{Ecuación} & \textbf{Verificación} & \textbf{¿Solución?} \\
				\hline
				$x + 3 = 8$ \newline con $x = 5$ &
				$5 + 3 = 8$ \newline $8 = 8$ \checkmark &
				\textcolor{valgreen}{\textbf{SÍ}} \\
				\hline
				$x + 3 = 8$ \newline con $x = 4$ &
				$4 + 3 = 8$ \newline $7 \neq 8$ \texttimes &
				\textcolor{valred}{\textbf{NO}} \\
				\hline
			\end{tabular}
		\end{center}
	\end{concepto}

		\textbf{C) Completa el concepto:}
		
		\vspace{0.2cm}
		\begin{enumerate}[label=\textbf{\arabic*.}, itemsep=0.4cm, start=13]
			\item Cualquier número que haga una ecuación verdadera al sustituir la variable se llama \underline{\hspace{4cm}}.
			\item El símbolo $=$ separa el \underline{\hspace{3cm}} izquierdo del \underline{\hspace{3cm}} derecho.
			\item Si sustituyo $x = 3$ en $x + 2 = 5$ obtengo $5 = 5$, lo cual es \underline{\hspace{2cm}}, por lo tanto $x = 3$ \underline{\hspace{3cm}} (es/no es) una solución.
			\item Una ecuación puede ser verdadera o \underline{\hspace{3cm}} según el valor que se asigne a la variable.
			\item La variable en una ecuación también se llama \underline{\hspace{3cm}}.
		\end{enumerate}

	\section*{Misión 2: Propiedad Uniforme de la Suma}
	
	\begin{concepto}{Resolver una ecuación}
		Relosver una ecuación significa encontrar \textbf{todos} los valores de la variable que hacen \textbf{verdadera} la ecuación.
	\end{concepto}
	
	\begin{concepto}{Deshacer una Resta — Suma igual a ambos lados}
		Para resolver ecuaciones del tipo $x - a = b$, se \textbf{suma} el mismo
		número a \textbf{ambos lados} para mantener el equilibrio en la balanza, entre el lado izquiero y el lado derecho, los dos deben seguir "pesando" lo mismo:
		
		\vspace{0.4cm}
		\begin{center}
			\begin{tikzpicture}[scale=0.92]
				% Balance beam
				\draw[valdark, line width=3pt] (0,2.4) -- (10,2.4);
				\fill[valdark] (5,2.2) -- (4.4,0.8) -- (5.6,0.8) -- cycle;
				\draw[valdark, line width=2pt] (4,0.8) -- (6,0.8) -- (6,0.5) -- (4,0.5) -- cycle;
				% Left pan
				\draw[valblue, line width=2pt] (1.5,2.4) -- (1.5,1.4);
				\draw[valblue, line width=2pt] (0.3,1.4) -- (2.7,1.4);
				\draw[valblue, line width=2pt] (0.3,1.4) -- (0.3,0.6) -- (2.7,0.6) -- (2.7,1.4);
				\node[font=\large\bfseries, valblue] at (1.5,1.0) {$x - 2$};
				% Right pan
				\draw[valorange, line width=2pt] (8.5,2.4) -- (8.5,1.4);
				\draw[valorange, line width=2pt] (7.3,1.4) -- (9.7,1.4);
				\draw[valorange, line width=2pt] (7.3,1.4) -- (7.3,0.6) -- (9.7,0.6) -- (9.7,1.4);

				% Arrow and action

				{$+2$ a ambos lados: \quad $x - 2 + 2 = 3 + 2$ \quad $\Rightarrow$ \quad $x = 5$};
			\end{tikzpicture}
		\end{center}
		
		\vspace{0.4cm}
		\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue, arc=3pt]
			\centering\large\bfseries
			Sumar el mismo número a ambos lados de la ecuación no cambia su solución, para cualesquiera números a,b,c.\vspace{0.5cm}
			
			Si $a = b$, entonces $a + c = b + c$\\[4pt]
			\normalsize\normalfont

		\end{tcolorbox}
	\end{concepto}

	\section*{Misión 3: Propiedad Uniforme de la Resta}
	
	\begin{concepto}{Deshacer una Suma — Resta igual a ambos lados}
		Para resolver ecuaciones del tipo $x + a = b$, se \textbf{resta} el mismo
		número a \textbf{ambos lados}:
		
		\vspace{0.3cm}
		\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue, arc=3pt]
			\centering\large\bfseries
			Si $a = b$, entonces $a - c = b - c$\\[4pt]
			\normalsize\normalfont
			Para despejar $x$ en $x + a = b$:\; resta $a$ de \textbf{ambos} lados $\rightarrow$ $x = b - a$
		\end{tcolorbox}
		
		\vspace{0.3cm}
		\textbf{Ejemplo:} $x + 3 = 5$ \hfill
		$x + 3 - 3 = 5 - 3$ \hfill $x = 2$
	\end{concepto}
	
	\begin{atencion}{¡Atención! La variable puede quedar en cualquier lado}
		Si al resolver obtienes $-12 = y$ , esa respuesta es \textbf{válida}
		y equivalente a $y = -12$. La variable no tiene que estar siempre a la izquierda del igual.\\
		Ejemplo: $8 + n = 3 \;\rightarrow\; n = 3 - 8 \;\rightarrow\; n = -5$.
	\end{atencion}
	
	\begin{ejemploresuelto}{Resolver usando la Propiedad de la Resta}
		
		Cuando la ecuación es $x + a = b$, el $+a$ ``estorba'' la variable. Para eliminarlo usamos la \textbf{operación inversa}: restar.
		
		\vspace{0.3cm}

					\node[draw=valorange, fill=vallightorange, rounded corners=4pt,
					font=\small\bfseries, text width=4cm, align=center] at (8.5,1.5)
					{Solución: restar \textbf{3}\\a ambos lados};
					\draw[-{Stealth}, valorange, line width=2pt, dashed](5.8,1.1)--(5.0,0.3);
				\end{tikzpicture}
			\end{center}
		\end{pasocaja}

	\section*{Misión 4: Propiedad Uniforme de la Multiplicación}
	
	\begin{concepto}{Deshacer una Division — Multiplica ambos lados}
		Para resolver $\dfrac{x}{a} = b$, se \textbf{multiplica} por $a$ a ambos lados:
		
		\vspace{0.3cm}
		\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue, arc=3pt]
			\centering\large\bfseries
			Si $a = b$, entonces $c \cdot a = c \cdot b \quad (c \neq 0)$
		\end{tcolorbox}
		
		\vspace{0.4cm}
		\textbf{Caso especial — Coeficiente fraccionario y recíprocos:}
		
		Si la ecuación es $\dfrac{2}{3}x = 6$, el coeficiente de $x$ es $\dfrac{2}{3}$. Para que ese coeficiente se convierta en $1$, se multiplica por su
		\textbf{recíproco}: $\dfrac{3}{2}$.
		\vspace{0.3cm}
		\begin{center}
			\begin{tikzpicture}
				\node[draw=valblue, fill=vallightblue, rounded corners=5pt,
				minimum width=2.8cm, minimum height=0.9cm,
				align=center, font=\bfseries, line width=1.5pt] at (0,0)
				{$\dfrac{2}{3}$};
				\draw[-{Stealth}, valdark, line width=1.5pt](1.6,0)--(2.8,0)
				node[midway, above, font=\small\bfseries]{recíproco};
				\node[draw=valorange, fill=vallightorange, rounded corners=5pt,
				minimum width=2.8cm, minimum height=0.9cm,
				align=center, font=\bfseries, line width=1.5pt] at (4.4,0)
				{$\dfrac{3}{2}$};
				\node[font=\small, valdark] at (2.2,-0.7)
				{$\dfrac{2}{3} \times \dfrac{3}{2} = \dfrac{6}{6} = 1$};
			\end{tikzpicture}
		\end{center}
		
		\vspace{0.3cm}
		\textbf{Ejemplo completo:}\quad
		$\dfrac{2}{3}x = 6$\;\;
		$\xrightarrow{\times\,\frac{3}{2}}$\;\;
		$\dfrac{3}{2}\cdot\dfrac{2}{3}x = 6\cdot\dfrac{3}{2}$\;\;
		$\rightarrow$\;\;$x = \mathbf{9}$
	\end{concepto}

	\section*{Misión 5: Propiedad Uniforme de la División}
	
	\begin{concepto}{Deshacer una Multiplicación — Divide ambos lados}
		Para resolver $a \cdot x = b$, se \textbf{divide} ambos lados entre $a$:
		
		\vspace{0.3cm}
		\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue, arc=3pt]
			\centering\large\bfseries
			Si $a = b$, entonces $\dfrac{a}{c} = \dfrac{b}{c} \quad (c \neq 0)$
		\end{tcolorbox}
		
		\vspace{0.3cm}
		\textbf{Ejemplo:}\quad$2t = 80$\;\;
		$\xrightarrow{\div\,2}$\;\;
		$\dfrac{2t}{2} = \dfrac{80}{2}$\;\;$\rightarrow$\;\;$t = \mathbf{40}$
		
		\vspace{0.3cm}
		\textbf{Relación importante:}
		Dividir entre $c$ es lo mismo que multiplicar por su recíproco $\dfrac{1}{c}$.\\
		$\dfrac{2t}{2} = \dfrac{80}{2}$ \quad es equivalente a \quad
		$\dfrac{1}{2}\cdot 2t = \dfrac{1}{2}\cdot 80$
	\end{concepto}
	
	\begin{ejemploresuelto}{Resolver con la Propiedad de la División}
		
		Cuando la variable tiene un coeficiente numérico entero ($2t$, $-5x$, $9m$…),
		la operación que ``puso'' ese coeficiente fue una \textbf{multiplicación}. Para deshacerla, \textbf{dividimos} ambos lados entre ese mismo número.
