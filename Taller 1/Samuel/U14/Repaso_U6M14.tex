	% --- PUNTO 2 ---
	\noindent \textbf{2. Propiedad Uniforme: Suma y Resta}
	
	\vspace{0.2cm}
	\noindent \textit{Resuelve las ecuaciones aplicando la operación contraria a ambos lados.}
	
	\vspace{0.4cm}
	\begin{multicols}{2}
		\begin{enumerate}[label=\textbf{\Alph*)}, itemsep=1cm]
			\item \quad $m + 15 = 45$ \\
			Procedimiento: \\
			$m + 15 \ - $ \boxfill[1cm] $= 45 \ - $ \boxfill[1cm] \\
			$m = $ \boxfill[1.5cm]
			
			\item \quad $y - 22 = 18$ \\
			Procedimiento: \\
			$y - 22 \ + $ \boxfill[1cm] $= 18 \ + $ \boxfill[1cm] \\
			$y = $ \boxfill[1.5cm]
		\end{enumerate}
	\end{multicols}
	
	\vspace{0.8cm}
	
	% --- PUNTO 3 ---
	\noindent \textbf{3. Propiedad Uniforme: Multiplicación y División}
	
	\vspace{0.2cm}
	\noindent \textit{Aplica la operación contraria para despejar la incógnita.}
	
	\vspace{0.4cm}
	\begin{multicols}{2}
		\begin{enumerate}[label=\textbf{\Alph*)}, itemsep=1cm]
			\item \quad $7 \cdot x = 56$ \\
			Procedimiento: \\
			$\dfrac{7 \cdot x}{\boxfill[1cm]} = \dfrac{56}{\boxfill[1cm]}$ \\
			$x = $ \boxfill[1.5cm]
			
			\item \quad $\dfrac{p}{4} = 12$ \\
			Procedimiento: \\
			$\dfrac{p}{4} \cdot \boxfill[1cm] = 12 \cdot \boxfill[1cm]$ \\
			$p = $ \boxfill[1.5cm]
		\end{enumerate}
	\end{multicols}
	
	
	% --- PUNTO 4 ---
	\noindent \textbf{4. Reto de Dos Pasos (Ecuaciones Combinadas)}
	
	\vspace{0.2cm}
	\noindent \textit{A veces debes aplicar dos propiedades para dejar la letra sola. Sigue los pasos indicados.}
	
	\vspace{0.4cm}
	\begin{enumerate}[label=\textbf{\Alph*)}, itemsep=1.2cm]
		\item \quad $\mathbf{2x + 10 = 30}$ \\
		\textbf{Paso 1:} Resta 10 a ambos lados para eliminar la suma.\\
		$2x + 10 \ - $ \boxfill[1cm] $= 30 \ - $ \boxfill[1cm] \\
		$2x = $ \boxfill[1.5cm] \\
		\textbf{Paso 2:} Divide ambos lados entre 2 para despejar la $x$.\\
		$\dfrac{2x}{\boxfill[1cm]} = \dfrac{\boxfill[1.5cm]}{\boxfill[1cm]}$ \\
		$x = $ \boxfill[1.5cm]
		
		\item \quad $\mathbf{3y - 5 = 16}$ \\
		\textbf{Paso 1:} Suma 5 a ambos lados.\\
		$3y - 5 \ + $ \boxfill[1cm] $= 16 \ + $ \boxfill[1cm] \\
		$3y = $ \boxfill[1.5cm] \\
		\textbf{Paso 2:} Divide ambos lados entre 3.\\
		$\dfrac{3y}{\boxfill[1cm]} = \dfrac{\boxfill[1.5cm]}{\boxfill[1cm]}$ \\
		$y = $ \boxfill[1.5cm]
	\end{enumerate}
