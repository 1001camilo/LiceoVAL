\begin{objetivo}{Objetivo 2: Simplificar formas especiales}
    \begin{concepto}
        Existen cuatro reglas de oro para las fracciones especiales:
        \begin{enumerate}
            \item \textbf{Mismo número:} {\large$\dfrac{a}{a} = 1$}. (Siempre da 1, es cómo tomar toda la pizza completa).\\
            \item \textbf{Denominador 1:} {\large$\dfrac{a}{1} = a$}. (Siempre da el mismo numerador es cómo no ``cortar'' el objeto).\\
            \item \textbf{Numerador 0:} {\large$\dfrac{0}{a} = 0$}. (Siempre da 0, es cómo no tomar ningún pedazo).\\
            \item \textbf{Denominador 0:} {\large$\dfrac{a}{0} = $} \textbf{Indefinido}. (¡NUNCA se puede dividir por cero!).\\
        \end{enumerate}
    \end{concepto}

\begin{objetivo}{Objetivos 3 y 4: Definir y construir equivalentes}
    \begin{concepto}{Fracciones Equivalentes y Amplificación (Visual)}
    \textbf{1. ¿Qué son? (Objetivo 3)} \\
    Dos fracciones son \textbf{equivalentes} si valen exactamente lo mismo. Visualmente, significa que ocupan la misma cantidad de espacio, aunque estén cortadas en pedazos más pequeños.
    
    \vspace{0.4cm}
    \begin{center}
        \begin{tikzpicture}[scale=0.8]

            \fill[valblue!30] (-4,0) -- (-4,1.5) arc (90:-90:1.5) -- cycle;

            \draw[thick, valdark] (-4,-1.5) -- (-4,1.5);
            \node[below] at (-4,-1.8) {\Large $\mathbf{\dfrac{1}{2}}$};

            % Símbolo =
            \node at (-1.5,0) {\Huge $\mathbf{=}$};

            % Pizza 2/4
            \fill[valblue!30] (1,0) -- (1,1.5) arc (90:-90:1.5) -- cycle;
            \draw[thick, valdark] (1,0) circle (1.5cm);
            \draw[thick, valdark] (1,-1.5) -- (1,1.5);
            \draw[thick, valdark] (-0.5,0) -- (2.5,0); % Corte horizontal
            \node[below] at (1,-1.8) {\Large $\mathbf{\dfrac{2}{4}}$};
            
            % Símbolo =
            \node at (4,0) {\Huge $\mathbf{=}$};

            % Pizza 4/8
            \fill[valblue!30] (6.5,0) -- (6.5,1.5) arc (90:-90:1.5) -- cycle;
            \draw[thick, valdark] (6.5,0) circle (1.5cm);

            \node[below] at (6.5,-1.8) {\Large $\mathbf{\dfrac{4}{8}}$};
        \end{tikzpicture}
    \end{center}
    \vspace{0.2cm}
    \newpage
    \textbf{2. ¿Cómo construirlas?} \\
    Para \textbf{construir} una fracción equivalente más grande (a esto le llamamos \textbf{Amplificar}), el truco mágico es multiplicar el número de arriba y el de abajo por el \textbf{mismo número}. ¡Es como usar un cuchillo para cortar cada pedazo en porciones más pequeñas!
    
    \vspace{0.3cm}
    \begin{center}
        \begin{tikzpicture}
            \node (frac1) at (0,0) {\Huge $\mathbf{\dfrac{1}{2}}$};
            \node (frac2) at (4,0) {\Huge $\mathbf{\dfrac{3}{6}}$};
            
            \draw[->, ultra thick, valred] (0.8, 0.4) to[bend left=30] node[midway, above] {\textbf{Multiplica $\times 3$}} (3.2, 0.4);
            \draw[->, ultra thick, valgreen] (0.8, -0.4) to[bend right=30] node[midway, below] {\textbf{Multiplica $\times 3$}} (3.2, -0.4);
        \end{tikzpicture}
    \end{center}
\end{concepto}
    
    \begin{ejemplo}
        Construir una fracción equivalente a $\dfrac{3}{5}$ multiplicando por $2$:\\
        $\dfrac{3 \times 2}{5 \times 2} = \mathbf{\dfrac{6}{10}}$
    \end{ejemplo}
