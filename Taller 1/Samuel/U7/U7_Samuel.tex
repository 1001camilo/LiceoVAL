	\subsection*{Subtema 1: Fracciones Homogéneas (Igual Denominador)}
	
	\begin{concepto}{El Mismo Tamaño}
		Cuando las fracciones tienen el \textbf{mismo número abajo} (denominador), significa que los pedazos de pizza son del mismo tamaño.
		\\[0.2cm]
		\textbf{Regla:} El número de abajo \textbf{NUNCA} se suma ni se resta, se queda exactamente igual. Solo sumamos o restamos los números de arriba (los numeradores).
	\end{concepto}
	
	\begin{ejemplo}
		\begin{center}
			%\textbf{Suma Visual:}\\
			Tengo $\dfrac{1}{4}$ de pizza y me regalan $\dfrac{2}{4}$ más. Como ambas están cortadas en 4, solo sumo los pedazos que tengo: $1 + 2 = 3$. El 4 de abajo se queda igual.
			\\[0.4cm]
			\Large $\dfrac{1}{4} + \dfrac{2}{4} = \dfrac{1 + 2}{4} = \mathbf{\dfrac{3}{4}}$
			\\[0.6cm]
			\begin{tikzpicture}[scale=0.9]
				% 1/4
				\draw[thick] (0,0) circle (1.2cm);
				\fill[valblue!40] (0,0) -- (1.2,0) arc (0:90:1.2cm) -- cycle;
				\draw[thick] (-1.2,0) -- (1.2,0); \draw[thick] (0,-1.2) -- (0,1.2);
				
				\node at (1.8,0) {\Large $+$};
				
				% 2/4
				\begin{scope}[xshift=3.6cm]
					\draw[thick] (0,0) circle (1.2cm);
					\fill[valblue!40] (0,0) -- (1.2,0) arc (0:180:1.2cm) -- cycle;
					\draw[thick] (-1.2,0) -- (1.2,0); \draw[thick] (0,-1.2) -- (0,1.2);
				\end{scope}
				
				\node at (5.4,0) {\Large $=$};
				
				% 3/4
				\begin{scope}[xshift=7.2cm]
					\draw[thick] (0,0) circle (1.2cm);
					\fill[valblue!40] (0,0) -- (1.2,0) arc (0:270:1.2cm) -- cycle;
					\draw[thick] (-1.2,0) -- (1.2,0); \draw[thick] (0,-1.2) -- (0,1.2);
				\end{scope}
			\end{tikzpicture}
		\end{center}
	\end{ejemplo}

	\subsection*{Subtema 2: Fracciones Heterogéneas (Diferente Denominador)}
	
	\begin{concepto}{El Truco de la Mariposa (Productos Cruzados)}
		Cuando los números de abajo \textbf{son diferentes}, los pedazos son de distinto tamaño y ¡no podemos sumarlos ni restarlos directamente! Tenemos que usar el \textbf{Método de la Mariposa} para igualarlos.
	\end{concepto}
	
	\begin{ejemplo}
		\begin{center}
			\begin{tikzpicture}[scale=1.1]
				% --- Fracciones iniciales (Más separadas) ---
				\node (A) at (0, 1) {\Huge 3};
				\node (B) at (0, -1) {\Huge 4};
				\draw[line width=1.5pt] (-0.4, 0) -- (0.4, 0);
				
				\node at (1.75, 0) {\Huge $+$}; % Signo más centrado
				
				\node (C) at (3.5, 1) {\Huge 1}; % Segunda fracción más a la derecha
				\node (D) at (3.5, -1) {\Huge 5};
				\draw[line width=1.5pt] (3.1, 0) -- (3.9, 0);
				
				% --- Alas (Elipses) ---
				% Rotadas desde el nuevo centro (1.75, 0) y más anchas
				\draw[valred, ultra thick, rotate around={-35:(1.75,0)}] (1.75,0) ellipse (2.8cm and 0.7cm);
				\draw[valblue, ultra thick, rotate around={35:(1.75,0)}] (1.75,0) ellipse (2.8cm and 0.7cm);
				
				% --- Antenas (Multiplicación cruzada) ---
				% Flechas más altas y abiertas para evitar superposición
				
				% Antena Izquierda (3 * 5 = 15)
				\draw[valred, ultra thick, ->] (0, 1.8) to[out=110, in=-30] (-1.5, 3.8) node[above left] {\Large \textbf{15} \ \small ($3 \times 5$)};
				
				% Antena Derecha (4 * 1 = 4)
				\draw[valblue, ultra thick, ->] (3.5, 1.8) to[out=70, in=210] (5.0, 3.8) node[above right] {\small ($4 \times 1$) \ \Large \textbf{4}};
				
				% --- Cuerpo (Multiplicación denominadores) ---
				\draw[valgreen, ultra thick, <->] (0, -1.8) to[out=-90, in=-90] (3.5, -1.8);
				\node[valgreen, below] at (1.75, -2) {\Large \textbf{20} \ \small ($4 \times 5$)};
				
				% --- Proceso final ---
				\node[right, align=left] at (6.0, 0) { % Movido a la derecha
					\Large $\mathbf{= \frac{\textcolor{valred}{15} + \textcolor{valblue}{4}}{\textcolor{valgreen}{20}}}$ \\[0.4cm]
					\Large $\mathbf{= \frac{19}{20}}$
				};
			\end{tikzpicture}
		\end{center}
		\vspace{0.2cm}
		\textbf{Pasos de la Mariposa:}\\
		1. Multiplica cruzado (las alas).\\
		2. Multiplica los de abajo.\\
		3. Suma o resta los cruzados y mantén el resultado de la multiplicación de abajo igual
	\end{ejemplo}
	
	\begin{practica}{Misión 2: Sumas y Restas Heterogéneas}
		\textit{Multiplica cruzado para llenar las cajas de arriba, multiplica los de abajo para la caja inferior, y luego resuelve.}

	\subsection*{Subtema 3: Resolución de Problemas con Fracciones}
	
	\begin{concepto}{Traducir Historias a Fracciones}
		Lee con atención el problema para saber si debes \textbf{sumar o restar}.
		\begin{itemize}
			\item \textbf{Suma (+):} Juntar, añadir, comprar más, total entre los dos.
			\item \textbf{Resta (-):} Comer, regalar, gastar, cortar, cuánto le queda, la diferencia.
		\end{itemize}
		¡Fíjate bien en los denominadores (los números de abajo)! Si son iguales o si son diferentes.
	\end{concepto}
