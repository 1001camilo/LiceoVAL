	\mision{Misión 1: Clasificación de Ángulos}
	
	\begin{conceptbox}{La Abertura de las Líneas}
		Un \textbf{ángulo} está formado por dos líneas que se unen en un punto llamado \textbf{vértice}. La unidad de medida es el \textbf{grado ($^\circ$)}. Según qué tan abiertos estén, se clasifican así:
		\begin{itemize}[itemsep=0.1cm]
			\item \textbf{Agudo:} Mide menos de $90^\circ$ (Es muy cerrado).
			\item \textbf{Recto:} Mide exactamente $90^\circ$ (Forma una "L" perfecta).
			\item \textbf{Obtuso:} Mide más de $90^\circ$ pero menos de $180^\circ$ (Es muy abierto).
			\item \textbf{Llano:} Mide exactamente $180^\circ$ (Es una línea totalmente plana).
		\end{itemize}
	\end{conceptbox}
	
	\begin{center}
		\begin{tikzpicture}[scale=0.9, thick]
			% Agudo
			\draw[<->] (2,1.5) -- (0,0) -- (2.5,0);
			\draw[valblue, fill=valblue!20] (0.5,0) arc (0:36.8:0.5) -- (0,0) -- cycle;
			\node[valblue] at (1, 0.3) {$< 90^\circ$};
			\node at (1.2, -0.5) {\textbf{Agudo}};
			
			% Recto
			\begin{scope}[xshift=4cm]
				\draw[<->] (0,2) -- (0,0) -- (2.5,0);
				\draw[valred, fill=valred!20] (0,0.4) -- (0.4,0.4) -- (0.4,0) -- (0,0) -- cycle;
				\node[valred] at (0.8, 0.8) {$90^\circ$};
				\node at (1.2, -0.5) {\textbf{Recto}};
			\end{scope}

	\mision{Misión 2: Ángulos Complementarios y Suplementarios}
	
	\begin{conceptbox}{Equipos Perfectos}
		A veces los ángulos se unen para formar figuras perfectas:\\
		$\bullet$ \textbf{Complementarios:} Son dos ángulos que al juntarse suman exactamente \textbf{$\mathbf{90^\circ}$} (un ángulo Recto).\\
		$\bullet$ \textbf{Suplementarios:} Son dos ángulos que al juntarse suman exactamente \textbf{$\mathbf{180^\circ}$} (un ángulo Llano).
	\end{conceptbox}
	
	\begin{espaciotrabajo}{¡Tu Turno! - Descubre el ángulo escondido}

	\mision{Misión 3: El Teorema de los 180°}
	
	\begin{conceptbox}{La Regla de Oro de los Triángulos}
		No importa si un triángulo es grande, pequeño, flaco o gordito. \textbf{La suma de sus TRES ángulos internos siempre, siempre dará exactamente $\mathbf{180^\circ}$.}
	\end{conceptbox}
