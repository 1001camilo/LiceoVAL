	\mision{Misión 1: Clasificación de Ángulos}
	
	\begin{conceptbox}{La Abertura de las Líneas}
		Un \textbf{ángulo} está formado por dos líneas que se unen en un punto llamado \textbf{vértice}. La unidad de medida es el \textbf{grado ($^\circ$)}. Según qué tan abiertos estén, se clasifican así:
		\begin{itemize}[itemsep=0.1cm]
			\item \textbf{Agudo:} Mide menos de $90^\circ$ (Es muy cerrado).
			\item \textbf{Recto:} Mide exactamente $90^\circ$ (Forma una "L" perfecta).
			\item \textbf{Obtuso:} Mide más de $90^\circ$ pero menos de $180^\circ$ (Es muy abierto).
			\item \textbf{Llano:} Mide exactamente $180^\circ$ (Es una línea totalmente plana).
		\end{itemize}
	\end{conceptbox}
	
	\begin{center}
		\begin{tikzpicture}[scale=0.9, thick]
			% Agudo
			\draw[<->] (2,1.5) -- (0,0) -- (2.5,0);

	\mision{Misión 2: Parejas Especiales}
	
	\begin{conceptbox}{Equipos Perfectos}
		A veces los ángulos se unen para formar figuras perfectas o tienen relaciones especiales:\\
		$\bullet$ \textbf{Complementarios:} Son dos ángulos que al juntarse suman exactamente \textbf{$\mathbf{90^\circ}$} (forman un ángulo Recto perfecto).\\
		$\bullet$ \textbf{Suplementarios:} Son dos ángulos que al juntarse suman exactamente \textbf{$\mathbf{180^\circ}$} (forman una línea plana o ángulo Llano).\\
		$\bullet$ \textbf{Opuestos por el Vértice:} Son los ángulos que están frente a frente cuando dos líneas rectas se cruzan. \textbf{¡Siempre miden exactamente lo mismo!}
	\end{conceptbox}

		No importa si un triángulo es grande, pequeño, flaco o gordito. \textbf{La suma de sus TRES ángulos internos siempre, siempre dará exactamente $\mathbf{180^\circ}$.}

	\mision{Misión 4: Clasificación de Triángulos}
	
	\begin{conceptbox}{Nombres y Apellidos de los Triángulos}
		Todo triángulo tiene un "nombre" según la medida de sus LADOS y un "apellido" según la medida de sus ÁNGULOS internos.
		\begin{itemize}[itemsep=0.1cm]
			\item \textbf{Según sus lados (Nombre):}
			\begin{itemize}
				\item \textbf{Equilátero:} Sus 3 lados son iguales.
				\item \textbf{Isósceles:} Tiene 2 lados iguales.
				\item \textbf{Escaleno:} Tiene 0 lados iguales (todos son diferentes).
			\end{itemize}
			\item \textbf{Según sus ángulos internos (Apellido):}
			\begin{itemize}
				\item \textbf{Acutángulo:} Sus 3 ángulos son agudos (menores de 90°).
				\item \textbf{Rectángulo:} Tiene 1 ángulo recto (exactamente 90°).
				\item \textbf{Obtusángulo:} Tiene 1 ángulo obtuso (mayor de 90°).
			\end{itemize}
		\end{itemize}
	\end{conceptbox}
