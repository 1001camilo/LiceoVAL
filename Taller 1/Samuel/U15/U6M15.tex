% ---------- Macro genérica: TRIÁNGULO A ESCALA ----------
% \TrianguloEscala{alfa}{beta}{etiquetaA}{etiquetaB}{etiquetaC}
%   alfa = grados del ángulo en A (vértice inferior izq.)
%   beta = grados del ángulo en B (vértice inferior der.)
%   el ángulo en C se calcula solo: 180-alfa-beta
%   etiquetaA/B/C = lo que se ve en el vértice ("x^\circ" o el número)
\newcommand{\TrianguloEscala}[5]{%
	\begin{tikzpicture}[scale=1.3]
		\pgfmathsetmacro{\base}{2.6}
		\pgfmathsetmacro{\gama}{180-#1-#2}
		\pgfmathsetmacro{\bb}{\base*sin(#2)/sin(\gama)}
		\pgfmathsetmacro{\Cx}{\bb*cos(#1)}
		\pgfmathsetmacro{\Cy}{\bb*sin(#1)}
		\coordinate (A) at (0,0);
		\coordinate (B) at (\base,0);
		\coordinate (C) at (\Cx,\Cy);
		% ángulo en C (por diferencia de direcciones)
		\pgfmathsetmacro{\dCA}{atan2(-\Cy,-\Cx)}
		\pgfmathsetmacro{\dCB}{atan2(-\Cy,\base-\Cx)}
		\pgfmathsetmacro{\dCAn}{(\dCA<0)*(\dCA+360)+(\dCA>=0)*(\dCA)}
		\pgfmathsetmacro{\dCBn}{(\dCB<0)*(\dCB+360)+(\dCB>=0)*(\dCB)}
		\pgfmathsetmacro{\dlo}{min(\dCAn,\dCBn)}
		\pgfmathsetmacro{\dhi}{max(\dCAn,\dCBn)}
		% arcos rellenos de color en cada vértice
		\fill[orange!55,opacity=0.75] (A) -- ++(0:0.5) arc (0:#1:0.5) -- cycle;
		\fill[cyan!55,opacity=0.75]   (B) -- ++(180:0.5) arc (180:{180-#2}:0.5) -- cycle;
		\fill[violet!55,opacity=0.75] (C) -- ++(\dlo:0.5) arc (\dlo:\dhi:0.5) -- cycle;
		% triángulo
		\draw[thick] (A) -- (B) -- (C) -- cycle;
		% etiquetas
		\node[font=\bfseries\small] at ($(A)+({#1/2}:0.8)$) {#3};
		\node[font=\bfseries\small] at ($(B)+({180-(#2/2)}:0.8)$) {#4};
		\node[font=\bfseries\small] at ($(C)+({(\dlo+\dhi)/2}:0.8)$) {#5};
	\end{tikzpicture}%
}

	\begin{concepto}{¿Cómo se Nombra un Ángulo?}
		Un \textbf{ángulo} es la abertura formada por dos semirectas (llamados
		\textbf{lados}) que comparten el mismo punto de origen, llamado
		\textbf{vértice}. Se nombra con \textbf{tres letras}: un punto de un
		lado, el \textbf{vértice en el medio}, y un punto del otro lado. También
		puede nombrarse solo con la letra del vértice si no hay confusión posible.
		
		%\vspace{0.2cm}
		\begin{center}
			\anguloTikz{O}{A}{B}{62}{\hspace{7mm}$\angle AOB$}
		\end{center}
		
		\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue, arc=3pt]
			\centering\large\bfseries
			$\angle AOB = \angle BOA = \angle O$
			\quad\normalsize\normalfont (¡el vértice SIEMPRE va en el medio!)
		\end{tcolorbox}
	\end{concepto}

	\section*{Misión 2: Parejas Especiales}
	
	\begin{concepto}{Ángulos Opuestos por el Vértice}
		Cuando dos rectas se cruzan (se intersecan) en un punto, se forman
		\textbf{4 ángulos}, agrupados en \textbf{2 parejas de ángulos opuestos
			por el vértice} (uno frente al otro, sin compartir ningún lado).
		
		%\vspace{0.2cm}
		\begin{center}
			\begin{tikzpicture}[scale=1]
				\pgfmathsetmacro{\ang}{atan(0.5)} % ángulo real de las rectas (~26.565°)
				\draw[valdark,line width=1.4pt] (-2.2,-1.1) -- (2.2,1.1);
				\draw[valdark,line width=1.4pt] (-2.2,1.1) -- (2.2,-1.1);
				\fill[valorange,opacity=0.5] (0,0) -- (\ang:0.4) arc (\ang:{180-\ang}:0.4) -- cycle;
				\fill[valorange,opacity=0.5] (0,0) -- ({180+\ang}:0.4) arc ({180+\ang}:{360-\ang}:0.4) -- cycle;
				\fill[valblue,opacity=0.5] (0,0) -- ({-\ang}:0.4) arc ({-\ang}:{\ang}:0.4) -- cycle;
				\fill[valblue,opacity=0.5] (0,0) -- ({180-\ang}:0.4) arc ({180-\ang}:{180+\ang}:0.4) -- cycle;
				\fill[valdark] (0,0) circle(1.5pt);
				\node[valorange,font=\bfseries] at (0,0.55) {$a$};
				\node[valorange,font=\bfseries] at (0,-0.55) {$a'$};
				\node[valblue,font=\bfseries] at (0.95,0) {$b$};
				\node[valblue,font=\bfseries] at (-0.95,0) {$b'$};
			\end{tikzpicture}
		\end{center}
		
		\begin{tcolorbox}[enhanced, colback=vallightorange, colframe=valorange, arc=3pt]
			\centering\large\bfseries
			Los ángulos opuestos por el vértice son SIEMPRE
			\textbf{CONGRUENTES} (iguales): \quad $m\angle a = m\angle a'$
			\qquad $m\angle b = m\angle b'$
		\end{tcolorbox}
	\end{concepto}
	
	\begin{concepto}{Ángulos Complementarios y Suplementarios}
		Estas son dos parejas especiales que se definen \textbf{únicamente por
			la suma} de sus medidas (los ángulos NO necesitan estar juntos para
		tener estos nombres):
		
		%\vspace{0.2cm}
		\begin{center}
			\renewcommand{\arraystretch}{1.2}
			\begin{tabular}{cc}
				\angulosComplementarios{$A$}{$B$} & \angulosSuplementarios{$A$}{$B$} \\
				\textbf{\color{valblue}COMPLEMENTARIOS} & \textbf{\color{valred}SUPLEMENTARIOS}
			\end{tabular}
		\end{center}
		
		\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue, arc=3pt]
			\centering\bfseries
			Si $A$ y $B$ son \textbf{complementarios}:\quad $m\angle A + m\angle B = 90^\circ$\\[4pt]
			Si $A$ y $B$ son \textbf{suplementarios}:\quad $m\angle A + m\angle B = 180^\circ$
		\end{tcolorbox}
	\end{concepto}
	
	\begin{ejemplo}
		\textbf{Resuelve usando lo que sabes de parejas de ángulos:}
		
		\vspace{0.2cm}
		\textbf{1) Opuestos por el vértice:} $\angle a$ y $\angle b$ son opuestos
		por el vértice. Si $\angle a = (3x+10)^\circ$ y $\angle b = 70^\circ$,
		halla $x$.\\
		Como son opuestos por el vértice, $\angle a = \angle b$:
		$$3x + 10 = 70 \quad\rightarrow\quad 3x = 70 - 10 \quad\rightarrow\quad
		3x = 60 \quad\rightarrow\quad x=60\div 3 \quad\rightarrow\quad\mathbf{x = 20}$$
		
		\textbf{2) Complementarios:} $\angle A$ y $\angle B = 35^\circ$ son
		complementarios. Halla $\angle A$.
		$$\angle A + 35^\circ = 90^\circ \quad\rightarrow\quad
		\angle A = 90^\circ - 35^\circ \quad\rightarrow\quad \mathbf{\angle A = 55^\circ}$$
		
		\textbf{3) Suplementarios:} $\angle A = (2x-10)^\circ$ y
		$\angle B = 120^\circ$ son suplementarios. Halla $x$.
		$$(2x - 10) + 120 = 180 \quad\rightarrow\quad 2x + 110 = 180
		\quad\rightarrow\quad 2x = 70 \quad\rightarrow\quad \mathbf{x = 35}$$
	\end{ejemplo}

	\section*{Misión 4: El Secreto de los Triángulos ($180^\circ$)}
	
	\begin{concepto}{El Secreto de los $180^\circ$}
		La suma	de las medidas de los TRES ángulos internos de \textbf{CUALQUIER}
		triángulo —sin importar su forma o tamaño— es SIEMPRE $180^\circ$.
		
		%\vspace{0.2cm}
		\begin{center}
			\trianguloAngulos{$a$}{$b$}{$c$}
		\end{center}
		
		\begin{tcolorbox}[enhanced, colback=vallightblue, colframe=valblue, arc=3pt]
			\centering\Large\bfseries
			$m\angle a + m\angle b + m\angle c = 180^\circ$
		\end{tcolorbox}
		
		\textbf{¿Por qué pasa esto?} Si ``arrancas'' los tres ángulos de un
		triángulo y los juntas uno al lado del otro por su vértice, ¡forman
		exactamente una línea recta (media vuelta $=180^\circ$)!
		
		\begin{center}
			\begin{tikzpicture}[scale=1.1]
				\coordinate (O) at (0,0);
				\draw[valdark,line width=1.5pt] (-2.2,0) -- (2.2,0);
				\draw[valdark,line width=1.2pt] (O) -- (50:2.0);
				\draw[valdark,line width=1.2pt] (O) -- (120:2.0);
				\draw[valorange,fill=vallightorange,opacity=0.65] (O) -- (0:0.7) arc (0:50:0.7) -- cycle;
				\draw[valred,fill=vallightred,opacity=0.65] (O) -- (50:0.7) arc (50:120:0.7) -- cycle;
				\draw[valblue,fill=vallightblue,opacity=0.65] (O) -- (120:0.7) arc (120:180:0.7) -- cycle;
				\node[valorange,font=\bfseries\small] at (25:0.95) {$a$};
				\node[valred,font=\bfseries\small] at (85:0.95) {$b$};
				\node[valblue,font=\bfseries\small] at (150:0.95) {$c$};
			\end{tikzpicture}\\
			\footnotesize $a+b+c=180^\circ$ (media vuelta $=$ una recta)
		\end{center}
	\end{concepto}
	
	\begin{concepto}{Caso Especial: El Triángulo Isósceles}
		En un triángulo isósceles, los DOS ángulos que están junto a la base
		(el lado distinto) se llaman \textbf{ángulos de la base}, y son SIEMPRE
		\textbf{IGUALES} entre sí, porque cada uno queda ``enfrente'' de uno de
		los dos lados iguales.
		
		%\vspace{0.2cm}
		\begin{center}
			\begin{tikzpicture}[scale=1]
				\coordinate (A) at (0,0); \coordinate (B) at (3.2,0); \coordinate (C) at (1.6,2.6);
				\draw[valdark,line width=1.4pt,marcalado] (A) -- (C);
				\draw[valdark,line width=1.4pt,marcalado] (B) -- (C);
				\draw[valdark,line width=1.4pt] (A) -- (B);
				\node[valorange,font=\bfseries] at (0.55,0.28) {$x$};
				\node[valorange,font=\bfseries] at (2.65,0.28) {$x$};
				\node[valblue,font=\bfseries] at (1.6,2.05) {$v$};
				\node[below,font=\small] at (1.6,-0.28) {base};
			\end{tikzpicture}
		\end{center}
		
		\begin{tcolorbox}[enhanced, colback=vallightgreen, colframe=valgreen, arc=3pt]
			\centering\bfseries
			Si conoces el ángulo del vértice ($v$), cada ángulo de la base ($x$)
			cumple:
			$$2x + v = 180^\circ$$
		\end{tcolorbox}
	\end{concepto}
	
	\begin{truco}{El Tercer Ángulo}
		Si ya conoces dos ángulos de un triángulo (o puedes plantear una
		ecuación con ellos), el tercero siempre se calcula igual:
		$$x = 180^\circ - (\text{suma de los otros dos ángulos})$$
		Y en un triángulo isósceles, si conoces el ángulo del vértice entre los dos lados iguales, ¡ya
		sabes que los otros dos son IGUALES entre sí! Solo falta repartir en
		partes iguales lo que queda de $180^\circ$.
	\end{truco}
	
	\begin{ejemplo}
		\textbf{1) Triángulo general:} halla el valor de $x$.
		\begin{center}
			\trianguloAngulos{$40^\circ$}{$90^\circ$}{$x$}
		\end{center}
		$$x + 40^\circ + 90^\circ = 180^\circ \quad\rightarrow\quad
		x + 130^\circ = 180^\circ \quad\rightarrow\quad \mathbf{x = 50^\circ}$$
		
		\textbf{2) Triángulo isósceles:} el ángulo del vértice mide $40^\circ$.
		Halla cada ángulo de la base.
		\begin{center}
			\begin{tikzpicture}[scale=0.85]
				\coordinate (A) at (0,0); \coordinate (B) at (3.2,0); \coordinate (C) at (1.6,2.6);
				\draw[valdark,line width=1.4pt,marcalado] (A) -- (C);
				\draw[valdark,line width=1.4pt,marcalado] (B) -- (C);
				\draw[valdark,line width=1.4pt] (A) -- (B);
				\node[valorange,font=\bfseries] at (0.55,0.28) {$x$};
				\node[valorange,font=\bfseries] at (2.65,0.28) {$x$};
				\node[valblue,font=\bfseries] at (1.6,2.05) {$40^\circ$};
			\end{tikzpicture}
		\end{center}
		Como los ángulos de la base son iguales, llamamos $x$ a cada uno:
		$$x + x + 40^\circ = 180^\circ \quad\rightarrow\quad 2x = 140^\circ
		\quad\rightarrow\quad \mathbf{x = 70^\circ}$$
	\end{ejemplo}
