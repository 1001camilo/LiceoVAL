	\mision{Misión 1: Clasificación de Ángulos}
	
	\begin{conceptbox}[La Abertura de las Líneas]
		Un \textbf{ángulo} está formado por dos semirrectas que comparten un punto inicial llamado \textbf{vértice}. La unidad de medida estándar es el \textbf{grado ($^\circ$)}. Según su abertura, los clasificamos en:
		\begin{itemize}[itemsep=0.1cm]
			\item \textbf{Agudo:} Mide menos de $90^\circ$.
			\item \textbf{Recto:} Mide exactamente $90^\circ$ (forma una "L" perfecta).
			\item \textbf{Obtuso:} Mide más de $90^\circ$ pero menos de $180^\circ$.
			\item \textbf{Llano:} Mide exactamente $180^\circ$ (una línea recta plana).
		\end{itemize}
	\end{conceptbox}
	
	\begin{center}
		\begin{tikzpicture}[scale=0.9, thick]
			% Agudo
			\draw[<->] (2,1.5) -- (0,0) -- (2.5,0);
			\draw[valblue, fill=valblue!20] (0.5,0) arc (0:36.8:0.5) -- (0,0) -- cycle;
			\node[valblue] at (1, 0.3) {$< 90^\circ$};
			\node at (1.2, -0.5) {\textbf{Agudo}};
			
			% Recto
			\begin{scope}[xshift=4cm]
				\draw[<->] (0,2) -- (0,0) -- (2.5,0);
				\draw[valred, fill=valred!20] (0,0.4) -- (0.4,0.4) -- (0.4,0) -- (0,0) -- cycle;
				\node[valred] at (0.8, 0.8) {$90^\circ$};
				\node at (1.2, -0.5) {\textbf{Recto}};
			\end{scope}
			
			% Obtuso
			\begin{scope}[xshift=8cm]
				\draw[<->] (-1.5,1.5) -- (0,0) -- (2.5,0);
				\draw[valgreen, fill=valgreen!20] (0.5,0) arc (0:135:0.5) -- (0,0) -- cycle;
				\node[valgreen] at (0.5, 0.8) {$> 90^\circ$};
				\node at (0.5, -0.5) {\textbf{Obtuso}};
			\end{scope}
			
			% Llano

	\mision{Misión 2: Ángulos Complementarios y Suplementarios}
	
	\begin{conceptbox}[Equipos de Ángulos]
		A veces los ángulos trabajan en equipo para formar figuras perfectas:\\
		$\bullet$ \textbf{Complementarios:} Dos ángulos que al sumarse forman un ángulo recto (\textbf{$\mathbf{90^\circ}$}).\\
		$\bullet$ \textbf{Suplementarios:} Dos ángulos que al sumarse forman un ángulo llano (\textbf{$\mathbf{180^\circ}$}).
	\end{conceptbox}
	
	\begin{examplebox}
		Encuentra el valor del ángulo faltante $x$ sabiendo que los ángulos son \textbf{suplementarios}.\\[0.2cm]
		\begin{minipage}{0.35\textwidth}
			\begin{tikzpicture}[scale=0.8, thick]
				\draw[<->] (-3,0) -- (3,0);
				\draw[->] (0,0) -- (1.5,2.5);
				\draw[valblue, fill=valblue!10] (0.8,0) arc (0:59:0.8) -- (0,0) -- cycle;
				\draw[valred, fill=valred!10] (1.1,1.8) arc (59:180:1.2) -- (0,0) -- cycle;
				\node at (1.5, 0.5) {$60^\circ$};
				\node at (-1, 0.8) {$x$};
			\end{tikzpicture}
		\end{minipage}%
		\begin{minipage}{0.6\textwidth}
			\textbf{1. Plantea la ecuación:} \quad $x + 60^\circ = 180^\circ$\\[0.2cm]
			\textbf{2. Despeja (resta):} \qquad $x = 180^\circ - 60^\circ$\\[0.2cm]
			\textbf{3. Resultado:} \qquad \qquad $\mathbf{x = 120^\circ}$
		\end{minipage}
	\end{examplebox}
	
	\begin{espaciotrabajo}[¡Tu Turno! - Ecuaciones Geométricas]
		Plantea la ecuación lineal para encontrar el ángulo desconocido $x$ en cada caso y resuélvela.
		
		\vspace{0.4cm}
		\begin{multicols}{2}
			\textbf{A) Son Complementarios}\\
			\begin{tikzpicture}[scale=0.7, thick]
				\draw[<->] (0,3) -- (0,0) -- (3,0);
				\draw[->] (0,0) -- (2.5,1.5);
				% Símbolo de 90° general
				\draw (0,0.3) -- (0.3,0.3) -- (0.3,0);
				\node at (2, 0.5) {$35^\circ$};
				\node at (0.8, 1.8) {$x$};
			\end{tikzpicture}\\[0.2cm]
			\textbf{Ecuación:}\\
			$x + \boxfill[1cm] = 90^\circ$\\[0.2cm]
			\textbf{Despeje:}\\
			$x = 90^\circ - \boxfill[1cm]$\\[0.2cm]
			\textbf{Respuesta:} \quad $x = $ \boxfill[1.5cm]
			
			\columnbreak
			
			\textbf{B) Son Suplementarios}\\
			\begin{tikzpicture}[scale=0.7, thick]
				\draw[<->] (-3,0) -- (3,0);
				\draw[->] (0,0) -- (-1.5,2);
				\node at (1, 0.5) {$x$};
				\node at (-1.5, 0.6) {$110^\circ$};
			\end{tikzpicture}\\[0.2cm]
			\textbf{Ecuación:}\\
			$x + \boxfill[1cm] = \boxfill[1.5cm]$\\[0.2cm]
			\textbf{Despeje:}\\
			$x = \boxfill[1cm] - \boxfill[1cm]$\\[0.2cm]
			\textbf{Respuesta:} \quad $x = $ \boxfill[1.5cm]
		\end{multicols}
	\end{espaciotrabajo}

	\mision{Misión 3: Familias de Triángulos}
	
	\begin{conceptbox}[Según sus Lados y según sus Ángulos]
		\textbf{Por sus Lados:}\\
		$\bullet$ \textbf{Equilátero:} 3 lados iguales. \quad $\bullet$ \textbf{Isósceles:} 2 lados iguales. \quad $\bullet$ \textbf{Escaleno:} Ningún lado igual.\\[0.2cm]
		\textbf{Por sus Ángulos:}\\
		$\bullet$ \textbf{Acutángulo:} 3 ángulos agudos. \quad $\bullet$ \textbf{Rectángulo:} 1 ángulo recto ($90^\circ$). \quad $\bullet$ \textbf{Obtusángulo:} 1 ángulo obtuso.
	\end{conceptbox}

	\mision{Misión 4: El Teorema de los 180°}
	
	\begin{conceptbox}[La Regla de Oro de los Triángulos]
		No importa si el triángulo es grande, pequeño, rectángulo o escaleno. \textbf{La suma de los tres ángulos internos de CUALQUIER triángulo siempre es exactamente $\mathbf{180^\circ}$.}\\
		Ecuación Maestra: \quad $\boldsymbol{\angle A + \angle B + \angle C = 180^\circ}$
	\end{conceptbox}
	
	\begin{espaciotrabajo}[¡Tu Turno! - Ecuaciones Triangulares]
		Arma la ecuación sumando los 3 ángulos para que den $180^\circ$ y despeja la $x$.
		
		\vspace{0.4cm}
		\begin{multicols}{2}
			\textbf{A)}\\
			\begin{tikzpicture}[scale=0.8, thick]
				\draw (0,0) -- (4,0) -- (1,2.5) -- cycle;
				\node at (0.6, 0.3) {$x$};
				\node at (3.2, 0.3) {$40^\circ$};
				\node at (1.2, 1.8) {$80^\circ$};
			\end{tikzpicture}\\[0.2cm]
			\textbf{Ecuación:}\\
			$x + \boxfill[1cm] + \boxfill[1cm] = 180^\circ$\\[0.2cm]
			\textbf{Suma los conocidos:}\\
			$x + \boxfill[1.5cm] = 180^\circ$\\[0.2cm]
			\textbf{Despeja:}\\
			$x = 180^\circ - \boxfill[1.5cm]$\\[0.2cm]
			\textbf{Respuesta:} $x = $ \boxfill[1.5cm]
