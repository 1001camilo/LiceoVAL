	% --- PUNTO 2 ---
	\noindent \textbf{2. Parejas Geométricas en Paralelas}
	
	\vspace{0.2cm}
	\noindent Observa las líneas paralelas cortadas por una transversal. Sabiendo que el ángulo resaltado mide $\mathbf{130^\circ}$, deduce el valor de los ángulos indicados por las letras basándote en la lógica visual.
	
	\begin{center}
		\begin{tikzpicture}[scale=1.2]
			\draw[thick, <->] (-3, 2) -- (3, 2) node[right] {$L_1$};
			\draw[thick, <->] (-3, 0) -- (3, 0) node[right] {$L_2$};
			\draw[thick, <->, >=stealth] (-1.5, -1) -- (1.5, 3) node[above] {$T$};
			
			\coordinate (A) at (0.75, 2);
			\coordinate (B) at (-0.75, 0);
			
			% Ángulo principal: 130 grados (obtuso superior izquierdo)
			\draw[fill=valred!20, draw=valred, thick] (A) -- ++(53:0.5) arc (53:180:0.5) -- cycle;
			\node at ($(A) + (-0.6, 0.4)$) {$\mathbf{130^\circ}$};
			
			% Ángulo x: agudo superior derecho
			\draw[fill=valblue!20, draw=valblue, thick] (A) -- ++(0.5,0) arc (0:53:0.5) -- cycle;
			\node at ($(A) + (0.7, 0.25)$) {$x$};
			
			% Ángulo y: obtuso inferior derecho (Alterno interno)
			\draw[fill=valred!20, draw=valred, thick] (B) -- ++(0.5,0) arc (360:233:-0.5) -- cycle;
			\draw[fill=valred!20, draw=valred, thick] (B) -- ++(0.5,0) arc (0:-127:0.5) -- cycle;
			\draw[fill=valred!20, draw=valred, thick] (B) -- ++(233:0.5) arc (233:360:0.5) -- cycle;
			\node at ($(B) + (0.6, -0.4)$) {$y$};
			
			% Ángulo z: agudo inferior izquierdo
			\draw[fill=valblue!20, draw=valblue, thick] (B) -- ++(180:0.5) arc (180:233:0.5) -- cycle;
			\node at ($(B) + (-0.7, -0.25)$) {$z$};
		\end{tikzpicture}
	\end{center}
