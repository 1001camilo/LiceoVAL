	\section*{Misión 1: Conociendo el Mundo Decimal}
	
	\begin{concepto}{A la derecha de la coma}
		La \textbf{Coma Decimal} separa a los números enteros gigantes de los pedacitos más pequeños que 1. A la izquierda viven los enteros (Unidades, Decenas...). A la derecha de la coma viven los pedacitos: \textbf{Décimas}, \textbf{Centésimas} y \textbf{Milésimas}.
	\end{concepto}

	\section*{Misión 2: Adición y Sustracción}
	
	\begin{concepto}{}
		Para sumar o restar decimales, hay una sola regla obligatoria: \textbf{¡La coma decimal siempre debe ir una debajo de la otra.}
		\\
		Si a un número le faltan pedacitos para quedar del mismo largo, simplemente \textbf{rellenamos los huecos con ceros ($0$)}.
	\end{concepto}

	\section*{Misión 3: Multiplicación de Decimales}
	
	\begin{concepto}{Ignorar y Contar}
		Para multiplicar números con coma decimal, \textbf{no necesitas alinearlos}. Sigue estos dos pasos:
		\begin{enumerate}
			\item Multiplica normal, \textbf{ignorando por completo} las comas (como si no existieran).
			\item Al final, cuenta cuántos números hay \textbf{después} de la coma en toda la operación, y mueve la coma en tu respuesta esa misma cantidad de espacios hacia la izquierda.
		\end{enumerate}
	\end{concepto}
	
	\vspace{0.3cm}
	\begin{center}

	\section*{Misión 4: Multiplicar por 10, 100 y 1000}
	
	\begin{concepto}{Multiplicar por múltiplos de 10}
		Cuando multiplicas cualquier número por $10$, $100$ o $1.000$, \textbf{no tienes que hacer ninguna operación matemática complicada}. 
		\\[0.2cm]
		Solo debes hacer que la coma decimal "salte" hacia la \textbf{derecha} tantas veces como \textbf{ceros} tenga el número.
		\begin{itemize}
			\item Por \textbf{10} (1 cero) $\rightarrow$ Salta \textbf{1} vez a la derecha.
			\item Por \textbf{100} (2 ceros) $\rightarrow$ Salta \textbf{2} veces a la derecha.
			\item Por \textbf{1000} (3 ceros) $\rightarrow$ Salta \textbf{3} veces a la derecha.
		\end{itemize}
		\textit{Nota: Si se te acaban los números para saltar, ¡pones ceros en los huecos vacíos!}
	\end{concepto}
	
	\vspace{0.4cm}
	\begin{center}

	\section*{Misión 5: Estimación (Redondeo)}
	
	\begin{concepto}{Acercándose a la verdad}
		En la vida real, a veces no necesitamos el número exacto, sino uno aproximado. A esto le llamamos \textbf{Estimar} o \textbf{Redondear}. 
		\\[0.2cm]
		\textbf{Regla del Vecino de la Derecha:}
		\begin{itemize}
			\item Si el número vecino es \textbf{5, 6, 7, 8 o 9}: Sumas $1$ al número que estás redondeando (Sube).
			\item Si el número vecino es \textbf{0, 1, 2, 3 o 4}: Lo dejas exactamente igual (Se queda).
		\end{itemize}
		¡Todo lo que esté después del vecino desaparece!
	\end{concepto}
	
	\begin{ejemplo}
		\begin{multicols}{2}
			\textbf{1. Redondear $3,7$ a la Unidad:} \\
			El número entero es $3$. Su vecino es $7$. \\
			Como $7$ es grande (5 o más), el $3$ sube a $\mathbf{4}$.
			
			\columnbreak
			
			\textbf{2. Redondear $8,24$ a la Décima:} \\
			El número en la décima es $2$. Su vecino es $4$. \\
			Como $4$ es pequeño, el $2$ se queda igual: $\mathbf{8,2}$.
		\end{multicols}
	\end{ejemplo}
	
	\begin{practica}{Aproximación }

	\section*{Misión 6: Problemas del Mundo Real}
	
	\begin{concepto}{Traducir historias a Matemáticas}
		\begin{itemize}
			\item \textbf{Suma (+):} Juntar cosas, añadir dinero, distancia total.
			\item \textbf{Resta (-):} Gastar dinero, cortar algo, ver cuánto falta o cuánto sobra (la diferencia).
			\item \textbf{Multiplicar ($\times$):} Comprar varias cosas iguales, repetir una medida varias veces.
		\end{itemize}
	\end{concepto}
	
	\begin{practica}{Detective de Problemas (15 Ejercicios)}
		\textit{Lee el problema con atención. Anota la operación (puedes usar el espacio para calcular) y escribe la respuesta final con la coma en su lugar.}
		
		\vspace{0.4cm}
		\begin{enumerate}[itemsep=0.8cm]
			\item Fui a la tienda y compré unas papas de $\mathbf{\$2,50}$ y un jugo de $\mathbf{\$1,20}$. ¿Cuánto pagué en total? \\
			Operación: $2,50 + 1,20$ \hspace{2cm} \textbf{Respuesta:} \$ \boxfill[2cm]
			
			\item Tenía un billete de $\mathbf{\$10}$ y me gasté $\mathbf{\$4,50}$ en dulces. ¿Cuánto dinero me sobró?.\\
			Operación: $10,00 - 4,50$ \hspace{2cm} \textbf{Respuesta:} \$ \boxfill[2cm]
			
			\item Cada cuaderno vale $\mathbf{\$3,15}$. Si compro \textbf{3} cuadernos para el colegio, ¿cuánto pago en total?\\
			Operación: $3,15 \times 3$ \hspace{2cm} \textbf{Respuesta:} \$ \boxfill[2cm]
			\newpage
			
			\item Un deportista corrió $\mathbf{4,5}$ km el lunes y $\mathbf{3,25}$ km el martes. ¿Cuántos km corrió en los dos días?.\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] km.
			
			\item Un pedazo de madera medía $\mathbf{5,8}$ metros y el carpintero cortó $\mathbf{2,3}$ metros. ¿Cuánto mide el pedazo que quedó?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] metros.
			
			\item Un paquete de galletas pesa $\mathbf{0,25}$ kilos. Si compro \textbf{10} paquetes, ¿cuánto pesan en total?.\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] kilos.
			
			\item Mi temperatura era de $\mathbf{36,5}$ grados. Luego subió $\mathbf{1,2}$ grados porque tuve fiebre. ¿Cuál es mi temperatura ahora?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] grados.
			
			\item Un frasco de miel pesa $\mathbf{1,4}$ kilos. Si en la caja hay \textbf{2} frascos iguales, ¿cuánto pesan juntos?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] kilos.
			
			\item Quieres comprar un juguete de $\mathbf{\$15,50}$, pero solo tienes $\mathbf{\$12,00}$. ¿Cuánto dinero te falta?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \$ \boxfill[2cm]
			
			\item El colegio regaló $\mathbf{0,5}$ litros de leche a cada niño. Si hay \textbf{100} niños, ¿cuántos litros de leche regalaron en total?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] litros.
			
			\item Para hacer un pastel, mezclo $\mathbf{1,25}$ kg de harina y $\mathbf{0,50}$ kg de azúcar. ¿Cuánto pesa la mezcla completa?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] kilos.
			
			\item Una tela medía $\mathbf{8,0}$ metros y cortaron $\mathbf{4,5}$ metros para unas cortinas. ¿Cuánta tela sobró?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] metros.
			\newpage
			
			\item El kilo de manzanas vale $\mathbf{\$2,10}$. Si llevo $\mathbf{3}$ kilos, ¿cuánto me cuestan?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \$ \boxfill[2cm]
			
			\item Tu hermano mide $\mathbf{1,45}$ metros y tú mides $\mathbf{1,60}$ metros. ¿Cuántos metros de diferencia hay entre los dos?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \boxfill[2cm] metros.
			
			\item Ganas $\mathbf{\$5,25}$ diarios por ayudar en casa. Si ayudas durante \textbf{10} días seguidos, ¿cuánto dinero tendrás?\\
			Operación: \boxline[3cm] \hspace{2cm} \textbf{Respuesta:} \$ \boxfill[2cm]
		\end{enumerate}
