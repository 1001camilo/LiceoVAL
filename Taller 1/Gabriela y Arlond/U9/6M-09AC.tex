	%  MISIÓN 1 – ¿Qué es una Fracción?
	% ══════════════════════════════════════════════════════════════════════════
	\mision{valblue}{Misión 1: {¿}Qué es una Fracción?}
	
	\begin{concepto}{$\square$ Aprende el Concepto}
		Una \textbf{fracción} representa una \textbf{PARTE} de un \textbf{TODO},
		que ha sido dividido en partes \textbf{IGUALES}.
		
		%\vspace{8pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm,y=1cm]
				%── Pizza dividida en 4, 1 parte coloreada ──────────────
				\begin{scope}[shift={(2.3,0)}]
					\draw[valdark,line width=1.6pt,fill=valorange]
					(0,0) -- (0:2.2) arc (0:90:2.2) -- cycle;
					\draw[valdark,line width=1.6pt,fill=white]
					(0,0) -- (90:2.2) arc (90:180:2.2) -- cycle;
					\draw[valdark,line width=1.6pt,fill=white]
					(0,0) -- (180:2.2) arc (180:270:2.2) -- cycle;
					\draw[valdark,line width=1.6pt,fill=white]
					(0,0) -- (270:2.2) arc (270:360:2.2) -- cycle;
					\node[font=\bfseries\normalsize,color=valdark] at (0,-2.8)
					{Pizza dividida en 4 partes};
				\end{scope}
				
				%── Flecha ───────────────────────────────────────────────
				\draw[flechaop,valdark] (5.0,0) -- (6.0,0);
				
				%── Resultado fracción ───────────────────────────────────
				\node[cajao,minimum width=3.6cm,minimum height=2.0cm,
				font=\bfseries\Huge,inner sep=8pt] at (8.4,0)
				{$\dfrac{1}{4}$};
				\node[font=\bfseries\normalsize,color=valdark,align=center,
				text width=4.2cm] at (8.4,-1.7)
				{Tomamos \textbf{1} parte\\ de un total de \textbf{4}};
			\end{tikzpicture}
		\end{center}
		
		%\vspace{6pt}
		\begin{center}
			\begin{tikzpicture}
				\node[cajaz,minimum width=14.5cm,minimum height=0.9cm,
				font=\bfseries\large,inner sep=7pt] at (0,0)
				{El signo de fracción es una raya: $\dfrac{\;\;}{\;\;}$};
				\node[cajaz,minimum width=14.5cm,minimum height=0.9cm,
				font=\bfseries\large,inner sep=7pt] at (0,-1.5)
				{$\dfrac{1}{4}$\ Se lee: ``un cuarto'' o ``1 de 4''};
			\end{tikzpicture}
			
		\end{center}
	\end{concepto}
	
	%\vspace{4pt}
	
	\begin{truco}{¡Regla de Oro: Las Partes Deben Ser IGUALES!}
		Para que algo sea una fracción correcta, \textbf{todas las partes}
		deben tener el \textbf{mismo tamaño}.
		
		%\vspace{8pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm,y=1cm]
				%── CORRECTO: rectángulo 4 partes iguales ───────────────
				\begin{scope}[shift={(0,0)}]
					\draw[valdark,line width=1.5pt,fill=vallightgreen]
					(0,0) rectangle (4.0,1.3);
					\draw[valdark,line width=1.2pt] (1.0,0) -- (1.0,1.3);
					\draw[valdark,line width=1.2pt] (2.0,0) -- (2.0,1.3);
					\draw[valdark,line width=1.2pt] (3.0,0) -- (3.0,1.3);
					\node[font=\Huge,color=valgreen] at (5.0,0.65) {\checkmark};
					\node[font=\bfseries\normalsize,color=valgreen,align=center,
					text width=3cm] at (2.0,-0.65)
					{CORRECTO\\ {\footnotesize partes iguales}};
				\end{scope}
				
				%── INCORRECTO: rectángulo 3 partes desiguales ──────────
				\begin{scope}[shift={(8.8,0)}]
					\draw[valdark,line width=1.5pt,fill=vallightred]
					(0,0) rectangle (4.0,1.3);
					\draw[valdark,line width=1.2pt] (0.7,0) -- (0.7,1.3);
					\draw[valdark,line width=1.2pt] (2.3,0) -- (2.3,1.3);

		\end{center}
	\end{truco}

	% ══════════════════════════════════════════════════════════════════════════
	%  MISIÓN 2 – Los Términos de la Fracción
	% ══════════════════════════════════════════════════════════════════════════
	\mision{valblue}{Misión 2: Los Términos de la Fracción}
	
	\begin{concepto}{$\square$ Aprende el Concepto: Las 2 Partes con Nombre}
		Toda fracción tiene \textbf{2 números} con nombre propio, separados por
		una \textbf{raya}.
		
		%\vspace{10pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm,y=1cm]
				%── Número de arriba ─────────────────────────────────────
				\node[btnazul,minimum width=2.1cm,minimum height=1.5cm,
				font=\bfseries\Huge] at (0,1.4) {3};
				%── Raya ──────────────────────────────────────────────────
				\draw[valdark,line width=3pt] (-1.4,0) -- (1.4,0);
				%── Número de abajo ──────────────────────────────────────
				\node[btnorange,minimum width=2.1cm,minimum height=1.5cm,
				font=\bfseries\Huge] at (0,-1.4) {4};
				
				%── Flecha y etiqueta NUMERADOR ──────────────────────────
				\draw[flechaop,valblue,line width=1.5pt] (1.9,1.4) -- (2.9,1.4);
				\node[cajaz,inner sep=6pt,font=\bfseries\large,align=left,
				text width=5.2cm] at (5.9,1.4)
				{NUMERADOR\\[2pt]
					{\normalsize cuántas partes\\ \textbf{tomamos}}};
				
				%── Flecha y etiqueta DENOMINADOR ────────────────────────
				\draw[flechaop,valorange,line width=1.5pt] (1.9,-1.4) -- (2.9,-1.4);
				\node[cajao,inner sep=6pt,font=\bfseries\large,align=left,
				text width=5.4cm] at (6.0,-1.4)
				{DENOMINADOR\\[2pt]
					{\normalsize en cuántas partes\\ se dividió el \textbf{entero}}};
				
				%── Pequeña pizza de apoyo (3/4) ─────────────────────────
				\begin{scope}[shift={(10.7,0)}]
					\draw[valdark,line width=1.3pt,fill=vallightblue]
					(0,0) -- (0:1.4) arc (0:90:1.4) -- cycle;
					\draw[valdark,line width=1.3pt,fill=vallightblue]
					(0,0) -- (90:1.4) arc (90:180:1.4) -- cycle;
					\draw[valdark,line width=1.3pt,fill=vallightblue]
					(0,0) -- (180:1.4) arc (180:270:1.4) -- cycle;
					\draw[valdark,line width=1.3pt,fill=white]
					(0,0) -- (270:1.4) arc (270:360:1.4) -- cycle;
					\node[font=\bfseries\small,color=valdark] at (0,-1.9)
					{3 de 4 partes};
				\end{scope}
			\end{tikzpicture}
		\end{center}
	\end{concepto}
	
	%\vspace{4pt}
	
	\begin{truco}{¡El de ABAJO Manda!}
		Un truco fácil para recordar:
		\begin{itemize}[leftmargin=*,itemsep=2pt]
			\item El \textcolor{valorange}{\textbf{DENOMINADOR}} (de \textbf{abajo})
			\textbf{manda}: dice en cuántas partes se cortó el entero.
			\item El \textcolor{valblue}{\textbf{NUMERADOR}} (de \textbf{arriba})
			\textbf{cuenta} cuántas de esas partes tomamos.
		\end{itemize}
	\end{truco}

	\mision{valred}{Misión 5: {¡}Matemáticas para la Vida! --- Fracciones en lo Cotidiano}

		Usamos fracciones \textbf{todos los días}: cuando vemos la hora, cuando
		repartimos comida, cuando cocinamos o cuando usamos dinero.
		¡Vamos a practicar con situaciones reales!
	\end{concepto}
