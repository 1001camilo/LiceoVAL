	\section*{Misión 1: Hacer Parejas}
	
	\begin{concepto}{¿Qué es un número PAR y qué es IMPAR?}
		\begin{itemize}
			\item Un número es \textbf{PAR} si al dibujar esa cantidad podemos agrupar todo de dos en dos y \textbf{NO SOBRA NINGUNO}.
			\item Un número es \textbf{IMPAR} si al agrupar de dos en dos, siempre \textbf{SOBRA UNO} que se queda sin pareja.
		\end{itemize}
	\end{concepto}

	\section*{Misión 2: El Truco de la Cola}
	
	\begin{concepto}{No importa qué tan grande sea}
		¿Qué pasa si te dan un número gigante como el $45.892$? No vas a dibujar 45.892 bolitas para hacer parejas, gastariamos mucho tiempo.
		\\[0.2cm]
		Para saber si un número grande es Par o Impar, el truco es \textbf{mirar SOLAMENTE el último número} (la cola o las \textbf{Unidades}). Si el último número es par, ¡todo el número gigante es par!
	\end{concepto}
	
	\begin{ejemplo}
		\begin{itemize}
			\item Observa el número $3.45\mathbf{\textcolor{valred}{7}}$. Termina en \textbf{7}. Como el 7 es Impar, \textbf{todo el número es Impar}.
			\item Observa el número $12.89\mathbf{\textcolor{valblue}{4}}$. Termina en \textbf{4}. Como el 4 es Par, \textbf{todo el número es Par}.
		\end{itemize}
	\end{ejemplo}
	\newpage
	\begin{practica}{Mirando la unidad}
		\textit{Subraya con tu lápiz el último número de cada cifra \textbf{(Las unidades)}. Luego, escribe si el número completo es \textbf{PAR} o \textbf{IMPAR}.}
		
		\vspace{0.4cm}
		\begin{multicols}{2}
			\raggedcolumns
			\begin{enumerate}[itemsep=1cm]
				\item $45 \rightarrow$ \boxfill[2.5cm]
				\item $128 \rightarrow$ \boxfill[2.5cm]
				\item $301 \rightarrow$ \boxfill[2.5cm]
				\item $890 \rightarrow$ \boxfill[2.5cm]
				\item $4.567 \rightarrow$ \boxfill[2.5cm]
				\item $10.002 \rightarrow$ \boxfill[2.5cm]
				\item $35.889 \rightarrow$ \boxfill[2.5cm]
				\item $100.000 \rightarrow$ \boxfill[2.5cm]
			\end{enumerate}
		\end{multicols}
		\vspace{0.2cm}
	\end{practica}

	\section*{Misión 3: Sumando pares e impares}
	
	\begin{concepto}{¿Qué pasa cuando los sumamos?}
		\begin{itemize}
			\item Si sumas dos números PARES, el resultado siempre será \textbf{PAR} (Ejemplo: $2 + 2 = 4$).
			\item Si sumas dos números IMPARES, se juntan los que sobraban y forman una nueva pareja. ¡El resultado será \textbf{PAR}! (Ejemplo: $3 + 3 = 6$).
			\item Si sumas un PAR con un IMPAR, uno se queda solo. ¡El resultado será \textbf{IMPAR}! (Ejemplo: $2 + 3 = 5$).
		\end{itemize}
	\end{concepto}
	
	\begin{practica}{La suma es \textbf{PAR} o \textbf{IMPAR}}
		\textit{Resuelve la suma6 y escribe su resultado en la cajita gris. Luego, marca con una \textbf{X} si ese resultado final es Par o Impar.}
		
		\vspace{0.6cm}
		\begin{enumerate}[itemsep=1.2cm]
			\item $10 + 2 = $ \boxfill[1cm] \quad $\rightarrow$ \quad $\square$ Par \hspace{0.5cm} $\square$ Impar
			
			\item $7 + 3 = $ \boxfill[1cm] \quad $\rightarrow$ \quad $\square$ Par \hspace{0.5cm} $\square$ Impar
			
			\item $8 + 5 = $ \boxfill[1cm] \quad $\rightarrow$ \quad $\square$ Par \hspace{0.5cm} $\square$ Impar
			
			\item $12 + 4 = $ \boxfill[1cm] \quad $\rightarrow$ \quad $\square$ Par \hspace{0.5cm} $\square$ Impar
			
			\item $9 + 9 = $ \boxfill[1cm] \quad $\rightarrow$ \quad $\square$ Par \hspace{0.5cm} $\square$ Impar
			
			\item $15 + 4 = $ \boxfill[1cm] \quad $\rightarrow$ \quad $\square$ Par \hspace{0.5cm} $\square$ Impar
			
			\item $20 + 10 = $ \boxfill[1cm] \quad $\rightarrow$ \quad $\square$ Par \hspace{0.5cm} $\square$ Impar
			
			\item $11 + 2 = $ \boxfill[1cm] \quad $\rightarrow$ \quad $\square$ Par \hspace{0.5cm} $\square$ Impar
		\end{enumerate}
		\vspace{0.2cm}
	\end{practica}

	\section*{Misión 4: Detectives de la Vida Real}
	
	\begin{truco}{Repartir entre Dos}
		En la vida diaria usamos estos números todo el tiempo. La regla más importante de la vida real es: \textbf{Solo las cantidades PARES se pueden repartir exactamente entre 2 personas sin tener que partir ni romper nada.} Si la cantidad es Impar, siempre le sobrará a alguien o habrá que cortarlo.
	\end{truco}
