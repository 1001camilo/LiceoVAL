	% ══════════════════════════════════════════════════════════════════════════
	%  MISIÓN 1 – ¿Qué es Dividir?
	% ══════════════════════════════════════════════════════════════════════════
	\mision{valblue}{Misión 1: Dividir es Repartir en Partes Iguales}
	
	\begin{concepto}{$\square$ Aprende el Concepto}
		Dividir es \textbf{repartir} una cantidad en \textbf{grupos iguales}.
		Cuando repartes, debes asegurarte de que todos los grupos reciban lo mismo.
		
		\vspace{8pt}
		\begin{center}
			\begin{tikzpicture}[x=0.9cm,y=0.9cm]
				% 12 circles in 4 groups of 3
				% Group 1 (blue box)
				\draw[cajaz,line width=1.5pt] (-0.4,-0.4) rectangle (2.0,1.1);
				\node[font=\LARGE,color=valblue] at (0.2,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (0.9,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (1.6,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (0.3,0.1) {};
				\node[font=\bfseries\small,color=valblue] at (0.8,-0.8) {Grupo 1};
				
				% Group 2
				\draw[cajaz,line width=1.5pt] (2.5,-0.4) rectangle (4.9,1.1);
				\node[font=\LARGE,color=valblue] at (3.1,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (3.8,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (4.5,0.7) {\textbullet};
				\node[font=\bfseries\small,color=valblue] at (3.7,-0.8) {Grupo 2};
				
				% Group 3
				\draw[cajaz,line width=1.5pt] (5.4,-0.4) rectangle (7.8,1.1);
				\node[font=\LARGE,color=valblue] at (6,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (6.7,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (7.4,0.7) {\textbullet};
				\node[font=\bfseries\small,color=valblue] at (6.6,-0.8) {Grupo 3};
				
				% Group 4
				\draw[cajaz,line width=1.5pt] (8.3,-0.4) rectangle (10.7,1.1);
				\node[font=\LARGE,color=valblue] at (8.9,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (9.5,0.7) {\textbullet};
				\node[font=\LARGE,color=valblue] at (10.1,0.7) {\textbullet};
				\node[font=\bfseries\small,color=valblue] at (9.5,-0.8) {Grupo 4};
				
				% Arrow + equation
				\draw[flechaop,valdark] (10.7,0.35) -- (13,0.35);
				\node[cajao,inner sep=6pt,font=\bfseries\LARGE] at (15,0.35)
				{$12 \div 4 = 3$};
				\node[font=\bfseries\normalsize,color=valdark] at (15,-0.6)
				{12 entre 4 grupos $=$ 3 en cada grupo};
			\end{tikzpicture}
		\end{center}
		
		\vspace{6pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm,y=1cm]
				\node[cajaz,minimum width=14cm,minimum height=1.0cm,
				font=\bfseries\large,inner sep=8pt] at (0,0)
				{El signo de dividir es: $\div$ \quad $4\div 12$ Se lee: ``12 dividido entre 4''};
			\end{tikzpicture}
		\end{center}
	\end{concepto}
	
	\vspace{4pt}
	
	\begin{truco}{¡Dividir y Multiplicar son Operaciones Hermanas!}
		Si sabes multiplicar, ¡ya sabes dividir!
		Cuando multiplicas $3 \times 4 = 12$, puedes obtener \textbf{dos divisiones} de regalo:
		
		\vspace{6pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm,y=1cm]
				% Multiplication center
				\node[cajaz,minimum width=3.5cm,minimum height=1.1cm,
				font=\bfseries\LARGE,inner sep=8pt] at (0,0)
				{$3 \times 4 = 12$};
				
				% Two arrows going down-left and down-right
				\draw[flechaop,valgreen,line width=2pt] (-0.8,-0.65) -- (-2.5,-1.6);
				\draw[flechaop,valorange,line width=2pt] (0.8,-0.65) -- (2.5,-1.6);
				
				% Left division
				\node[cajag,minimum width=3.5cm,minimum height=1.0cm,
				font=\bfseries\Large,inner sep=6pt] at (-3.5,-2.2)
				{$12 \div 4 = 3$};
				
				% Right division
				\node[cajao,minimum width=3.5cm,minimum height=1.0cm,
				font=\bfseries\Large,inner sep=6pt] at (3.5,-2.2)
				{$12 \div 3 = 4$};
			\end{tikzpicture}
		\end{center}
	\end{truco}
	
	\vspace{4pt}
	
	\begin{practica}{$\square$ ¡Tu Turno! --- Usa la multiplicación para dividir}
		\textit{Completa las familias de operaciones. Usa lo que ya sabes de multiplicar.}
		
		\vspace{8pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm,y=1cm]
				% Familia 1
				\node[cajaz,inner sep=5pt,font=\bfseries\normalsize] at (0,0)
				{Familia 1};
				\node[font=\bfseries\large,color=valdark] at (0,-0.8)
				{$5 \times 3 = 15$};
				\draw[valblue,line width=1pt] (-1.3,-1.6) rectangle (1.3,-0.95);
				\node[font=\bfseries\large,color=valdark] at (0,-1.28) {$15 \div 3 =$};
				\draw[valblue,line width=1pt] (-1.3,-2.3) rectangle (1.3,-1.65);
				\node[font=\bfseries\large,color=valdark] at (0,-1.98) {$15 \div 5 =$};
				
				% Familia 2
				\node[cajag,inner sep=5pt,font=\bfseries\normalsize] at (5.0,0)
				{Familia 2};
				\node[font=\bfseries\large,color=valdark] at (5.0,-0.8)
				{$4 \times 6 = 24$};
				\draw[valblue,line width=1pt] (3.7,-1.6) rectangle (6.3,-0.95);
				\node[font=\bfseries\large,color=valdark] at (5.0,-1.28) {$24 \div 6 =$};
				\draw[valblue,line width=1pt] (3.7,-2.3) rectangle (6.3,-1.65);
				\node[font=\bfseries\large,color=valdark] at (5.0,-1.98) {$24 \div 4 =$};
				
				% Familia 3
				\node[cajao,inner sep=5pt,font=\bfseries\normalsize] at (10.0,0)
				{Familia 3};
				\node[font=\bfseries\large,color=valdark] at (10.0,-0.8)
				{$7 \times 3 = 21$};
				\draw[valblue,line width=1pt] (8.7,-1.6) rectangle (11.3,-0.95);
				\node[font=\bfseries\large,color=valdark] at (10.0,-1.28) {$21 \div 3 =$};
				\draw[valblue,line width=1pt] (8.7,-2.3) rectangle (11.3,-1.65);
				\node[font=\bfseries\large,color=valdark] at (10.0,-1.98) {$21 \div 7 =$};
			\end{tikzpicture}
		\end{center}
	\end{practica}

	\mision{valblue}{Misión 2: Las Partes de la División}
	
	\begin{concepto}{$\square$ Aprende el Concepto: ¿Cómo se llaman?}
		Una división tiene \textbf{4 partes} con nombre propio:
		
		\vspace{10pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm,y=1cm]
				%── Caja principal: 25 ÷ 4 = 6 R 1 ─────────────────────
				% Dividendo
				\node[btnazul,minimum width=1.6cm,minimum height=1.4cm,
				font=\bfseries\Huge] at (0,0) {25};
				% Signo ÷
				\node[font=\bfseries\Huge,color=valdark] at (2.0,0) {$\div$};
				% Divisor
				\node[btnorange,minimum width=1.6cm,minimum height=1.4cm,
				font=\bfseries\Huge] at (4.0,0) {4};
				% Signo =
				\node[font=\bfseries\Huge,color=valdark] at (6.0,0) {$=$};
				% Cociente
				\node[btnverde,minimum width=1.6cm,minimum height=1.4cm,
				font=\bfseries\Huge] at (8.0,0) {6};
				% R (residuo label)
				\node[font=\bfseries\Large,color=valred] at (9.8,0) {R};
				% Residuo
				\node[btnrojo,minimum width=1.2cm,minimum height=1.4cm,
				font=\bfseries\Huge] at (11.2,0) {1};
				
				%── Flechas y etiquetas ─────────────────────────────────
				\draw[flechaop,valblue,line width=1.5pt] (0,-0.85) -- (0,-1.7);
				\node[cajaz,inner sep=5pt,font=\bfseries\normalsize] at (0,-2.25)
				{DIVIDENDO};
				\node[font=\normalsize,color=valdark,align=center,text width=3.2cm]
				at (0,-3.1) {El número que\\ se va a dividir};
				
				\draw[flechaop,valorange,line width=1.5pt] (4,-0.85) -- (4,-1.7);
				\node[cajao,inner sep=5pt,font=\bfseries\normalsize] at (4,-2.25)
				{DIVISOR};
				\node[font=\normalsize,color=valdark,align=center,text width=3.2cm]
				at (4,-3.1) {¿En cuántos\\ grupos?};
				
				\draw[flechaop,valgreen,line width=1.5pt] (8,-0.85) -- (8,-1.7);
				\node[cajag,inner sep=5pt,font=\bfseries\normalsize] at (8,-2.25)
				{COCIENTE};
				\node[font=\normalsize,color=valdark,align=center,text width=3.2cm]
				at (8,-3.1) {El resultado:\\ cuántos hay en\\ cada grupo};
				
				\draw[flechaop,valred,line width=1.5pt] (11.2,-0.85) -- (11.2,-1.7);
				\node[cajar,inner sep=5pt,font=\bfseries\normalsize] at (11.2,-2.25)
				{RESIDUO};
				\node[font=\normalsize,color=valdark,align=center,text width=2.8cm]
				at (11.2,-3.1) {Lo que sobra\\ al final};
			\end{tikzpicture}
		\end{center}
	\end{concepto}
	
	\vspace{4pt}
	
	\begin{practica}{$\square$ ¡Tu Turno! --- Identifica las Partes}

		\begin{center}

				%── División 1: 18 ÷ 3 = 6 R 0 ──────────────────────────

				\node[draw=valdark,fill=white,minimum width=1.1cm,minimum height=0.9cm,
				font=\bfseries\LARGE] at (3.5,0) {3};
				\node[font=\bfseries\LARGE,color=valdark] at (4.7,0) {$=$};
				\node[draw=valdark,fill=white,minimum width=1.1cm,minimum height=0.9cm,
				font=\bfseries\LARGE] at (5.9,0) {6};
				\node[font=\bfseries\large,color=valred] at (7.0,0) {R};
				\node[draw=valdark,fill=vallightred,minimum width=1.0cm,minimum height=0.9cm,
				font=\bfseries\LARGE] at (7.8,0) {0};
				% Blank lines for labels

				\node[draw=valdark,fill=white,minimum width=1.1cm,minimum height=0.9cm,
				font=\bfseries\LARGE] at (0.7,-2.0) {29};
				\node[font=\bfseries\LARGE,color=valdark] at (2.1,-2.0) {$\div$};
				\node[draw=valdark,fill=white,minimum width=1.1cm,minimum height=0.9cm,

		\vspace{8pt}
		\textit{Completa las oraciones:}
