	%  MISIÓN 1 – ¿Qué es Dividir?
	% ══════════════════════════════════════════════════════════════════════════
	\mision{valblue}{Misión 1: Dividir es Repartir en Partes Iguales}
	
	\begin{concepto}{$\square$ Aprende el Concepto}
		Dividir es \textbf{repartir} una cantidad en \textbf{grupos iguales}.
		Cuando repartes, debes asegurarte de que todos los grupos reciban lo mismo.\\
		\textbf{Ejemplo:} Si tenemos 12 canicas y queremos organizarlas en 4 cajas para que cada caja tenga la misma cantidad de canicas, tenemos que hacer una división, en este caso es $12\div4$.
		
		%\vspace{8pt}
		\begin{center}
			\begin{tikzpicture}[x=0.9cm,y=0.9cm]
				% 12 circles in 4 groups of 3
				% Group 1 (blue box)

				\node[font=\LARGE,color=valblue] at (0.2,2.35) {\textbullet};
				\node[font=\LARGE,color=valblue] at (0.9,2.35) {\textbullet};
				\node[font=\LARGE,color=valblue] at (1.6,2.35) {\textbullet};
				\node[font=\LARGE,color=valblue] at (0.3,3) {};
				\node[font=\bfseries\small,color=valblue] at (0.8,1.2) {Grupo 1};
				
				% Group 2
				\draw[cajaz,line width=1.5pt] (2.5,1.6) rectangle (4.9,3.1);

	\end{concepto}

	\newpage
	
	\begin{truco}{¡Dividir y Multiplicar son Operaciones Hermanas!}
		Si sabes multiplicar, ¡ya sabes dividir!
		Cuando multiplicas $3 \times 4 = 12$, puedes obtener \textbf{dos divisiones}, ya que la división es la operación contraria, cuando pienses en dividir, puedes pensar en una multiplicación, por ejemplo al $12\div4$ te puedes preguntar por que número multiplico a $4$ para que de $12$:
		
		%\vspace{6pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm,y=1cm]
				% Multiplication center
				\node[cajaz,minimum width=3.5cm,minimum height=1.1cm,
				font=\bfseries\LARGE,inner sep=8pt] at (0,0)
				{$3 \times 4 = 12$};
				
				% Two arrows going down-left and down-right
				\draw[flechaop,valgreen,line width=2pt] (-0.8,-0.65) -- (-2.5,-1.6);

				% Left division
				\node[cajag,minimum width=3.5cm,minimum height=1.0cm,

		\end{center}
	\end{truco}
