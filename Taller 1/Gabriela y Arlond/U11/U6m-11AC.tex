	\begin{concepto}{$\square$ Aprende el Concepto}
		
		% TEXTO: Definición de combinación para AC (lenguaje simple)
		Una \textbf{combinación} ocurre cuando elegimos \textbf{una cosa de
			un grupo} y la unimos con \textbf{una cosa de otro grupo}.
		
		\vspace{6pt}
		\textit{\textbf{Ejemplo:} Tienes 2 camisas y 2 pantalones.
			Cada camisa se puede usar con cada pantalón.
			{¿}Cuántos atuendos distintos puedes hacer?}
		
		\vspace{8pt}
		
		% ─────────────────────────────────────────────────────────────────────
		% DIAGRAMA: Cuadrícula 2×2 de combinaciones ropa.
		% EDITAR: Cambia los textos de los \node para otro ejemplo.

		%   Eje Y (filas)    = camisas:     "Roja"  y "Azul"

	\mision{valorange}{Misión 3: El Principio de Multiplicación}
	
	\begin{truco}{{¡}La Regla de Oro para Contar Combinaciones!}
		
		% TEXTO: Regla del principio multiplicativo
		Si tienes \textbf{A} opciones del Grupo 1 y \textbf{B} opciones del
		Grupo 2, el total de combinaciones es:
		
		%\vspace{8pt}
		\begin{center}
			\begin{tikzpicture}[x=1cm, y=1cm]
				% Caja A
				\node[cajag, minimum width=3cm, minimum height=1.1cm,
				font=\bfseries\LARGE, inner sep=8pt] at (0,0) {A};
				\node[font=\bfseries\small, color=valgreen] at (0,-0.88)
				{opciones Grupo 1};
				
				% Signo multiplicación
				\node[font=\bfseries\Huge, color=valdark] at (2.5,0) {$\times$};
				
				% Caja B
				\node[cajaz, minimum width=3cm, minimum height=1.1cm,
				font=\bfseries\LARGE, inner sep=8pt] at (5.0,0) {B};
				\node[font=\bfseries\small, color=valblue] at (5.0,-0.88)
				{opciones Grupo 2};
				
				% Signo igual
				\node[font=\bfseries\Huge, color=valdark] at (7.5,0) {$=$};
				
				% Caja resultado
				\node[cajar, minimum width=3.5cm, minimum height=1.1cm,
				font=\bfseries\LARGE, inner sep=8pt] at (10.5,0) {A $\times$ B};
				\node[font=\bfseries\small, color=valred] at (10.5,-0.88)
				{combinaciones totales};
			\end{tikzpicture}
		\end{center}
		
		%\vspace{6pt}
		% TEXTO: Ejemplo aplicado
		\textbf{Aplicando al ejemplo del helado:}
		\quad 3 sabores $\times$ 2 tama\~nos $= 6$ helados distintos.
	\end{truco}

			NO necesitas dibujar el árbol.}

		\begin{enumerate}[itemsep=0.8cm, label=\textbf{\arabic*.}]

			{¿}Cu\'antas combinaciones hay?\\[3pt]
			Grupo A $=$ \boxfill[1.5cm] \quad
			Grupo B $=$ \boxfill[1.5cm] \quad
			Total $=$ \boxfill[2.5cm]

			A $=$ \boxfill[1.2cm] \;
			B $=$ \boxfill[1.2cm] \;
			C $=$ \boxfill[1.2cm] \;
			Total $=$ A$\times$B$\times$C $=$ \boxfill[2.5cm]
		\end{enumerate}
	\end{practica}
