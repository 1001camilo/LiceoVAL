\begin{mision}{valorange}{\ding{72}\; Misión 3: ¿Cuánto cuesta todo?
  --- El Total}

\begin{aprender}{¿Qué es el TOTAL?}
El \textbf{TOTAL} es lo que tienes que pagar por todos los productos.\\[0.2cm]
Para encontrar el total:\\[0.2cm]
\begin{center}
  \Large
  \textbf{SUMA todos los precios en la calculadora}
\end{center}
\end{aprender}

\vspace{0.3cm}

\begin{paso}{PASO A PASO con la calculadora}
\begin{enumerate}[label=\textbf{Paso \arabic*.}, leftmargin=2cm,
    itemsep=0.2cm]
  \item Escribe el precio del primer producto.
  \item Presiona \textbf{[ + ]}.
  \item Escribe el precio del segundo producto.
  \item Presiona \textbf{[ + ]}.
  \item Sigue con todos los productos.
  \item Al final presiona \textbf{[ = ]}.
  \item El número que aparece es el \textbf{TOTAL}.
\end{enumerate}
\end{paso}

\begin{mision}{valred}{\ding{72}\; Misión 5: Las Vueltas}

\begin{aprender}{¿Qué son las Vueltas?}
Cuando pagas con un billete y el total es \textbf{menor},\\
la tienda te devuelve dinero.\\[0.2cm]
Ese dinero que te devuelven se llama \textbf{VUELTAS} o \textbf{CAMBIO}.\\[0.3cm]
\begin{center}
  \begin{tcolorbox}[colback=valred!10, colframe=valred,
      arc=3mm, boxrule=2pt, width=0.8\linewidth, halign=center]
    {\LARGE\bfseries
      VUELTAS = Billete que diste $-$ Total}
  \end{tcolorbox}
\end{center}
\end{aprender}

\vspace{0.3cm}

\begin{paso}{PASO A PASO para calcular las Vueltas}
\begin{enumerate}[label=\textbf{Paso \arabic*.}, leftmargin=2cm,
    itemsep=0.25cm]
  \item Escribe el valor del billete que entregaste.
  \item Presiona \textbf{[ $-$ ]}.
  \item Escribe el \textbf{TOTAL} de la compra.
  \item Presiona \textbf{[ = ]}.
  \item El número que aparece es tus \textbf{VUELTAS}.
\end{enumerate}
\end{paso}
