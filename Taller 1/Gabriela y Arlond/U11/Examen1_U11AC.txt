\newcommand{\billete}[2]{%

			\centering

				\tcbox[colback=white,colframe=black!40,arc=2mm]{\small Billete (Frente)}\\[2pt]%
			}%
			{\tiny\textcolor{black!70}{\textsc{Frente}}}

				\tcbox[colback=white,colframe=black!40,arc=2mm]{\small Billete (Reverso)}\\[2pt]%
