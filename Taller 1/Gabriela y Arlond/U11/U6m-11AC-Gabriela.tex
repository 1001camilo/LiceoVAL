	\end{tcolorbox}

	\vspace{0.4cm}
	\begin{enumerate}[label=\textbf{\arabic*.}, itemsep=0.8cm]
		\item Compro un \textbf{Sándwich de \$ 4.000} y una \textbf{Gaseosa de \$ 2.500}. \\
		Mi operación en la calculadora: $4000 + 2500 = $ \\
		\textbf{Total a pagar: \$} \boxfill[3cm]
		
		\item Compro un \textbf{Pan de \$ 1.500} y un \textbf{Jugo de \$ 2.000}. \\
		Voy a pagar con un \billete{2000}, una \moneda{1000} y una \moneda{500}. \\
		Mi operación en la calculadora: $2000 + 1000 + 500 = $ \\
		\textbf{Total a pagar: \$} \boxfill[3cm]
		
		\item Compro un \textbf{Cuaderno de \$ 8.500} y un \textbf{Lápiz de \$ 1.200}. \\
		Mi operación en la calculadora: \boxfill[2cm] $+$ \boxfill[2cm] $=$ \\
		\textbf{Total a pagar: \$} \boxfill[3cm]
	\end{enumerate}

	% MISIÓN 4
	% ---------------------------------------------------------
	\begin{tcolorbox}[colback=valdark!10, colframe=valdark, title=\textbf{Misión 4: Mis Vueltas (Lo que me sobra)}]
		Si te alcanzó para pagar, el cajero te debe devolver dinero. Eso se llama \textbf{vueltas}. Para saber de cuánto son tus vueltas, debes \textbf{RESTAR} en tu calculadora. \\
		\textbf{Fórmula: (Valor de tu Dinero) $-$ (Total a Pagar) = Vueltas.}
	\end{tcolorbox}
	\vspace{0.2cm}
	
	\noindent \textit{Ejemplo: Pago con \$10.000. El total es \$8.000. \\
		Mi calculadora: 10000 $-$ 8000 = \textbf{2.000 de vueltas.}}
	
	\vspace{0.4cm}
	\begin{enumerate}[label=\textbf{\arabic*.}, itemsep=0.8cm]
		\item Compro un combo de \textbf{\$ 15.000}. Pago con un \billete{20000}. \\
		Mi operación en la calculadora: $20000 - 15000 = $ \\
		\textbf{Me dan de vueltas: \$} \boxfill[3cm]
		
		\item Compro un dulce de \textbf{\$ 500}. Pago con una \moneda{1000}. \\
		Mi operación en la calculadora: $1000 - 500 = $ \\
		\textbf{Me dan de vueltas: \$} \boxfill[3cm]
	\end{enumerate}
