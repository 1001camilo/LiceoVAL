	\end{tcolorbox}
	\vspace{0.2cm}
	
	\noindent \textit{Ejemplo: Jugo (\$ 2.000) + Galletas (\$ 1.500) = \textbf{Total: \$ 3.500}}
	
	\vspace{0.4cm}
	\begin{enumerate}[label=\textbf{\arabic*.}, itemsep=0.6cm]
		\item Compro un \textbf{Sándwich de \$ 4.000} y una \textbf{Gaseosa de \$ 2.500}. \\
		Mi operación en la calculadora: $4000 + 2500 = $ \\
		\textbf{Total a pagar: \$} \boxfill[2.5cm]
		
		\item Compro un \textbf{Pan de \$ 1.500} y un \textbf{Jugo de \$ 2.000}. \\
		Voy a pagar con un billete de 2.000, una moneda de 1.000 y una de 500. \\
		Mi operación: $2000 + 1000 + 500 = $ \\
		\textbf{Total a pagar: \$} \boxfill[2.5cm]
	\end{enumerate}

	\begin{tcolorbox}[colback=valdark!10, colframe=valdark, title=\textbf{Misión 4: Mis Vueltas (Lo que me sobra)}]
		Si te alcanzó para pagar, el cajero te debe devolver dinero. Eso se llama \textbf{vueltas}. Para saber de cuánto son tus vueltas, debes \textbf{RESTAR} en tu calculadora. \\
		\textbf{Fórmula: (Valor de tu Dinero) $-$ (Total a Pagar) = Vueltas.}
	\end{tcolorbox}
	\vspace{0.2cm}
	
	\noindent \textit{Ejemplo: Pago con \$10.000. El total es \$9.150. \\
		Mi calculadora: 10000 $-$ 9150 = \textbf{850 de vueltas (¡Te darán monedas!).}}
	
	\vspace{0.4cm}
	\begin{enumerate}[label=\textbf{\arabic*.}, itemsep=0.8cm]
		\item Compro un combo de \textbf{\$ 14.150}. Pago con un billete de 20.000. \\
		Mi operación en la calculadora: $20000 - 14150 = $ \\
		\textbf{Me dan de vueltas: \$} \boxfill[3cm]
		
		\item Compro un dulce de \textbf{\$ 350}. Pago con una moneda de 1.000. \\
		Mi operación en la calculadora: $1000 - 350 = $ \\
		\textbf{Me dan de vueltas: \$} \boxfill[3cm]
	\end{enumerate}
