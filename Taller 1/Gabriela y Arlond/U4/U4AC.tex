	\section*{Misión 1: La Magia de Quitar}
	
	\begin{concepto}{Restar es buscar la diferencia}
		Restar (o sustraer) significa \textbf{quitar} una cantidad a otra que ya teníamos, para saber cuánto nos queda al final. También sirve para saber qué diferencia hay entre dos números.
		\\[0.2cm]
		Toda resta tiene tres partes importantes que debes conocer:
		\begin{itemize}
			\item \textbf{Minuendo:} Es el número gigante, la cantidad que tienes al principio. \textbf{¡Siempre va arriba!}
			\item \textbf{Sustraendo:} Es el número que vas a quitar, gastar o regalar. \textbf{¡Va abajo!}
			\item \textbf{Diferencia:} Es el resultado final, lo que te sobra.
		\end{itemize}
	\end{concepto}
	
	\vspace{0.2cm}
	\begin{center}
		\begin{tikzpicture}[scale=1.2]
			\node[font=\Huge\bfseries] (min) at (0, 1.5) {9};
			\node[font=\Huge\bfseries] (sus) at (0, 0.5) {4};
			\node[font=\Huge\bfseries] (dif) at (0, -0.7) {5};
			\node[font=\Huge\bfseries] at (-1, 0.5) {$-$};
			\draw[ultra thick] (-1.5, 0) -- (1, 0);
			
			% Flechas y nombres
			\draw[->, thick, valblue] (0.5, 1.5) -- (2, 1.5) node[right, text=valblue] {\Large \textbf{Minuendo} (El grande)};
			\draw[->, thick, valred] (0.5, 0.5) -- (2, 0.5) node[right, text=valred] {\Large \textbf{Sustraendo} (Lo que quito)};
			\draw[->, thick, valgreen] (0.5, -0.7) -- (2, -0.7) node[right, text=valgreen] {\Large \textbf{Diferencia} (El resultado)};
		\end{tikzpicture}
	\end{center}

	\section*{Misión 2: El Edificio de la Resta}
	
	\begin{concepto}{El Secreto de las Columnas}
		Para restar números más grandes, los organizamos hacia abajo como si fueran pisos de un edificio. \textbf{Es obligatorio} que las unidades queden debajo de las unidades, y las decenas debajo de las decenas.
		\\[0.2cm]
		Como estamos haciendo restas \textbf{"Sin Prestar"}, te darás cuenta de que el número de arriba siempre es igual o más grande que el de abajo.
	\end{concepto}
	
	\begin{ejemplo}
		\begin{multicols}{2}
			\textbf{Tengo 78 galletas y me como 25.}
			\\ ¿Cuántas me quedan?
			
			\vspace{0.4cm}
			\textbf{Pasos:}
			\begin{enumerate}
				\item Pongo el más grande arriba (78).
				\item Pongo el pequeño abajo (25).
				\item Resto primero las Unidades (8 - 5 = 3).
				\item Resto luego las Decenas (7 - 2 = 5).
			\end{enumerate}
			
			\columnbreak
			
			\begin{center}
				\Large
				\begin{tabular}{r c c}
					& \textbf{D} & \textbf{U} \\
					& 7 & 8 \\
					$-$ & 2 & 5 \\
					\hline
					& \textbf{5} & \textbf{3}
				\end{tabular}
				\\[0.4cm]
				\normalsize \textbf{¡Me quedan 53 galletas!}
			\end{center}
		\end{multicols}
	\end{ejemplo}
	
	\begin{practica}{Restas de 2 y 3 cifras (8 Ejercicios)}

	\section*{Misión 3: Restas de Gigantes}
	
	\begin{concepto}{Muchos números, el mismo proceso}
		No importa si el número tiene 4, 5 o 6 cifras. El proceso sigue siendo el mismo: empieza por las Unidades (derecha), luego las Decenas, y así sucesivamente. No olvides poner el punto que indica "mil" en tu respuesta.
	\end{concepto}
	\newpage
	\begin{practica}{Restas de 4 y 5 cifras }
		\textit{Resta columna por columna. ¡Asegúrate de bajar el punto de los miles en el mismo lugar!}

	\section*{Misión 4: Detectives de Problemas}
	
	\begin{truco}{Las Palabras Clave de la Resta}
		Cuando leas un problema, ¿cómo saber que tienes que restar? Busca estas palabras clave: \textbf{Quedan, Sobran, Gastó, Perdió, Comió, Regaló, Faltan, o ¿Cuál es la Diferencia?}
		\\[0.2cm]
		Recuerda: En la resta, \textbf{el número más grande siempre va arriba}.
	\end{truco}
	\newpage
	\begin{practica}{Resuelve los casos}
		\textit{Lee el caso. Escribe la resta (de forma vertical) en el espacio a la izquierda, y anota el resultado en la respuesta a la derecha.}
		
		\vspace{0.4cm}
		\begin{enumerate}[itemsep=1.5cm]
			\item Tenía \textbf{\$ 5.800} en mi bolsillo y gasté \textbf{\$ 2.500} en un jugo. ¿Cuánto dinero me queda? \\
			\begin{tabular}{c c}
				\makebox[5cm]{ \boxfill[3cm] } & \textbf{Respuesta:} Me quedan \$ \boxfill[1.5cm]
			\end{tabular}
			
			\item Un libro tiene \textbf{458} páginas. Si ya leí \textbf{123} páginas, ¿cuántas páginas me faltan por leer? \\
			\begin{tabular}{c c}
				\makebox[5cm]{ \boxfill[3cm] } & \textbf{Respuesta:} Me faltan \boxfill[1.5cm] páginas.
			\end{tabular}
			
			\item En una granja hay \textbf{875} gallinas, y el granjero vende \textbf{341} de ellas. ¿Cuántas gallinas sobran en la granja? \\
			\begin{tabular}{c c}
				\makebox[5cm]{ \boxfill[3cm] } & \textbf{Respuesta:} Sobran \boxfill[1.5cm] gallinas.
			\end{tabular}
			
			\item Mi mamá me dio \textbf{\$ 8.500}. Fui a la papelería y compré un lápiz de \textbf{\$ 3.200}. ¿Cuánto dinero me sobró? \\
			\begin{tabular}{c c}
				\makebox[5cm]{ \boxfill[3cm] } & \textbf{Respuesta:} Me sobraron \$ \boxfill[1.5cm]
			\end{tabular}
			
			\item En un estadio caben \textbf{9.850} personas. Si ya han entrado \textbf{4.210} personas, ¿cuántos asientos vacíos quedan? \\
			\begin{tabular}{c c}
				\makebox[5cm]{ \boxfill[3cm] } & \textbf{Respuesta:} Quedan \boxfill[1.5cm] asientos vacíos.
			\end{tabular}
			
			\item Una fábrica hizo \textbf{6.785} juguetes y repartió \textbf{2.340} a las tiendas. ¿Cuántos juguetes le quedaron en la fábrica? \\
			\begin{tabular}{c c}
				\makebox[5cm]{ \boxfill[3cm] } & \textbf{Respuesta:} Le quedaron \boxfill[1.5cm] juguetes.
			\end{tabular}
			
			\item El camión repartidor llevaba \textbf{798} cajas de leche. Dejó \textbf{254} cajas en el supermercado. ¿Cuántas cajas le quedan en el camión? \\
			\begin{tabular}{c c}
				\makebox[5cm]{ \boxfill[3cm] } & \textbf{Respuesta:} Le quedan \boxfill[1.5cm] cajas.
			\end{tabular}
			
			\item Un rompecabezas tiene \textbf{5.968} fichas. Si ya hemos armado \textbf{1.453} fichas, ¿cuántas faltan por poner? \\
			\begin{tabular}{c c}
				\makebox[5cm]{ \boxfill[3cm] } & \textbf{Respuesta:} Faltan \boxfill[1.5cm] fichas.
			\end{tabular}
		\end{enumerate}
		\vspace{0.2cm}
	\end{practica}
