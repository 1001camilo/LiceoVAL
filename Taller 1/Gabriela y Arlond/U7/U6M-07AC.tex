	\mision{valblue}{Misión 2: Las Partes de la Multiplicación}
	
	\begin{concepto}{$\square$ Aprende el Concepto: ¿Cómo se llaman?}
		\begin{center}
			\begin{tikzpicture}[x=1cm, y=1cm]
				%── Cajas Factor × Factor = Producto ────────────────
				\node[btnazul, minimum width=2.6cm, minimum height=1.4cm]

				\node[font=\bfseries\Huge, color=valdark] at (2.1, 0) {$\times$};
				
				\node[btnazul, minimum width=2.6cm, minimum height=1.4cm]
				at (4.2, 0) {\Huge 4};
				\node[font=\bfseries\large, color=valblue] at (4.2,-1.1) {FACTOR};
				
				\node[font=\bfseries\Huge, color=valdark] at (6.3, 0) {$=$};
				
				\node[btnrojo, minimum width=2.6cm, minimum height=1.4cm]
				at (8.4, 0) {\Huge 20};
				\node[font=\bfseries\large, color=valred]  at (8.4,-1.1) {PRODUCTO};
				
				%── Regla general ────────────────────────────────────
				\node[cajao, minimum width=11.5cm, minimum height=1.0cm,
				font=\bfseries\large, inner sep=6pt] at (4.2,-2.5)
				{FACTOR \quad $\times$ \quad FACTOR \quad $=$ \quad PRODUCTO};
			\end{tikzpicture}
		\end{center}
		
		\vspace{6pt}
		\begin{itemize}[leftmargin=*, itemsep=2pt]
			\item Los \textbf{\color{valblue}FACTORES} son los números que
			\textbf{se multiplican}.
			\item El \textbf{\color{valred}PRODUCTO} es el \textbf{resultado}
			de la multiplicación.
		\end{itemize}
	\end{concepto}
	
	\vspace{4pt}
