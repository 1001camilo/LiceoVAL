	\section*{Misión 1: ¿Qué es multiplicar?}
	
	\begin{concepto}{Multiplicar es una suma repetida}
		Multiplicar es una forma rápida de sumar el mismo número muchas veces.
		Imagina que tienes 4 cajas y en cada caja hay 3 juguetes.
		Podrías sumar: $3 + 3 + 3 + 3 = 12$.
		Pero es más rápido multiplicar: $3 \times 4 = 12$.
		\begin{itemize}
			\item El signo es una equis ($\times$) o un punto ($\cdot$).
			\item Se lee: ``3 \textbf{por} 4''.
		\end{itemize}
	\end{concepto}
	
	\begin{ejemplo}
		\begin{center}
			\begin{tikzpicture}[scale=1]
				% Grupo 1
				\draw[ultra thick, valblue, rounded corners, fill=valblue!5] (0,0) rectangle (2.5,2);
				\fill[valorange] (0.7,0.6) circle (0.25); \fill[valorange] (1.8,0.6) circle (0.25);
				\fill[valorange] (0.7,1.4) circle (0.25); \fill[valorange] (1.8,1.4) circle (0.25);
				\node[font=\Large\bfseries, valdark] at (1.25, -0.5) {4};
				
				\node[font=\Huge\bfseries, valdark] at (3.25, 1) {$+$};
				
				% Grupo 2
				\draw[ultra thick, valblue, rounded corners, fill=valblue!5] (4,0) rectangle (6.5,2);
				\fill[valorange] (4.7,0.6) circle (0.25); \fill[valorange] (5.8,0.6) circle (0.25);
				\fill[valorange] (4.7,1.4) circle (0.25); \fill[valorange] (5.8,1.4) circle (0.25);
				\node[font=\Large\bfseries, valdark] at (5.25, -0.5) {4};
				
				\node[font=\Huge\bfseries, valdark] at (7.25, 1) {$+$};
				
				% Grupo 3
				\draw[ultra thick, valblue, rounded corners, fill=valblue!5] (8,0) rectangle (10.5,2);
				\fill[valorange] (8.7,0.6) circle (0.25); \fill[valorange] (9.8,0.6) circle (0.25);
				\fill[valorange] (8.7,1.4) circle (0.25); \fill[valorange] (9.8,1.4) circle (0.25);
				\node[font=\Large\bfseries, valdark] at (9.25, -0.5) {4};

	\section*{Misión 2: Multiplicar Hacia Abajo}
	
	\begin{concepto}{El Edificio de la Multiplicación}
		Cuando tenemos un número grande, lo multiplicamos en forma vertical (hacia abajo). El número pequeño de abajo multiplica \textbf{a cada uno} de los números de arriba, empezando \textbf{siempre por la derecha} (Unidades), luego las Decenas.
	\end{concepto}
	
	\begin{ejemplo}
		\begin{multicols}{2}
			\textbf{Queremos multiplicar $23 \times 3$:}\\
			Imagina que cada paquete trae 23 galletas y compramos 3 paquetes.
			
			\vspace{0.4cm}
			\textbf{Pasos:}
			\begin{enumerate}
				\item Multiplica el de abajo por la Unidad: $3 \times 3 = \mathbf{9}$.
				\item Multiplica el de abajo por la Decena: $3 \times 2 = \mathbf{6}$.
			\end{enumerate}
			
			\columnbreak
			
			\begin{center}
				\Large
				\begin{tabular}{r c c}
					& \textbf{D} & \textbf{U} \\
					& 2 & 3 \\
					$\times$ & & \textbf{3} \\
					\hline
					& \textbf{6} & \textbf{9}
				\end{tabular}
				\\[0.4cm]
				\normalsize \textbf{¡En total tenemos 69 galletas!}
			\end{center}
		\end{multicols}
	\end{ejemplo}
	
	\begin{practica}{Multiplicar directo (8 Ejercicios)}
		\textit{Multiplica el número de abajo por cada uno de los números de arriba y anota el resultado abajo. ¡Aquí no tienes que llevar nada!}

	\section*{Misión 3: El Regalo para el Vecino (Llevando)}
	
	\begin{concepto}{Cuando el resultado es muy grande}
		Si al multiplicar una columna el resultado es \textbf{10 o más}, ¡no caben los dos números abajo! Dejas el número de la derecha abajo y \textbf{"llevas"} el otro número hacia arriba de la siguiente columna. ¡Escríbelo en la \textbf{cajita azul}! Luego, \textbf{SE LO SUMAS} al resultado de la siguiente multiplicación.
	\end{concepto}
	
	\begin{ejemplo}
		\begin{multicols}{2}
			\textbf{Queremos multiplicar $14 \times 3$:}\\
			\textbf{Paso 1:} Multiplica $3 \times 4 = 12$. Ponemos el \textbf{2} abajo y \textcolor{valblue}{\textbf{llevamos 1}} a la cajita del vecino.
			
			\vspace{0.2cm}
			\textbf{Paso 2:} Multiplica el de abajo por la Decena: $3 \times 1 = 3$. ¡Y le SUMAMOS el 1 que teníamos guardado en la cajita! $3 + 1 = \mathbf{4}$.
			
			\columnbreak
			
			\begin{center}
				\Large
				\begin{tabular}{r c c}
					& \cajitalleva & \\[-0.5ex]
					& 1 & 4 \\
					$\times$ & & \textbf{3} \\
					\hline
					& \textbf{4} & \textbf{2}
				\end{tabular}
				\\[0.4cm]
				\normalsize \textbf{¡El resultado es 42!}
			\end{center}
		\end{multicols}
	\end{ejemplo}
	
	\begin{practica}{Multiplicar llevando (8 Ejercicios)}

		Cuando leas un problema en la vida real, ¿cómo saber que tienes que multiplicar? Busca estas palabras clave: \textbf{Cada uno}, \textbf{El doble} (multiplicar por 2), \textbf{El triple} (multiplicar por 3), \textbf{Veces}, o cuando te pidan \textbf{un total de muchos objetos exactamente iguales}.
	\end{truco}
