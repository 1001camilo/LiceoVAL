\begin{espaciotrabajonb}{Problema 1 --- El Jardín Escolar}
\begin{minipage}[t]{0.58\linewidth}
  % EDITAR: dimensiones del jardín
  \textit{El jardín escolar tiene una cerca de \textbf{360 cm} de largo.
  El maestro quiere pintar letreros cada \textbf{1 m}.
  {¿}Cuántos letreros puede poner?}

  \vspace{8pt}
  \textbf{Paso 1:} Convierte 360 cm a metros.\\[3pt]
  \hspace{1cm} 360 cm $\div$ 100 $=$ \boxfill[2.5cm] m

  \vspace{5pt}
  \textbf{Paso 2:} Divide entre 1 m para saber cuántos letreros.\\[3pt]
  \hspace{1cm} \boxfill[1.8cm] $\div$ 1 $=$ \boxfill[2.5cm] letreros
\end{minipage}%
\hfill
\begin{minipage}[t]{0.36\linewidth}

\end{minipage}
\end{espaciotrabajonb}

\begin{espaciotrabajonb}{Problema 2 --- La Maratón Escolar}
\begin{minipage}[t]{0.58\linewidth}
  \textit{La ruta de la maratón escolar es de \textbf{5 Km}.
  Juan ya corrió \textbf{2\,300 m}.
  {¿}Cuántos metros le faltan para terminar?}

  \vspace{8pt}
  \textbf{Paso 1:} Convierte 5 Km a metros.\\[3pt]
  \hspace{1cm} 5 Km $\times$ 1\,000 $=$ \boxfill[3cm] m

  \vspace{5pt}
  \textbf{Paso 2:} Resta lo que ya corrió.\\[3pt]
  \hspace{1cm} \boxfill[2.5cm] $-$ 2\,300 $=$ \boxfill[3cm] m
