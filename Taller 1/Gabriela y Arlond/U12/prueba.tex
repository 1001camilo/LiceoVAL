	\begin{concepto}{$\square$ Aprende el Concepto}
		
		La \textbf{longitud} es la medida de qué tan \textbf{largo} es algo:

	\begin{truco}{{¡}Aprende a Leer los Símbolos de las Unidades!}
		
		Cada unidad de longitud tiene un \textbf{símbolo corto} (abreviatura)
		que usamos para escribir más rápido.
		El símbolo \textbf{nunca} lleva punto al final ni va en plural.
		
		\vspace{6pt}
		
		\renewcommand{\arraystretch}{1.6}
		\begin{center}
			\begin{tabular}{|c|l|l|p{5.2cm}|}
				\hline
				\rowcolor{vallightorange}
				\textbf{Símbolo} & \textbf{Nombre} &
				\textbf{Cómo se lee} & \textbf{Para medir\ldots} \\
				\hline
				\rowcolor{vallightteal!50}
				\textbf{km} & kilómetro  & ki-LÓ-me-tro &
				Distancias entre ciudades \\
				\hline
				\rowcolor{valgray}
				\textbf{hm} & hectómetro & hec-TÓ-me-tro &
				Grandes extensiones de tierra \\
				\hline
				\rowcolor{valgray}
				\textbf{dam} & decámetro & de-CÁ-me-tro &
				Estadios, cuadras \\
				\hline
				\rowcolor{vallightblue!80}
				\textbf{m}  & \textbf{metro (BASE)} & ME-tro &
				Habitaciones, personas, puertas \\
				\hline
				\rowcolor{valgray}
				\textbf{dm} & decímetro  & de-CÍ-me-tro &
				Libros, cuadernos, sillas \\
				\hline
				\rowcolor{valgray}
				\textbf{cm} & centímetro & cen-TÍ-me-tro &
				Lápices, manos, fotos \\
				\hline
				\rowcolor{vallightred!50}
				\textbf{mm} & milímetro  & mi-LÍ-me-tro &
				Monedas, granos de arroz, telas \\
				\hline
			\end{tabular}
		\end{center}
		
		\vspace{6pt}
		
		{\bfseries {¿}Qué significa cada prefijo?}

	\begin{practica}{$\square$ {¡}Tu Turno! --- Estima la medida}
		
		\textit{Escoge la medida más razonable para cada objeto.
			Marca con una \textbf{X} la respuesta correcta.}
		
		\vspace{6pt}
		\begin{enumerate}[itemsep=0.7cm, label=\textbf{\arabic*.}]

	\begin{espaciotrabajonb}{Problema 1 --- El Jardín Escolar}
		\begin{minipage}[t]{0.58\linewidth}
			\textit{El jardín escolar tiene una cerca de \textbf{360 cm} de largo.
				El maestro quiere pintar letreros cada \textbf{1 m}.
				{¿}Cuántos letreros puede poner?}
			
			\vspace{8pt}
			\textbf{Paso 1:} Convierte 360 cm a metros.\\[3pt]
			\hspace{1cm} 360 cm $\div$ 100 $=$ \boxfill[2.5cm] m
			
			\vspace{5pt}
			\textbf{Paso 2:} Divide entre 1 m para saber cuántos letreros.\\[3pt]
			\hspace{1cm} \boxfill[1.8cm] $\div$ 1 $=$ \boxfill[2.5cm] letreros
		\end{minipage}%
		\hfill
		\begin{minipage}[t]{0.36\linewidth}
			\textbf{Espacio de trabajo:}\vspace{2.8cm}
		\end{minipage}
	\end{espaciotrabajonb}

	\begin{espaciotrabajonb}{Problema 2 --- La Maratón Escolar}
		\begin{minipage}[t]{0.58\linewidth}
			\textit{La ruta de la maratón escolar es de \textbf{5 km}.
				Juan ya corrió \textbf{2\,300 m}.
				{¿}Cuántos metros le faltan para terminar?}
			
			\vspace{8pt}
			\textbf{Paso 1:} Convierte 5 km a metros.\\[3pt]
			\hspace{1cm} 5 km $\times$ 1\,000 $=$ \boxfill[3cm] m
			
			\vspace{5pt}
			\textbf{Paso 2:} Resta lo que ya corrió.\\[3pt]
			\hspace{1cm} \boxfill[2.5cm] $-$ 2\,300 $=$ \boxfill[3cm] m
		\end{minipage}%
		\hfill
		\begin{minipage}[t]{0.36\linewidth}
			\textbf{Espacio de trabajo:}\vspace{2.8cm}
