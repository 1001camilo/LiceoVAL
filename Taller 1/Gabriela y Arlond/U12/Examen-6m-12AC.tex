	% --- PUNTO 4 ---
	\noindent \textbf{4. Una modista tiene una cinta de tela de 120 cm de largo y necesita cortarla en 3 partes exactamente iguales. ¿Cuántos centímetros medirá cada parte? ¿A cuántos milímetros equivale esa medida final? (25\%)}
	
	\vspace{0.5cm}
	\noindent \textbf{Paso 1:} Divide el largo de la cinta entre 3 partes para encontrar los centímetros.
	\begin{center}
		\boxfill[1.5cm] $\div$ \boxfill[1cm] $=$ \boxfill[2cm] \textbf{cm}
	\end{center}
	
	\vspace{0.3cm}
	\noindent \textbf{Paso 2:} Convierte el resultado de centímetros (cm) a milímetros (mm).
	\begin{center}
		\boxfill[1.5cm] $\times$ \boxfill[1cm] $=$ \boxfill[2cm] \textbf{mm}
	\end{center}
