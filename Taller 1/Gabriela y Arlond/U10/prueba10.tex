\newcommand{\dibujarbarra}[4]{
	% #1=posicion x, #2=altura, #3=color, #4=etiqueta
	\foreach \y in {1,...,#2} {
		\draw[fill=#3, draw=valdark, thick] (#1, \y-1) rectangle (#1+1, \y);
	}
	\node[font=\bfseries\scriptsize, align=center, text width=1.2cm] at (#1+0.5, -0.3) {#4};
}

\newcommand{\dibujarbarrahorizontal}[4]{
	% #1=posicion y, #2=ancho, #3=color, #4=etiqueta
	\foreach \x in {1,...,#2} {
		\draw[fill=#3, draw=valdark, thick] (\x-1, #1) rectangle (\x, #1+1);
	}
	\node[font=\bfseries\scriptsize, left] at (-0.1, #1+0.5) {#4};
}

	\mision{valblue}{Misión 1: ¿Para qué sirve un Gráfico y sus Ejes?}
	
	\begin{concepto}{$\square$ Aprende el Concepto: Las dos líneas importantes}
		Un \textbf{Diagrama de Barras} sirve para organizar información y entenderla rápido, ¡como en una foto! Todo gráfico tiene dos líneas gruesas llamadas \textbf{EJES}:
		\begin{itemize}[leftmargin=*, itemsep=2pt]
			\item La \textbf{línea horizontal} (abajo) tiene palabras y nos dice \textbf{DE QUÉ} hablamos.
			\item La \textbf{línea vertical} (al lado) tiene números y nos dice \textbf{CUÁNTA CANTIDAD} hay.
		\end{itemize}
	\end{concepto}
	
	\titulopractica{$\square$ Práctica: Leer información y contar}
	\textit{Mira las líneas del gráfico con atención para entender de qué trata y responde.}

		\textbf{Interpretar} un gráfico es usar sus datos para responder preguntas y encontrar diferencias. Por ejemplo, en matemáticas, la \textbf{Moda} es el dato que más se repite o más se eligió. ¡Es muy fácil de encontrar porque siempre es la barra más alta!
	\end{concepto}
